5/// <summary>
6/// Utility for encoding, decoding, importing, and exporting <see cref="LevelMap"/> data.
7/// </summary>
8/// <remarks>
9/// @ingroup game_data
10/// @brief Converts level maps to compact shareable strings and reconstructs level maps from those strings.
11///
12/// The exported level format stores level metadata and run-length encoded tile data inside a
13/// Base64-encoded string. Clipboard helpers use <see cref="GUIUtility.systemCopyBuffer"/> for simple
14/// copy/paste sharing.
15///
16/// Encoded data contains:
17/// - Level name.
18/// - Level height and width.
19/// - Circuit flag.
20/// - Start and finish coordinates.
21/// - Lap count.
22/// - Checkpoint count per lap.
23/// - Road tile count.
24/// - Day/night track flag.
25/// - Flattened tile grid encoded with run-length encoding.
26///
27/// Tile values are stored as integers matching <see cref="LevelMap.LevelTileTypes"/>.
28/// </remarks>
29public static class ImportExportManager
30{

64 /// <summary>
65 /// Attempts to decode a shareable level string into a <see cref="LevelMap"/>.
66 /// </summary>
67 /// <param name="encoded">Base64-encoded level string.</param>
68 /// <param name="levelMap">Decoded level map when decoding succeeds; otherwise <c>null</c>.</param>
69 /// <returns><c>true</c> if decoding succeeded; otherwise <c>false</c>.</returns>
70 /// <remarks>
71 /// The method expects a decoded pipe-separated metadata string with thirteen parts. The final part contains
72 /// run-length encoded tile data that is reconstructed into <see cref="LevelMap.Tiles"/>.
73 /// </remarks>
74 public static bool TryDecodeLevel(string encoded, out LevelMap levelMap)
75 {
76 levelMap = null;
77
78 try
79 {
80 string decoded = Encoding.UTF8.GetString(Convert.FromBase64String(encoded));
81 string[] parts = decoded.Split('|');
82
83 if (parts.Length != PartCount)
84 {
85 Debug.LogWarning("[ImportExportManager] Invalid level code format.");
86 return false;
87 }
88
89 if (!int.TryParse(parts[1], out int height) || !int.TryParse(parts[2], out int width) || !int.TryParse(parts[4], out int startX) || !int.TryParse(parts[5], out int startY) || !int.TryParse(parts[6], out int finishX) ||
90 !int.TryParse(parts[7], out int finishY) || !int.TryParse(parts[8], out int laps) || !int.TryParse(parts[9], out int checkpointCountPerLap) || !int.TryParse(parts[10], out int roadTileCount))
91 {
92 Debug.LogWarning("[ImportExportManager] Invalid level metadata number format.");
93 return false;
94 }
95
96 if (width <= 0 || height <= 0)
97 {
98 Debug.LogWarning("[ImportExportManager] Invalid level dimensions.");
99 return false;
100 }
101
102 LevelMap map = new LevelMap();
103 map.Name = Encoding.UTF8.GetString(Convert.FromBase64String(parts[0]));
104 map.Height = height;
105 map.Width = width;
106 map.Circuit = parts[3] == "1";
107 map.StartPoint = new Coordinates(startX, startY);
108 map.FinishPoint = new Coordinates(finishX, finishY);
109 map.Laps = laps;
110 map.CheckpointCountPerLap = checkpointCountPerLap;
111 map.RoadTileCount = roadTileCount;
112 map.IsDayTrack = parts[11] == "1";
113
114 if (!TryDecodeFlatTilesRLE(parts[12], map.Width * map.Height, out int[] flatTiles))
115 {
116 Debug.LogWarning("[ImportExportManager] Failed to decode tile data.");
117 return false;
118 }
119
120 map.Tiles = LevelMap.GetUnflattenedTiles(flatTiles, map.Height, map.Width);
121 levelMap = map;
122 return true;
123 }
124 catch (Exception e)
125 {
126 Debug.LogWarning($"[ImportExportManager] Failed to parse level metadata: {e.Message}");
127 return false;
128 }
129 }

131 /// <summary>
132 /// Encodes a <see cref="LevelMap"/> into a compact shareable string.
133 /// </summary>
134 /// <param name="levelMap">Level map to encode.</param>
135 /// <returns>Base64-encoded level string.</returns>
136 /// <remarks>
137 /// The tile grid is flattened with <see cref="LevelMap.GetFlatTiles"/>, converted to signed bytes,
138 /// encoded with run-length encoding, and then included in the exported metadata string.
139 /// </remarks>
140 public static string EncodeLevel(LevelMap levelMap)
141 {
142 StringBuilder sb = new StringBuilder();
143 int[] flatTiles = levelMap.GetFlatTiles();
144 string tilesRLE = EncodeFlatTilesRLE(ConvertIntArrayToSByteArray(flatTiles));
145
146 sb.Append(Convert.ToBase64String(Encoding.UTF8.GetBytes(levelMap.Name ?? string.Empty)));
147 sb.Append('|');
148 sb.Append(levelMap.Height);
149 sb.Append('|');
150 sb.Append(levelMap.Width);
151 sb.Append('|');
152 sb.Append(levelMap.Circuit ? 1 : 0);
153 sb.Append('|');
154 sb.Append(levelMap.StartPoint.X);
155 sb.Append('|');
156 sb.Append(levelMap.StartPoint.Y);
157 sb.Append('|');
158 sb.Append(levelMap.FinishPoint.X);
159 sb.Append('|');
160 sb.Append(levelMap.FinishPoint.Y);
161 sb.Append('|');
162 sb.Append(levelMap.Laps);
163 sb.Append('|');
164 sb.Append(levelMap.CheckpointCountPerLap);
165 sb.Append('|');
166 sb.Append(levelMap.RoadTileCount);
167 sb.Append('|');
168 sb.Append(levelMap.IsDayTrack ? 1 : 0);
169 sb.Append('|');
170 sb.Append(tilesRLE);
171
172 return Convert.ToBase64String(Encoding.UTF8.GetBytes(sb.ToString()));
173 }

242 /// <summary>
243 /// Encodes a flat tile array with run-length encoding.
244 /// </summary>
245 /// <param name="flatTiles">Flattened tile values to encode.</param>
246 /// <returns>Run-length encoded tile string.</returns>
247 /// <remarks>
248 /// The produced format stores runs as <c>count:symbol</c> pairs separated by semicolons.
249 /// For example, three road tiles followed by two grass tiles would be encoded as
250 /// <c>3:1;2:0</c>.
251 /// </remarks>
252 public static string EncodeFlatTilesRLE(sbyte[] flatTiles)
253 {
254 if (flatTiles == null || flatTiles.Length == 0)
255 {
256 return string.Empty;
257 }
258
259 StringBuilder sb = new StringBuilder();
260 int symbolCounter = 1;
261 sbyte lastSymbol = flatTiles[0];
262
263 for (int i = 1; i < flatTiles.Length; i++)
264 {
265 sbyte symbol = flatTiles[i];
266
267 if (symbol == lastSymbol)
268 {
269 symbolCounter++;
270 }
271 else
272 {
273 sb.Append(symbolCounter);
274 sb.Append(':');
275 sb.Append(lastSymbol);
276 sb.Append(';');
277
278 lastSymbol = symbol;
279 symbolCounter = 1;
280 }
281 }
282
283 sb.Append(symbolCounter);
284 sb.Append(':');
285 sb.Append(lastSymbol);
286
287 return sb.ToString();
288 }

290 /// <summary>
291 /// Attempts to decode run-length encoded tile data into a flat integer tile array.
292 /// </summary>
293 /// <param name="rle">Run-length encoded tile string.</param>
294 /// <param name="expectedLength">Expected number of decoded tile values.</param>
295 /// <param name="flatTiles">Decoded flat tile array when decoding succeeds; otherwise <c>null</c>.</param>
296 /// <returns><c>true</c> if decoding succeeded and the decoded length matched <paramref name="expectedLength"/>; otherwise <c>false</c>.</returns>
297 public static bool TryDecodeFlatTilesRLE(string rle, int expectedLength, out int[] flatTiles)
298 {
299 flatTiles = null;
300
301 if (string.IsNullOrEmpty(rle))
302 {
303 return false;
304 }
305
306 int[] tempTiles = new int[expectedLength];
307 string[] pairs = rle.Split(';');
308 int index = 0;
309
310 foreach (string pair in pairs)
311 {
312 string[] parts = pair.Split(':');
313 if (parts.Length != 2)
314 {
315 Debug.LogWarning("[ImportExportManager] Invalid RLE format.");
316 return false;
317 }
318
319 if (!int.TryParse(parts[0], out int count) || !int.TryParse(parts[1], out int symbol))
320 {
321 Debug.LogWarning("[ImportExportManager] Invalid RLE number format.");
322 return false;
323 }
324
325 for (int i = 0; i < count; i++)
326 {
327 if (index >= expectedLength)
328 {
329 Debug.LogWarning("[ImportExportManager] Decoded length exceeds expected length.");
330 return false;
331 }
332
333 tempTiles[index++] = symbol;
334 }
335 }
336
337 if (index != expectedLength)
338 {
339 Debug.LogWarning("[ImportExportManager] Decoded length does not match expected length.");
340 return false;
341 }
342
343 flatTiles = tempTiles;
344 return true;
345 }
