6/// <summary>
7/// Unity-serializable wrapper around <see cref="Dictionary{TKey, TValue}"/>.
8/// </summary>
9/// <remarks>
10/// @ingroup core_utils
11/// @brief Stores dictionary data through parallel serialized key and value lists.
12///
13/// Unity cannot serialize generic dictionaries directly. This class mirrors the runtime
14/// dictionary into two parallel lists before serialization and reconstructs it after loading.
15///
16/// Notes:
17/// - Keys must be unique.
18/// - Duplicate keys found during deserialization are ignored with a warning.
19/// - Keys and values must be Unity-serializable types or custom serializable types.
20/// - Iteration order is not guaranteed, matching normal <see cref="Dictionary{TKey, TValue}"/> behavior.
21///
22/// Serialization:
23/// - <c>keys</c> and <c>values</c> are kept in sync during serialization.
24/// - Runtime dictionary operations are forwarded to the internal dictionary.
25/// </remarks>
26[Serializable]
27public class SerializableDictionary<TKey, TValue> :
28 ISerializationCallbackReceiver,
29 IDictionary<TKey, TValue>
30{

38 /// </remarks>

53 #endregion

64 #region Unity Methods (Serialization)

66 /// <summary>
67 /// Copies runtime dictionary contents into serialized key and value lists before Unity serialization.
68 /// </summary>
69 public void OnBeforeSerialize()
70 {
71 keys.Clear();
72 values.Clear();
73
74 foreach (var kvp in dictionary)
75 {
76 keys.Add(kvp.Key);
77 values.Add(kvp.Value);
78 }
79 }
80
81 /// <summary>
82 /// Rebuilds the runtime dictionary from serialized key and value lists after Unity deserialization.
83 /// </summary>
84 /// <remarks>
85 /// If the serialized lists have different lengths, extra entries are ignored. Duplicate keys are also
86 /// ignored so reconstruction does not throw an exception.
87 /// </remarks>
88 public void OnAfterDeserialize()
89 {
90 dictionary.Clear();
91
92 if (keys.Count != values.Count)
93 {
94 Debug.LogWarning(
95 $"[SerializableDictionary] Keys ({keys.Count}) and Values ({values.Count}) count mismatch; " +
96 $"extra entries will be ignored."
97 );
98 }
99
100 var comparer = EqualityComparer<TKey>.Default;
101 var seen = new HashSet<TKey>(comparer);
102 int count = Mathf.Min(keys.Count, values.Count);
103
104 for (int i = 0; i < count; i++)
105 {
106 var key = keys[i];
107
108 // Deduplicate keys to mirror Dictionary behavior and avoid exceptions.
109 if (!seen.Add(key))
110 {
111 Debug.LogWarning($"[SerializableDictionary] Duplicate key ignored during load: {key}");
112 continue;
113 }
114
115 dictionary[key] = values[i];
116 }
117 }
118
119 #endregion
