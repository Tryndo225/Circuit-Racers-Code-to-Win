6/// <summary>
7/// Unity-serializable wrapper around <see cref="Dictionary{TKey, TValue}"/> that round-trips
8/// keys and values via parallel serialized lists.
9/// </summary>
10/// <remarks>
11/// @ingroup core_utils
12/// @details
13/// Unity cannot serialize generic dictionaries directly. This class mirrors the runtime
14/// dictionary into two parallel lists during serialization and reconstructs it after load.
15/// <para>
16/// Notes:
17/// - Keys must be unique. Duplicate keys found on load are ignored with a warning.
18/// - Keys and values must be Unity-serializable types (or custom serializable types).
19/// - Order is not guaranteed (this behaves like a normal <c>Dictionary</c>).
20/// </para>
21/// @invariant <c>keys.Count</c> and <c>values.Count</c> are kept in sync during serialization.
22/// @thread Unity main thread for Unity methods; standard collection access otherwise.
23/// </remarks>
24[Serializable]
25public class SerializableDictionary<TKey, TValue> :
26 ISerializationCallbackReceiver,
27 IDictionary<TKey, TValue>
28{

37 #endregion

39 #region Runtime Storage

49 #region Unity Methods (Serialization)

51 /// <summary>
52 /// Unity method: called before serialization. Copies runtime contents
53 /// into the parallel key/value lists.
54 /// </summary>
55 public void OnBeforeSerialize()
56 {
57 keys.Clear();
58 values.Clear();
59
60 foreach (var kvp in dictionary)
61 {
62 keys.Add(kvp.Key);
63 values.Add(kvp.Value);
64 }
65 }
66
67 /// <summary>
68 /// Unity method: called after deserialization. Rebuilds the runtime dictionary
69 /// from the parallel key/value lists, guarding against length mismatches and duplicates.
70 /// </summary>
71 public void OnAfterDeserialize()
72 {
73 dictionary.Clear();
74
75 if (keys.Count != values.Count)
76 {
77 Debug.LogWarning(
78 $"[SerializableDictionary] Keys ({keys.Count}) and Values ({values.Count}) count mismatch; " +
79 $"extra entries will be ignored."
80 );
81 }
82
83 var comparer = EqualityComparer<TKey>.Default;
84 var seen = new HashSet<TKey>(comparer);
85 int count = Mathf.Min(keys.Count, values.Count);
86
87 for (int i = 0; i < count; i++)
88 {
89 var key = keys[i];
90
91 // Deduplicate keys to mirror Dictionary behavior and avoid exceptions.
92 if (!seen.Add(key))
93 {
94 Debug.LogWarning($"[SerializableDictionary] Duplicate key ignored during load: {key}");
95 continue;
96 }
97
98 dictionary[key] = values[i];
99 }
100 }
101
102 #endregion
