136 #region Example of Character or Sprite Selection
137 int charIndex = TMP_TextUtilities.FindIntersectingCharacter(m_TextComponent, Input.mousePosition, m_Camera, true);
138 if (charIndex != -1 && charIndex != m_lastCharIndex)
139 {
140 m_lastCharIndex = charIndex;
141
142 TMP_TextElementType elementType = m_TextComponent.textInfo.characterInfo[charIndex].elementType;
143
144 // Send event to any event listeners depending on whether it is a character or sprite.
145 if (elementType == TMP_TextElementType.Character)
146 SendOnCharacterSelection(m_TextComponent.textInfo.characterInfo[charIndex].character, charIndex);
147 else if (elementType == TMP_TextElementType.Sprite)
148 SendOnSpriteSelection(m_TextComponent.textInfo.characterInfo[charIndex].character, charIndex);
149 }
150 #endregion

152
153 #region Example of Word Selection
154 // Check if Mouse intersects any words and if so assign a random color to that word.
155 int wordIndex = TMP_TextUtilities.FindIntersectingWord(m_TextComponent, Input.mousePosition, m_Camera);
156 if (wordIndex != -1 && wordIndex != m_lastWordIndex)
157 {
158 m_lastWordIndex = wordIndex;
159
160 // Get the information about the selected word.
161 TMP_WordInfo wInfo = m_TextComponent.textInfo.wordInfo[wordIndex];
162
163 // Send the event to any listeners.
164 SendOnWordSelection(wInfo.GetWord(), wInfo.firstCharacterIndex, wInfo.characterCount);
165 }
166 #endregion

168
169 #region Example of Line Selection
170 // Check if Mouse intersects any words and if so assign a random color to that word.
171 int lineIndex = TMP_TextUtilities.FindIntersectingLine(m_TextComponent, Input.mousePosition, m_Camera);
172 if (lineIndex != -1 && lineIndex != m_lastLineIndex)
173 {
174 m_lastLineIndex = lineIndex;
175
176 // Get the information about the selected word.
177 TMP_LineInfo lineInfo = m_TextComponent.textInfo.lineInfo[lineIndex];
178
179 // Send the event to any listeners.
180 char[] buffer = new char[lineInfo.characterCount];
181 for (int i = 0; i < lineInfo.characterCount && i < m_TextComponent.textInfo.characterInfo.Length; i++)
182 {
183 buffer[i] = m_TextComponent.textInfo.characterInfo[i + lineInfo.firstCharacterIndex].character;
184 }
185
186 string lineText = new string(buffer);
187 SendOnLineSelection(lineText, lineInfo.firstCharacterIndex, lineInfo.characterCount);
188 }
