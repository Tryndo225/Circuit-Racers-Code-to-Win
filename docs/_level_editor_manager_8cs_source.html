118 /// <summary>
119 /// Handles clicks on the level grid and applies the currently selected edit operation.
120 /// </summary>
121 /// <param name="eventData">Pointer event data provided by the Unity EventSystem.</param>
122 /// <remarks>
123 /// The pointer position is converted from screen space into local coordinates of <see cref="levelGridRect"/>.
124 /// The local point is then mapped to tile coordinates in <see cref="editedLevel_"/>.
125 ///
126 /// Depending on the current editor mode, the click either:
127 /// - Moves the start point.
128 /// - Moves the finish point.
129 /// - Paints the clicked tile using <see cref="currentFill_"/>.
130 /// </remarks>
131 public void OnLevelGridClick(BaseEventData eventData)
132 {
133 if (editedLevel_ == null || editedLevel_.Tiles == null)
134 {
135 return;
136 }
137
138 PointerEventData pointerData = eventData as PointerEventData;
139
140 if (pointerData == null)
141 {
142 return;
143 }
144
145 bool converted = RectTransformUtility.ScreenPointToLocalPointInRectangle(levelGridRect, pointerData.position, pointerData.pressEventCamera, out Vector2 localPoint);
146
147 if (!converted)
148 {
149 return;
150 }
151
152 Rect rect = levelGridRect.rect;
153
154 if (!rect.Contains(localPoint))
155 {
156 return;
157 }
158
159 float normalizedX = (localPoint.x - rect.xMin) / rect.width;
160 float normalizedY = (localPoint.y - rect.yMin) / rect.height;
161
162 int tileX = Mathf.FloorToInt(normalizedX * editedLevel_.Width);
163 int tileY = Mathf.FloorToInt((1f - normalizedY) * editedLevel_.Height);
164
165 tileX = Mathf.Clamp(tileX, 0, editedLevel_.Width - 1);
166 tileY = Mathf.Clamp(tileY, 0, editedLevel_.Height - 1);
167
168 if (startSelected_)
169 {
170 editedLevel_.StartPoint = new Coordinates(tileX, tileY);
171 UpdateLevelPreview();
172 return;
173 }
174
175 if (finishSelected_)
176 {
177 editedLevel_.FinishPoint = new Coordinates(tileX, tileY);
178 UpdateLevelPreview();
179 return;
180 }
181
182 editedLevel_.Tiles[tileX, tileY] = (int)currentFill_;
183
184 Debug.Log($"Changed tile [{tileX}, {tileY}] to {(int)currentFill_}");
185 UpdateLevelPreview();
186 }
