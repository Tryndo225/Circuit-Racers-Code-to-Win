6/// <summary>
7/// Custom inspector drawer for <see cref="StringTrackPieceDictionary"/> that presents
8/// a reorderable, foldout-friendly key/value list (string -> <c>TrackPiece</c>).
9/// </summary>
10/// <remarks>
11/// @ingroup editor_util
12/// Features:
13/// - Uses <see cref="ReorderableList"/> for drag reordering, add/remove, and custom element drawing.
14/// - Keeps the hidden <c>keys</c> and <c>values</c> backing arrays strictly in sync (size &amp; order).
15/// - Prevents duplicate keys at edit time with a simple dialog warning.
16/// - Draws the value (<c>TrackPiece</c>) generically, including prefab, position, and rotation fields.
17/// Threading: Unity Editor thread only; excluded from player builds by <c>#if UNITY_EDITOR</c>.
18/// </remarks>
19[CustomPropertyDrawer(typeof(StringTrackPieceDictionary))]
20public class StringTrackPieceDictionaryDrawer : PropertyDrawer
21{
22 /// <summary>Backed <see cref="ReorderableList"/> used to render and manage the dictionary items.</summary>
23 private ReorderableList list;
24
25 /// <summary>Serialized reference to the hidden <c>keys</c> array.</summary>
26 private SerializedProperty keysProp;
27
28 /// <summary>Serialized reference to the hidden <c>values</c> array.</summary>
29 private SerializedProperty valuesProp;
30
31 // ReorderableList draws a drag handle at the far left; leave room so our foldout isn't covered
32
33 /// <summary>
34 /// Horizontal padding reserved for the built-in drag handle, so the foldout icon doesn't overlap.
35 /// </summary>
36 private const float HANDLE_GUTTER = 18f;
37
38 /// <summary>Small vertical padding used in element drawing.</summary>
39 private const float PAD = 2f;
40
41 /// <summary>
42 /// Returns the full height required to draw the property, delegating to the
43 /// underlying <see cref="ReorderableList"/> once initialized.
44 /// </summary>
45 /// <param name="property">Root property for the dictionary.</param>
46 /// <param name="label">GUI label.</param>
47 /// <returns>Required pixel height.</returns>
48 public override float GetPropertyHeight(SerializedProperty property, GUIContent label)
49 {
50 Ensure(property);
51 return list != null ? list.GetHeight() : EditorGUIUtility.singleLineHeight;
52 }
53
54 /// <summary>
55 /// Main GUI entry point: ensures the list is built, syncs sizes, and renders the reorderable list.
56 /// Shows an error help box if the expected backing arrays cannot be found.
57 /// </summary>
58 /// <param name="position">Draw rect.</param>
59 /// <param name="property">Root property for the dictionary.</param>
60 /// <param name="label">GUI label.</param>
61 public override void OnGUI(Rect position, SerializedProperty property, GUIContent label)
62 {
63 Ensure(property);
64 if (list == null)
65 {
66 EditorGUI.HelpBox(position, "Backing fields 'keys'/'values' not found.", MessageType.Error);
67 return;
68 }
69
70 SyncSizes(keysProp, valuesProp);
71 list.DoList(position);
72 }

74 /// <summary>
75 /// Lazily initializes the <see cref="ReorderableList"/> and wires all callbacks (header, add, remove,
76 /// reorder, element draw, element height). Also binds <see cref="keysProp"/> and <see cref="valuesProp"/>.
77 /// </summary>
78 /// <param name="property">Root property for the dictionary.</param>
79 private void Ensure(SerializedProperty property)
80 {
81 if (list != null) return;
82
83 keysProp = property.FindPropertyRelative("keys");
84 valuesProp = property.FindPropertyRelative("values");
85
86 if (keysProp == null || valuesProp == null) return;
87
88 SyncSizes(keysProp, valuesProp);
89
90 list = new ReorderableList(property.serializedObject, keysProp, true, true, true, true);
91
92 // Header
93 list.drawHeaderCallback = rect =>
94 {
95 EditorGUI.LabelField(rect, property.displayName + " (string -> TrackPiece)");
96 };
97
98 // Add
99 list.onAddCallback = l =>
100 {
101 int i = keysProp.arraySize;
102 keysProp.arraySize++;
103 valuesProp.arraySize++;
104
105 keysProp.GetArrayElementAtIndex(i).stringValue = "";
106
107 var v = valuesProp.GetArrayElementAtIndex(i);
108 if (v != null)
109 {
110 var prefabSP = v.FindPropertyRelative("prefab");
111 var posSP = v.FindPropertyRelative("position");
112 var rotSP = v.FindPropertyRelative("rotation");
113 if (prefabSP != null) prefabSP.objectReferenceValue = null;
114 if (posSP != null) posSP.vector3Value = Vector3.zero;
115 if (rotSP != null) rotSP.quaternionValue = Quaternion.identity;
116 }
117
118 property.serializedObject.ApplyModifiedProperties();
119 };
120
121 // Remove
122 list.onRemoveCallback = l =>
123 {
124 int i = l.index;
125 if (i < 0 || i >= keysProp.arraySize) return;
126 keysProp.DeleteArrayElementAtIndex(i);
127 if (i < valuesProp.arraySize) valuesProp.DeleteArrayElementAtIndex(i);
128 property.serializedObject.ApplyModifiedProperties();
129 SyncSizes(keysProp, valuesProp);
130 };
131
132 // Reorder (keep values aligned with keys)
133 list.onReorderCallbackWithDetails = (l, oldIndex, newIndex) =>
134 {
135 MoveArrayElementSafe(valuesProp, oldIndex, newIndex);
136 property.serializedObject.ApplyModifiedProperties();
137 };
138
139 // Per-element draw
140 list.drawElementCallback = (rect, index, active, focused) =>
141 {
142 var key = keysProp.GetArrayElementAtIndex(index);
143 var val = valuesProp.GetArrayElementAtIndex(index);
144 if (key == null || val == null) return;
145
146 float line = EditorGUIUtility.singleLineHeight;
147 float vsp = EditorGUIUtility.standardVerticalSpacing;
148 float pad = PAD;
149
150 // Foldout (leave space for reorderable-list handle on the far left)
151 var foldRect = new Rect(rect.x + HANDLE_GUTTER, rect.y + pad, 16f, line);
152 val.isExpanded = EditorGUI.Foldout(foldRect, val.isExpanded, GUIContent.none, true);
153
154 // Key field
155 var keyRect = new Rect(
156 foldRect.xMax + 4f,
157 rect.y + pad,
158 rect.width - (foldRect.width + HANDLE_GUTTER + 6f),
159 line
160 );
161
162 EditorGUI.BeginChangeCheck();
163 string newKey = EditorGUI.TextField(keyRect, key.stringValue);
164 if (EditorGUI.EndChangeCheck())
165 {
166 if (!IsDuplicateKey(keysProp, index, newKey)) key.stringValue = newKey;
167 else EditorUtility.DisplayDialog("Duplicate Key", $"Key '{newKey}' already exists.", "OK");
168 }
169
170 if (!val.isExpanded) return;
171
172 // Draw all children of TrackPiece generically (prefab/position/rotation, etc.)
173 float x = rect.x + HANDLE_GUTTER + 14f;
174 float w = rect.width - (HANDLE_GUTTER + 14f);
175 float y = keyRect.yMax + vsp;
176
177 var child = val.Copy();
178 var end = val.GetEndProperty();
179 bool enterChildren = true;
180
181 while (child.NextVisible(enterChildren) && !SerializedProperty.EqualContents(child, end))
182 {
183 enterChildren = false; // only descend once from the value
184 float h = EditorGUI.GetPropertyHeight(child, true);
185 var r = new Rect(x, y, w, h);
186 EditorGUI.PropertyField(r, child, true);
187 y += h + vsp;
188 }
189 };
190
191 // Dynamic element height: key line + all children heights when expanded
192 list.elementHeightCallback = index =>
193 {
194 var val = valuesProp.GetArrayElementAtIndex(index);
195 float line = EditorGUIUtility.singleLineHeight;
196 float vsp = EditorGUIUtility.standardVerticalSpacing;
197
198 float h = line + 6f; // key row
199 if (val != null && val.isExpanded)
200 {
201 var child = val.Copy();
202 var end = val.GetEndProperty();
203 bool enterChildren = true;
204 while (child.NextVisible(enterChildren) && !SerializedProperty.EqualContents(child, end))
205 {
206 enterChildren = false;
207 h += EditorGUI.GetPropertyHeight(child, true) + vsp;
208 }
209 }
210 return h;
211 };
212 }

214 #region Helpers

216 /// <summary>
217 /// Ensures the <c>values</c> array matches the size of the <c>keys</c> array.
218 /// </summary>
219 /// <param name="keys">Serialized array of keys.</param>
220 /// <param name="values">Serialized array of values.</param>
221 private static void SyncSizes(SerializedProperty keys, SerializedProperty values)
222 {
223 if (keys == null || values == null) return;
224 if (values.arraySize != keys.arraySize)
225 values.arraySize = keys.arraySize; // keep paired
226 }
227
228 /// <summary>
229 /// Safe wrapper around <see cref="SerializedProperty.MoveArrayElement(int, int)"/> with bounds checks.
230 /// </summary>
231 /// <param name="array">Array property to reorder.</param>
232 /// <param name="oldIndex">Original index.</param>
233 /// <param name="newIndex">Destination index.</param>
234 private static void MoveArrayElementSafe(SerializedProperty array, int oldIndex, int newIndex)
235 {
236 if (array == null) return;
237 int n = array.arraySize;
238 if (oldIndex < 0 || oldIndex >= n || newIndex < 0 || newIndex >= n) return;
239 array.MoveArrayElement(oldIndex, newIndex);
240 }
241
242 /// <summary>
243 /// Returns true if <paramref name="candidate"/> already exists in <paramref name="keys"/>
244 /// at any index other than <paramref name="selfIndex"/>.
245 /// </summary>
246 /// <param name="keys">Key array to scan.</param>
247 /// <param name="selfIndex">Index being edited.</param>
248 /// <param name="candidate">Proposed new key value.</param>
249 /// <returns>True when a duplicate is found; otherwise false.</returns>
250 private static bool IsDuplicateKey(SerializedProperty keys, int selfIndex, string candidate)
251 {
252 for (int i = 0; i < keys.arraySize; i++)
253 {
254 if (i == selfIndex) continue;
255 if (keys.GetArrayElementAtIndex(i).stringValue == candidate)
256 return true;
257 }
258 return false;
259 }
