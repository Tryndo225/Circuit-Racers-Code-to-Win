110/// <summary>
111/// Serializable race replay containing an ordered list of recorded <see cref="ReplaySnapshot"/> values.
112/// </summary>
113/// <remarks>
114/// @ingroup replay_system
115/// @brief Stores replay duration metadata and the recorded snapshots used for replay playback or preview.
116///
117/// A replay can be built incrementally with <see cref="AddSnapshot"/>, copied with <see cref="Copy"/>,
118/// or cleared with <see cref="Reset"/>.
119/// </remarks>
120[Serializable]
121public class Replay
122{
123 /// <summary>
124 /// Overall replay/race time in seconds.
125 /// </summary>
126 [Tooltip("Overall replay time in seconds.")]
127 public float OverallTime;
128
129 /// <summary>
130 /// Recorded replay snapshots.
131 /// </summary>
132 [Tooltip("Recorded replay snapshots used to reconstruct replay playback.")]
133 public List<ReplaySnapshot> Snapshots = new List<ReplaySnapshot>();
134
135 /// <summary>
136 /// Creates an empty replay.
137 /// </summary>
138 public Replay()
139 {
140 Snapshots = new List<ReplaySnapshot>();
141 }
142
143 /// <summary>
144 /// Creates a replay from an overall time and a snapshot list.
145 /// </summary>
146 /// <param name="overallTime">Overall replay time in seconds.</param>
147 /// <param name="snapshots">Snapshots to copy into this replay. If null, an empty list is used.</param>
148 public Replay(float overallTime, List<ReplaySnapshot> snapshots)
149 {
150 OverallTime = overallTime;
151 Snapshots = snapshots != null ? new List<ReplaySnapshot>(snapshots) : new List<ReplaySnapshot>();
152 }
153
154 /// <summary>
155 /// Creates a copy of another replay.
156 /// </summary>
157 /// <param name="other">Replay to copy. If null, an empty replay is created.</param>
158 public Replay(Replay other)
159 {
160 OverallTime = other != null ? other.OverallTime : 0f;
161 Snapshots = other != null ? new List<ReplaySnapshot>(other.Snapshots) : new List<ReplaySnapshot>();
162 }
163
164 /// <summary>
165 /// Gets the effective replay duration.
166 /// </summary>
167 /// <remarks>
168 /// If snapshots exist, this returns the larger value between <see cref="OverallTime"/> and the last
169 /// snapshot time. If no snapshots exist, it returns <see cref="OverallTime"/>.
170 /// </remarks>
171 public float Duration
172 {
173 get
174 {
175 if (Snapshots.Count == 0)
176 {
177 return OverallTime;
178 }
179
180 return Mathf.Max(OverallTime, Snapshots[Snapshots.Count - 1].TrackTime);
181 }
182 }
183
184 /// <summary>
185 /// Adds a new snapshot to the replay and updates <see cref="OverallTime"/> if needed.
186 /// </summary>
187 /// <param name="trackTime">Replay time, in seconds, for the new snapshot.</param>
188 /// <param name="trackPosition">Recorded vehicle position.</param>
189 /// <param name="trackRotation">Recorded vehicle rotation.</param>
190 /// <param name="currentLap">Lap index at the time of recording.</param>
191 /// <param name="currentSpeed">Vehicle speed at the time of recording.</param>
192 /// <param name="steeringAngle">Steering angle at the time of recording.</param>
193 /// <param name="snapshotType">Snapshot type describing whether this is a regular sample or replay event.</param>
194 public void AddSnapshot(float trackTime, Vector3 trackPosition, Quaternion trackRotation, int currentLap, float currentSpeed, float steeringAngle, SnapshotType snapshotType)
195 {
196 Snapshots.Add(new ReplaySnapshot(trackTime, trackPosition, trackRotation, currentLap, currentSpeed, steeringAngle, snapshotType));
197 OverallTime = Mathf.Max(OverallTime, trackTime);
198 }
199
200 /// <summary>
201 /// Creates a copy of this replay.
202 /// </summary>
203 /// <returns>A new <see cref="Replay"/> instance containing copied replay data.</returns>
204 public Replay Copy()
205 {
206 return new Replay(this);
207 }
208
209 /// <summary>
210 /// Clears this replay and resets its overall time to zero.
211 /// </summary>
212 public void Reset()
213 {
214 OverallTime = 0f;
215 Snapshots.Clear();
216 }
