5/// <summary>
6/// Simple automatic transmission: computes engine RPM from wheel RPM and current gear,
7/// auto-shifts up/down based on RPM thresholds and slip, and exposes shift coroutines

74 /// <summary>
75 /// Updates RPM and decides whether to start an upshift/downshift based on thresholds and slip.
76 /// Returns true if torque should be cut this frame (during shift or when starting one).
77 /// </summary>
78 /// <param name="wheelRPM">Average RPM of driven wheels (grounded).</param>
79 /// <param name="wheelSlip">Average slip measure; higher means more slip.</param>
80 /// <returns>True if torque should be cut.</returns>
81 public bool HandleShifting(float wheelRPM, float wheelSlip)
82 {
83 if (_isShifting)
84 {
85 return true;
86 }
87
88 EngineRPM = CalculateRPM(wheelRPM);
89
90 if (wheelSlip > slipThreshold)
91 {
92 return false;
93 }
94
95 if (EngineRPM >= shiftUpRPM && CurrentGear < forwardGears.Length - 1)
96 {
97 StartCoroutine(ShiftUp());
98 return true;
99 }
100 else if (EngineRPM <= shiftDownRPM && CurrentGear > 0)
101 {
102 StartCoroutine(ShiftDown());
103 return true;
104 }
105
106 return false;
107 }

111 #region Private Helpers
112
113 /// <summary>Returns the ratio of the current forward gear.</summary>
114 private float CurrentGearRatio()
115 {
116 return forwardGears[CurrentGear];
117 }
118
119 /// <summary>
120 /// Computes engine RPM from wheel RPM via current gear and final drive, then clamps to [idleRPM, redlineRPM].
121 /// </summary>
122 private float CalculateRPM(float wheelRPM)
123 {
124 // engine rpm = wheel rpm * (gear * final drive)
125 float gr = Mathf.Abs(CurrentGearRatio()) * Mathf.Abs(finalDrive);
126 float engine = wheelRPM * gr;
127
128 // keep above idle, clamp to redline
129 engine = Mathf.Max(idleRPM, engine);
130 engine = Mathf.Min(engine, redlineRPM);
131
132 return engine;
133 }

139 /// <summary>
140 /// Starts an upshift: invokes OnShift callbacks, eases EngineRPM toward idle during shiftDuration,
141 /// then increments CurrentGear and clears shifting flag.
142 /// </summary>
143 private IEnumerator ShiftUp()
144 {
145 if (_isShifting || CurrentGear >= forwardGears.Length - 1)
146 yield break;
147
148 _isShifting = true;
149
150 if (OnShift != null)
151 {
152 foreach (var action in OnShift)
153 {
154 action?.Invoke();
155 }
156 }
157
158 float originalRPM = EngineRPM;
159
160 for (float t = 0; t < shiftDuration; t += Time.deltaTime)
161 {
162 EngineRPM = Mathf.Lerp(originalRPM, idleRPM, t / shiftDuration);
163 yield return null;
164 }
165
166 // Shift up
167 CurrentGear++;
168 _isShifting = false;
169 }
170
171 /// <summary>
172 /// Starts a downshift: invokes OnShift callbacks, eases EngineRPM toward redline during shiftDuration,
173 /// then decrements CurrentGear and clears shifting flag.
174 /// </summary>
175 private IEnumerator ShiftDown()
176 {
177 if (_isShifting || CurrentGear <= 0)
178 yield break;
179
180 _isShifting = true;
181
182 if (OnShift != null)

188 }
189

191

193 {
194 EngineRPM = Mathf.Lerp(originalRPM, redlineRPM, t / shiftDuration);

196 }
