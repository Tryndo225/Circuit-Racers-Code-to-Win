5/// <summary>
6/// Simple automatic transmission controller for a vehicle.
7/// </summary>
8/// <remarks>
9/// @ingroup car_ctrl
10/// @brief Computes engine RPM from wheel RPM, handles automatic gear changes, and exposes shift callbacks.
11///
12/// The controller maps driven-wheel RPM through the current gear ratio and final drive to estimate
13/// engine RPM. It then shifts up or down based on configured RPM thresholds, as long as wheel slip
14/// is below the configured slip threshold.
15///
16/// Behaviour:
17/// - Keeps track of the current forward gear.
18/// - Calculates clamped engine RPM.
19/// - Starts upshift/downshift coroutines when thresholds are reached.
20/// - Temporarily reports torque cut while shifting.
21/// - Invokes registered shift callbacks when a shift starts.
22///
23/// Requirements:
24/// - <see cref="forwardGears"/> should contain at least one gear ratio.
25/// - <see cref="idleRPM"/> should be less than or equal to <see cref="redlineRPM"/>.
26/// - <see cref="shiftDownRPM"/> should usually be lower than <see cref="shiftUpRPM"/>.
27///
28/// Threading:
29/// - Unity main thread only.
30/// - Shift timing uses Unity coroutines.
31/// </remarks>
32public class TransmissionController : MonoBehaviour
33{

126 #region Public API
127
128 /// <summary>
129 /// Updates engine RPM and handles automatic shifting.
130 /// </summary>
131 /// <param name="wheelRPM">Average RPM of grounded driven wheels.</param>
132 /// <param name="wheelSlip">Average wheel slip measure.</param>
133 /// <returns>
134 /// <c>true</c> if drive torque should be cut because a shift is active or has just started;
135 /// otherwise <c>false</c>.
136 /// </returns>
137 /// <remarks>
138 /// Shifting is suppressed while wheel slip is above <see cref="slipThreshold"/>.
139 /// </remarks>
140 public bool HandleShifting(float wheelRPM, float wheelSlip)
141 {
142 if (_isShifting)
143 {
144 return true;
145 }
146
147 EngineRPM = CalculateRPM(wheelRPM);
148
149 if (wheelSlip > slipThreshold)
150 {
151 return false;
152 }
153
154 if (EngineRPM >= shiftUpRPM && CurrentGear < forwardGears.Length - 1)
155 {
156 StartCoroutine(ShiftUp());
157 return true;
158 }
159 else if (EngineRPM <= shiftDownRPM && CurrentGear > 0)
160 {
161 StartCoroutine(ShiftDown());
162 return true;
163 }
164
165 return false;
166 }
167
168 /// <summary>
169 /// Gets the current engine RPM normalized between idle and redline.
170 /// </summary>
171 /// <returns>Normalized RPM value, usually in the range 0 to 1.</returns>
172 public float GetNormalizedRPM()
173 {
174 return Mathf.InverseLerp(idleRPM, redlineRPM, EngineRPM);
175 }

179 #region Private Helpers

181 /// <summary>
182 /// Gets the ratio of the current forward gear.
183 /// </summary>
184 /// <returns>Current gear ratio.</returns>
185 private float CurrentGearRatio()
186 {
187 return forwardGears[CurrentGear];
188 }
189
190 /// <summary>
191 /// Calculates engine RPM from wheel RPM.
192 /// </summary>
193 /// <param name="wheelRPM">Average driven-wheel RPM.</param>
194 /// <returns>Engine RPM clamped between <see cref="idleRPM"/> and <see cref="redlineRPM"/>.</returns>
195 /// <remarks>
196 /// Engine RPM is estimated as wheel RPM multiplied by the current gear ratio and final drive ratio.
197 /// </remarks>
198 private float CalculateRPM(float wheelRPM)
199 {
200 // engine rpm = wheel rpm * (gear * final drive)
201 float gr = Mathf.Abs(CurrentGearRatio()) * Mathf.Abs(finalDrive);
202 float engine = wheelRPM * gr;
203
204 // keep above idle, clamp to redline
205 engine = Mathf.Max(idleRPM, engine);
206 engine = Mathf.Min(engine, redlineRPM);
207
208 return engine;
209 }

213 #region Coroutines: Shifting
214
215 /// <summary>
216 /// Performs an automatic upshift.
217 /// </summary>
218 /// <returns>Coroutine enumerator.</returns>
219 /// <remarks>
220 /// Shift callbacks are invoked when the shift starts. During the shift duration,
221 /// <see cref="EngineRPM"/> is eased toward idle RPM before the current gear is incremented.
222 /// </remarks>
223 private IEnumerator ShiftUp()
224 {
225 if (_isShifting || CurrentGear >= forwardGears.Length - 1)
226 yield break;
227
228 _isShifting = true;
229
230 if (OnShift != null)
231 {
232 foreach (var action in OnShift)
233 {
234 action?.Invoke();
235 }
236 }
237
238 float originalRPM = EngineRPM;
239
240 for (float t = 0; t < shiftDuration; t += Time.deltaTime)
241 {
242 EngineRPM = Mathf.Lerp(originalRPM, idleRPM, t / shiftDuration);
243 yield return null;
244 }
245
246 // Shift up
247 CurrentGear++;
248 _isShifting = false;
249 }
250
251 /// <summary>
252 /// Performs an automatic downshift.
253 /// </summary>
254 /// <returns>Coroutine enumerator.</returns>
255 /// <remarks>
256 /// Shift callbacks are invoked when the shift starts. During the shift duration,
257 /// <see cref="EngineRPM"/> is eased toward redline RPM before the current gear is decremented.
258 /// </remarks>
259 private IEnumerator ShiftDown()
260 {
261 if (_isShifting || CurrentGear <= 0)
262 yield break;
263
264 _isShifting = true;
265
266 if (OnShift != null)
267 {
268 foreach (var action in OnShift)
269 {
270 action?.Invoke();
271 }
272 }
273
274 float originalRPM = EngineRPM;
275
276 for (float t = 0; t < shiftDuration; t += Time.deltaTime)
277 {
278 EngineRPM = Mathf.Lerp(originalRPM, redlineRPM, t / shiftDuration);
279 yield return null;
280 }
281
282 // Shift down
283 CurrentGear--;
284 _isShifting = false;
285 }
