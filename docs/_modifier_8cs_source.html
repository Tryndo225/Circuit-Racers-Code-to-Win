46 public static void SetBlendMode(Material mat, BlendMode mode)
47 {
48 //Note: The source is actually the current filter, destination the result from previous filters
49 //Meaning operations are reversed
50
51 switch (mode)
52 {
53 case BlendMode.Multiply:
54 {
55 mat.SetInt(SrcFactorID, (int)UnityEngine.Rendering.BlendMode.DstColor);
56 mat.SetInt(DstFactorID, (int)UnityEngine.Rendering.BlendMode.SrcColor);
57 mat.SetInt(BlendOpID, (int)BlendOp.Multiply);
58 } break;
59 case BlendMode.Add:
60 {
61 mat.SetInt(SrcFactorID, (int)UnityEngine.Rendering.BlendMode.SrcColor);
62 mat.SetInt(DstFactorID, (int)UnityEngine.Rendering.BlendMode.DstColor);
63 mat.SetInt(BlendOpID, (int)BlendOp.Add);
64 } break;
65 case BlendMode.Subtract:
66 {
67 mat.SetInt(SrcFactorID, (int)UnityEngine.Rendering.BlendMode.SrcColor);
68 mat.SetInt(DstFactorID, (int)UnityEngine.Rendering.BlendMode.DstColor);
69 mat.SetInt(BlendOpID, (int)BlendOp.ReverseSubtract);
70 } break;
71 case BlendMode.Min:
72 {
73 mat.SetInt(SrcFactorID, (int)UnityEngine.Rendering.BlendMode.SrcColor);
74 mat.SetInt(DstFactorID, (int)UnityEngine.Rendering.BlendMode.DstColor);
75 mat.SetInt(BlendOpID, (int)BlendOp.Min);
76 } break;
77 case BlendMode.Max:
78 {
79 mat.SetInt(SrcFactorID, (int)UnityEngine.Rendering.BlendMode.SrcColor);
80 mat.SetInt(DstFactorID, (int)UnityEngine.Rendering.BlendMode.DstColor);
81 mat.SetInt(BlendOpID, (int)BlendOp.Max);
82 } break;
83 default:
84 throw new ArgumentOutOfRangeException(nameof(mode), mode, null);
85 }
