3/// <summary>
4/// Floating replay camera that follows a target from above using static and target-relative offsets.
5/// </summary>
6/// <remarks>
7/// @ingroup replay_system
8/// @brief Follows a replay target transform with smoothed position and rotation.
9///
10/// The camera position is calculated from:
11/// - The target position.
12/// - A static world-space <see cref="offset"/>.
13/// - A target-relative <see cref="relativeOffset"/>.
14///
15/// The camera can optionally flatten the target direction so pitch and roll do not affect the
16/// relative offset. This is useful when the camera should behave more like a top/floating replay camera
17/// than a fully target-rotated chase camera.
18///
19/// Threading:
20/// - Unity main thread only.
21/// - Camera motion is updated in <see cref="LateUpdate"/>.
22/// </remarks>
23public class FloatCamera : MonoBehaviour
24{
25 #region Inspector
26
27 [Header("Target")]
28 /// <summary>
29 /// Transform followed by this camera.
30 /// </summary>
31 /// <remarks>
32 /// If this reference is null, the camera does not update.
33 /// </remarks>
34 [Tooltip("Transform followed by this camera. If empty, camera movement is skipped.")]
35 public Transform target;
36
37 [Header("Offset")]
38 /// <summary>
39 /// Static world-space offset added to the target position.
40 /// </summary>
41 /// <remarks>
42 /// This offset does not rotate with the target.
43 /// </remarks>
44 [Tooltip("Static world-space offset added to the target position.")]
45 public Vector3 offset = Vector3.zero;
46
47 /// <summary>
48 /// Offset relative to the target orientation.
49 /// </summary>
50 /// <remarks>
51 /// X controls forward/backward distance, Y controls height, and Z controls side offset.
52 /// Negative X places the camera behind the target when using the target forward direction.
53 /// </remarks>
54 [Tooltip("Target-relative offset. X is forward/backward, Y is height, and Z is side offset.")]
55 public Vector3 relativeOffset = new Vector3(-10f, 20f, 0f);
56
57 /// <summary>
58 /// Determines whether the relative offset uses only the target yaw direction.
59 /// </summary>
60 /// <remarks>
61 /// When enabled, target pitch and roll are ignored while calculating the relative offset.
62 /// </remarks>
63 [Tooltip("If enabled, relative offset follows only target yaw and ignores pitch/roll.")]
64 public bool useFlatTargetDirection = true;
65
66 [Header("Smoothing")]
67 /// <summary>
68 /// Smooth damp time used for camera position movement.
69 /// </summary>
70 /// <remarks>
71 /// Smaller values make the camera react faster to target movement.
72 /// </remarks>
73 [Tooltip("SmoothDamp time for camera position. Smaller values make the camera more responsive.")]
74 public float positionSmoothTime = 0.2f;
75
76 /// <summary>
77 /// Rotation smoothing speed used when rotating toward the desired camera orientation.
78 /// </summary>
79 /// <remarks>
80 /// Higher values make the camera rotate faster.
81 /// </remarks>
82 [Tooltip("Rotation smoothing speed. Higher values rotate the camera faster.")]
83 public float rotationSpeed = 10f;
84
85 [Header("Orientation")]
86 /// <summary>
87 /// Determines whether the camera's up direction follows the target forward direction.
88 /// </summary>
89 /// <remarks>
90 /// When enabled, the top of the screen follows the target's forward direction.
91 /// When disabled, world forward is used as the screen-up reference.
92 /// </remarks>
93 [Tooltip("If enabled, the top of the screen follows the target's forward direction.")]
94 public bool followTargetRotation = true;
95
96 #endregion
97
98 #region State
99
100 /// <summary>
101 /// Velocity accumulator used by <see cref="Vector3.SmoothDamp(Vector3, Vector3, ref Vector3, float)"/>.
102 /// </summary>
103 private Vector3 positionVelocity;

105 #endregion

107 #region Unity Methods
108
109 /// <summary>
110 /// Updates the camera position and rotation after the target has moved for the frame.
111 /// </summary>
112 private void LateUpdate()
113 {
114 if (target == null)
115 {
116 return;
117 }
118
119 Vector3 desiredPosition = GetDesiredPosition();
120
121 transform.position = Vector3.SmoothDamp(
122 transform.position,
123 desiredPosition,
124 ref positionVelocity,
125 positionSmoothTime);
126
127 Quaternion desiredRotation = GetDesiredRotation();
128
129 transform.rotation = Quaternion.Slerp(
130 transform.rotation,
131 desiredRotation,
132 rotationSpeed * Time.deltaTime);
133 }
134
135 #endregion

139 /// <summary>
140 /// Instantly places and rotates the camera to match the target.
141 /// </summary>
142 /// <remarks>
143 /// Useful after respawn, teleport, or scene loading because it removes smoothing delay.
144 /// </remarks>
145 public void SyncCamera()
146 {
147 if (target == null)

162 public void SetTarget(Transform newTarget)

170 #region Helpers
171
172 /// <summary>
173 /// Calculates the desired world position of the camera.
174 /// </summary>
175 /// <returns>Target position plus world-space and target-relative offsets.</returns>
176 private Vector3 GetDesiredPosition()
177 {
178 return target.position + offset + GetRelativeWorldOffset();
179 }
180
181 /// <summary>
182 /// Calculates the target-relative offset in world space.
183 /// </summary>
184 /// <returns>World-space offset derived from <see cref="relativeOffset"/> and target orientation.</returns>
185 private Vector3 GetRelativeWorldOffset()
186 {
187 Vector3 forward = target.forward;
188 Vector3 right = target.right;
189 Vector3 up = Vector3.up;
190
191 if (useFlatTargetDirection)
192 {
193 forward.y = 0f;
194
195 if (forward.sqrMagnitude < 0.001f)
196 {
197 forward = Vector3.forward;
198 }
199
200 forward.Normalize();
201 right = Vector3.Cross(Vector3.up, forward).normalized;
202 }
203 else
204 {
205 up = target.up;
206 }
207
208 return
209 forward * relativeOffset.x +
210 up * relativeOffset.y +
211 right * relativeOffset.z;
212 }
213
214 /// <summary>
215 /// Calculates the desired camera rotation looking toward the target.
216 /// </summary>
217 /// <returns>Desired camera rotation, or the current rotation if the look direction is too small.</returns>
218 private Quaternion GetDesiredRotation()
219 {
220 Vector3 lookDirection = target.position - transform.position;
221
222 if (lookDirection.sqrMagnitude < 0.001f)
223 {
224 return transform.rotation;
225 }
226
227 Vector3 upDirection = followTargetRotation ? target.forward : Vector3.forward;
228 upDirection.y = 0f;
229
230 if (upDirection.sqrMagnitude < 0.001f)
231 {
232 upDirection = Vector3.forward;
233 }
234
235 return Quaternion.LookRotation(lookDirection.normalized, upDirection.normalized);
236 }
