5/// <summary>
6/// Per-layer configuration for a UI parallax effect.
7/// </summary>
8/// <remarks>
9/// @ingroup ui
10/// Each layer tracks a <see cref="RectTransform"/>, a 2D parallax strength, and an optional Z-tilt.
11/// A larger absolute value in <see cref="Strength"/> moves the layer more as the pointer moves;
12/// <see cref="TiltZ"/> applies a subtle roll (in degrees) around Z. Set <see cref="InvertedTilt"/> to
13/// flip the tilt direction relative to pointer X.
14/// </remarks>
15[Serializable]
16public struct UIParallaxLayer
17{
18 /// <summary>
19 /// The RectTransform to move and optionally tilt.
20 /// </summary>
21 public RectTransform RTransform;
22
23 /// <summary>
24 /// Normalized parallax influence per axis. (1,1) = full movement; (0,0) = frozen.
25 /// </summary>
26 public Vector2 Strength;
27
28 /// <summary>
29 /// Z-axis tilt (in degrees) applied based on normalized pointer X position.
30 /// </summary>
31 public float TiltZ;
32
33 /// <summary>
34 /// If true, flips the sign of the Z-tilt response.
35 /// </summary>
36 public bool InvertedTilt;
37
38 /// <summary>
39 /// Convenience cast: treat a RectTransform as a layer with zero strength and no tilt.
40 /// </summary>
41 /// <param name="rt">Target RectTransform.</param>
42 /// <returns>A layer pointing at <paramref name="rt"/> with default settings.</returns>
43 public static implicit operator UIParallaxLayer(RectTransform rt) => new UIParallaxLayer { RTransform = rt, Strength = Vector2.zero, TiltZ = 0f, InvertedTilt = false };
44}

46/// <summary>
47/// Pointer-driven UI parallax controller for layered RectTransforms.
48/// </summary>
49/// <remarks>
50/// @ingroup ui
51/// @brief Applies smooth anchored-position offsets (and optional Z-tilt) to a set of RectTransforms
52/// based on the pointer position within a reference <see cref="Canvas"/>.
53///
54/// Coordinate space:
55/// - Works in the canvas local space; supports ScreenSpaceOverlay and world/screen camera modes.
56///
57/// Motion model:
58/// - Pointer position is normalized to [-1,1] over the canvas rect. Each layer uses its
59/// <see cref="UIParallaxLayer.Strength"/> to scale a maximum pixel offset derived from <see cref="maxOffset"/>
60/// and the canvas <see cref="Canvas.scaleFactor"/>. Motion is eased toward the target each frame using
61/// exponential smoothing controlled by <see cref="damping"/>.
62///
63/// Threading:
64/// - Unity main thread only (reads <see cref="Input.mousePosition"/> and manipulates transforms).
65///
66/// Usage:
67/// - Assign a reference canvas (or leave null to auto-find in parents on <see cref="Awake"/>).
68/// - Populate <see cref="layers"/> with RectTransforms and per-layer parameters.
69/// - Tune <see cref="maxOffset"/> and <see cref="damping"/> for the desired feel.
70/// </remarks>
71public class UIParallax : MonoBehaviour
72{
73 [Header("Setup")]
74 /// <summary>
75 /// Reference canvas defining the parallax area and scale. Auto-resolved from parents on <see cref="Awake"/> if not set.
76 /// </summary>
77 [SerializeField] private Canvas canvas;
78
79 /// <summary>
80 /// Ordered list of parallax layers to animate (near-to-far or vice versa as you prefer).
81 /// </summary>
82 [SerializeField] private List<UIParallaxLayer> layers = new();
83
84 [Header("Motion")]
85 /// <summary>
86 /// Maximum pixel offset (at Strength = 1) applied at the canvas edges (normalized ±1),
87 /// before division by <see cref="Canvas.scaleFactor"/>.
88 /// </summary>
89 [SerializeField] private float maxOffset = 40f;
90
91 /// <summary>
92 /// Exponential smoothing factor (units: 1/seconds). Higher values converge faster.
93 /// </summary>
94 [SerializeField] private float damping = 8f;
95
96 /// <summary>
97 /// Cached initial anchored positions per layer (set in <see cref="Awake"/>).
98 /// </summary>
99 private Vector2[] initialAnchored;
100
101 /// <summary>
102 /// Cached initial local rotations per layer (set in <see cref="Awake"/>).
103 /// </summary>
104 private Quaternion[] initialRot;
105
106 /// <summary>
107 /// Unity Awake: cache the reference canvas (if missing) and per-layer initial transforms.
108 /// </summary>
109 void Awake()
110 {
111 if (!canvas) canvas = GetComponentInParent<Canvas>();
112 int n = layers.Count;
113 initialAnchored = new Vector2[n];
114 initialRot = new Quaternion[n];
115
116 for (int i = 0; i < n; i++)
117 {
118 if (!layers[i].RTransform) continue;
119 initialAnchored[i] = layers[i].RTransform.anchoredPosition;
120 initialRot[i] = layers[i].RTransform.localRotation;
121 }
122 }
123
124 /// <summary>
125 /// Unity Update: compute normalized pointer position in canvas space and ease layers toward their targets.
126 /// </summary>
127 void Update()
128 {
129 if (!canvas || layers.Count == 0) return;
130
131 RectTransform canvasRect = canvas.transform as RectTransform;
132 Vector2 local;
133 Camera cam = canvas.renderMode == RenderMode.ScreenSpaceOverlay ? null : canvas.worldCamera;
134 RectTransformUtility.ScreenPointToLocalPointInRectangle(canvasRect, Input.mousePosition, cam, out local);
135
136 Vector2 half = canvasRect.rect.size * 0.5f;
137 float nx = Mathf.Clamp(half.x > 0 ? local.x / half.x : 0f, -1f, 1f);
138 float ny = Mathf.Clamp(half.y > 0 ? local.y / half.y : 0f, -1f, 1f);
139
140 float scale = canvas.scaleFactor;
141 float px = (maxOffset / Mathf.Max(scale, 0.0001f));
142
143 // Exponential smoothing toward the target each frame (unscaled time for UI responsiveness).
144 float t = 1f - Mathf.Exp(-damping * Time.unscaledDeltaTime);
145
146 for (int i = 0; i < layers.Count; i++)
147 {
148 var layer = layers[i];
149 if (!layer.RTransform) continue;
150
151 Vector2 target = initialAnchored[i] + new Vector2(
152 nx * layer.Strength.x * px,
153 ny * layer.Strength.y * px
154 );
155
156 layer.RTransform.anchoredPosition = Vector2.Lerp(layer.RTransform.anchoredPosition, target, t);
157
158 if (layer.TiltZ != 0f)
159 {
160 Quaternion targetRot = Quaternion.Euler(0f, 0f, nx * layer.TiltZ * (layer.InvertedTilt ? -1f : 1f)) * initialRot[i];
161 layer.RTransform.localRotation = Quaternion.Slerp(layer.RTransform.localRotation, targetRot, t);
162 }
163 }
164 }
165}
