165 private void Awake()

172 /// <summary>
173 /// Updates smoothed RPM and throttle, computes band weights and limiter, and applies volume/pitch
174 /// to each on/off source.
175 /// </summary>
176 private void Update()
177 {
178 // Smooth inputs
179 RPM = Mathf.Clamp(RPM, minRPM, maxRPM);
180 float dt = Mathf.Max(Time.unscaledDeltaTime, 1e-4f);
181
182 // Exponential smoothing: x += (target - x) * (1 - exp(-k*dt))
183 _rpmSmoothed = Mathf.Lerp(_rpmSmoothed, RPM, 1f - Mathf.Exp(-rpmLerpSpeed * dt));
184 _thrSmoothed = Mathf.Lerp(_thrSmoothed, Mathf.Clamp01(throttle), 1f - Mathf.Exp(-throttleLerpSpeed * dt));
185
186 float n = Mathf.InverseLerp(minRPM, maxRPM, _rpmSmoothed);
187 float onWeight = Mathf.Pow(_thrSmoothed, throttleShape);
188 float offWeight = 1f - onWeight;
189
190 // Limiter gating near redline
191 float limiterGate = 1f;
192 if (n > limiterStart)
193 {
194 float t = Mathf.InverseLerp(limiterStart, 1f, n);
195 float blink = Mathf.PingPong(Time.time * 18f, 1f);
196 limiterGate = Mathf.Lerp(1f, 1f - limiterDepth, blink * t);
197 }
198
199 BandWeights(n, ref _bandWeights);
200
201 // Pitch
202 float pitchBase = pitchVsRpm.Evaluate(n);
203
204 // Shift flare
205 float flare = 1f;
206 if (_shiftFlareT > 0f)
207 {
208 float u = 1f - (_shiftFlareT / shiftFlareTime);
209 flare = 1f + shiftFlareAmount * Mathf.SmoothStep(1f, 0f, u);
210 _shiftFlareT -= Time.deltaTime;
211 }
212
213 for (int i = 0; i < _bands; i++)
214 {
215 float bandVol = _bandWeights[i] * limiterGate * masterVolume;
216
217 // ON
218 if (_onSrc[i] != null)
219 {
220 _onSrc[i].volume = bandVol * onWeight * (1f + onThrottleBoost * 0.5f);
221 _onSrc[i].pitch = pitchBase * flare;
222 }
223
224 // OFF
225 if (_offSrc[i] != null)
226 {
227 _offSrc[i].volume = bandVol * offWeight;
228 _offSrc[i].pitch = pitchBase;
229 }
230 }
231 }

266 /// <summary>
267 /// Computes normalized Gaussian-like weights for the four RPM bands around bandCenters,
268 /// then normalizes weights to sum to 1.
269 /// </summary>
270 /// <param name="n">Normalized RPM in [0,1].</param>
271 /// <param name="bandWeights">Output array of length 4.</param>
272 private void BandWeights(float n, ref float[] bandWeights)
273 {
274 float s = Mathf.Max(1e-3f, bandSharpness);
275 for (int i = 0; i < _bands; i++)
276 {
277 float c = (i == 0) ? bandCenters.x : (i == 1) ? bandCenters.y : (i == 2) ? bandCenters.z : bandCenters.w;
278 float d = Mathf.Abs(n - c);
279 bandWeights[i] = Mathf.Exp(-d * d * s);
280 }
281 // Normalize to sum = 1
282 float sum = bandWeights[0] + bandWeights[1] + bandWeights[2] + bandWeights[3];
283 if (sum > 1e-5f)
284 {
285 float inv = 1f / sum;
286 bandWeights[0] *= inv; bandWeights[1] *= inv; bandWeights[2] *= inv; bandWeights[3] *= inv;
287 }
288 }

308 /// <summary>
309 /// Creates a band source if the clip exists and stores it in the on/off arrays.
310 /// </summary>
311 /// <param name="band">Band index [0..3].</param>
312 /// <param name="clip">Audio clip for this band.</param>
313 /// <param name="onThrottle">True for on-throttle layer, false for off-throttle.</param>
314 private void SetupBandSource(int band, AudioClip clip, bool onThrottle)
315 {
316 if (!clip)
317 return;
318
319 var src = MakeSrc((onThrottle ? "On" : "Off") + "_Band" + band, clip, 0f, true);
320
321 if (onThrottle)
322 _onSrc[band] = src;
323 else
324 _offSrc[band] = src;
325 }
