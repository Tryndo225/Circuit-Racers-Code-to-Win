74 /// <summary>
75 /// Computes an order-sensitive hash value based on the contents of the sequence.
76 /// </summary>
77 /// <typeparam name="T">Element type.</typeparam>
78 /// <param name="source">Sequence to hash. If null, returns 0.</param>
79 /// <param name="comparer">
80 /// Optional equality comparer used when hashing elements. If null,
81 /// <see cref="EqualityComparer{T}.Default"/> is used.
82 /// </param>
83 /// <returns>Hash code reflecting sequence elements and order.</returns>
84 /// <remarks>
85 /// Uses <see cref="System.HashCode"/> aggregation for good distribution.
86 /// Suitable for cache keys within a session; not intended for persistent identities.
87 /// </remarks>
88 public static int GetContentHash<T>(this IEnumerable<T> source, IEqualityComparer<T>? comparer = null)
89 {
90 if (source is null) return 0;
91
92 comparer ??= EqualityComparer<T>.Default;
93
94 var hc = new HashCode();
95 foreach (var item in source)
96 hc.Add(item, comparer);
97
98 return hc.ToHashCode();
99 }
100 }

111 /// <summary>
112 /// Minimal <c>DontDestroyOnLoad</c> singleton base.
113 /// Ensures a single live instance of <typeparamref name="T"/>; duplicates self-destruct in <see cref="Awake"/>.
114 /// </summary>
115 /// <typeparam name="T">Concrete type deriving from <see cref="Singleton{T}"/>.</typeparam>
116 /// <remarks>
117 /// @ingroup core_utils
118 /// @invariant At most one live instance exists for a given <typeparamref name="T"/>.
119 /// @thread Unity main thread (Unity lifecycle methods).
120 /// </remarks>
121 [DisallowMultipleComponent]
122 [DefaultExecutionOrder(-100)] // Awake early so managers are available to others.
123 public class Singleton<T> : MonoBehaviour where T : Singleton<T>
124 {
125 /// <summary>Backer for <see cref="Instance"/>.</summary>
126 private static T _instance;
127
128 /// <summary>
129 /// Global accessor to the current singleton instance (or null if none has awoken yet).
130 /// </summary>
131 public static T Instance => _instance;
132
133 #region Unity Methods
134
135 /// <summary>
136 /// Unity method: enforces the singleton rule and marks the instance as persistent.
137 /// </summary>
138 protected virtual void Awake()
139 {
140 // If an instance already exists and it's not this one, destroy this duplicate.
141 if (_instance != null && _instance != this)
142 {
143 Destroy(gameObject);
144 return;
145 }
146
147 // This object becomes the live instance and persists across scene loads.
148 _instance = (T)this;
149 DontDestroyOnLoad(gameObject);
150 }
151
152 /// <summary>
153 /// Unity method: clears the static instance reference if this object owned it.
154 /// </summary>
155 protected virtual void OnDestroy()
156 {
157 if (_instance == this)
158 _instance = null;
159 }
160
161 #endregion Unity Methods
162 }
163}
