7/// <summary>
8/// Serializable level definition for a grid-based race track.
9/// </summary>
10/// <remarks>
11/// @ingroup level_gen
12/// @brief Stores track dimensions, metadata, start/finish positions, lap settings, day/night state, and tile data.
13///
14/// Tile values are defined by <see cref="LevelTileTypes"/>:
15/// - <see cref="LevelTileTypes.CP"/>: checkpoint.
16/// - <see cref="LevelTileTypes.Spacer"/>: spacer.
17/// - <see cref="LevelTileTypes.Grass"/>: empty/grass.
18/// - <see cref="LevelTileTypes.Track"/>: road.
19/// - <see cref="LevelTileTypes.PlaceHolder"/> and higher values: temporary placeholders used during generation.
20///
21/// The runtime tile grid is stored in <see cref="Tiles"/>. Because Unity does not serialize rectangular 2D arrays directly,
22/// <see cref="tilesFlatRLEString"/> is used as a flattened and run-length encoded serialization representation.
23/// </remarks>
24[Serializable]

224 /// <summary>
225 /// Creates a flattened copy of the current <see cref="Tiles"/> grid.
226 /// </summary>
227 /// <returns>
228 /// A one-dimensional array containing tile values in row-major order.
229 /// </returns>
230 public int[] GetFlatTiles()
231 {
232 if (Tiles == null)
233 UnityEngine.Debug.LogError("Tiles is null when trying to flatten. Ensure Tiles is initialized before serialization.");
234
235 int[] flatTiles = new int[Width * Height];
236 for (int y = 0; y < Height; y++)
237 for (int x = 0; x < Width; x++)
238 flatTiles[y * Width + x] = Tiles[x, y];
239
240 return flatTiles;
241 }

243 /// <summary>
244 /// Flattens <see cref="Tiles"/> and writes the result into <see cref="tilesFlatRLEString"/>
245 /// using run-length encoding.
246 /// </summary>
247 private void FlattenTilesToRLEString()
248 {
249 if (Tiles == null) return;
250 int[] tilesFlat = GetFlatTiles();
251 tilesFlatRLEString = ImportExportManager.EncodeFlatTilesRLE(ImportExportManager.ConvertIntArrayToSByteArray(tilesFlat));
252 }

255 /// <summary>
256 /// Rebuilds a 2D tile grid from a flattened tile array.
257 /// </summary>

259 /// <param name="height">Height of the resulting 2D grid.</param>

262 public static int[,] GetUnflattenedTiles(int[] flatTiles, int height, int width)
263 {
264 if (width <= 0 || height <= 0 || flatTiles == null)
265 UnityEngine.Debug.LogError("Invalid dimensions or tilesFlat is null when trying to unflatten. Ensure Width, Height, and tilesFlat are properly initialized before deserialization.");
266
267 int[,] Tiles = new int[width, height];
268 for (int y = 0; y < height; y++)
269 for (int x = 0; x < width; x++)
270 Tiles[x, y] = flatTiles[y * width + x];
271
272 return Tiles;
273 }

275 /// <summary>
276 /// Decodes <see cref="tilesFlatRLEString"/> and rebuilds the runtime <see cref="Tiles"/> grid.
277 /// </summary>
278 private void UnflattenTilesFromRLEString()
279 {
280 if (Width <= 0 || Height <= 0 || tilesFlatRLEString == null) return;
281
282 int[] tilesFlat = new int[Width * Height];
283 bool resultCheck = ImportExportManager.TryDecodeFlatTilesRLE(tilesFlatRLEString, Width * Height, out tilesFlat);
284
285 Tiles = GetUnflattenedTiles(tilesFlat, Height, Width);
286 }

399 /// <summary>
400 /// Checks whether two tile grids contain the same values for the given dimensions.
401 /// </summary>
402 /// <param name="first">First tile grid.</param>
403 /// <param name="second">Second tile grid.</param>
404 /// <param name="width">Width to compare.</param>
405 /// <param name="height">Height to compare.</param>
406 /// <returns><c>true</c> when both grids match; otherwise <c>false</c>.</returns>
407 private static bool TilesEqual(int[,] first, int[,] second, int width, int height)
408 {
409 if (ReferenceEquals(first, second))
410 {
411 return true;
412 }
413
414 if (first == null || second == null)
415 {
416 return false;
417 }
418
419 if (first.GetLength(0) != second.GetLength(0) ||
420 first.GetLength(1) != second.GetLength(1))
421 {
422 return false;
423 }
424
425 for (int y = 0; y < height; y++)
426 {
427 for (int x = 0; x < width; x++)
428 {
429 if (first[x, y] != second[x, y])
430 {
431 return false;
432 }
433 }
434 }
435
436 return true;
437 }
