83 /// <summary>
84 /// Computes an order-sensitive hash value based on the contents of the sequence.
85 /// </summary>
86 /// <typeparam name="T">Element type.</typeparam>
87 /// <param name="source">Sequence to hash. If null, returns zero.</param>
88 /// <param name="comparer">
89 /// Optional equality comparer used when hashing elements. If null,
90 /// <see cref="EqualityComparer{T}.Default"/> is used.
91 /// </param>
92 /// <returns>Hash code reflecting sequence elements and order.</returns>
93 /// <remarks>
94 /// Uses <see cref="System.HashCode"/> aggregation. Suitable for cache keys within a session,
95 /// but not intended for persistent identities.
96 /// </remarks>
97 public static int GetContentHash<T>(this IEnumerable<T> source, IEqualityComparer<T>? comparer = null)
98 {
99 if (source is null)
100 return 0;
101
102 comparer ??= EqualityComparer<T>.Default;
103
104 var hc = new HashCode();
105 foreach (var item in source)
106 hc.Add(item, comparer);
107
108 return hc.ToHashCode();
109 }
110 }

123 /// <summary>
124 /// Minimal persistent singleton base for Unity components.
125 /// </summary>
126 /// <typeparam name="T">Concrete type deriving from <see cref="Singleton{T}"/>.</typeparam>
127 /// <remarks>
128 /// @ingroup core_utils
129 /// @brief Ensures a single live instance of <typeparamref name="T"/> and keeps it across scene loads.
130 ///
131 /// Behaviour:
132 /// - The first instance becomes <see cref="Instance"/>.
133 /// - Later duplicate instances destroy their own GameObject during <see cref="Awake"/>.
134 /// - The surviving instance is marked with <see cref="UnityEngine.Object.DontDestroyOnLoad(UnityEngine.Object)"/>.
135 ///
136 /// Threading:
137 /// - Unity main thread only, because this class uses Unity lifecycle methods and GameObject destruction.
138 /// </remarks>
139 [DisallowMultipleComponent]
140 [DefaultExecutionOrder(-100)]
141 public class Singleton<T> : MonoBehaviour where T : Singleton<T>
142 {
143 /// <summary>
144 /// Backing field for <see cref="Instance"/>.
145 /// </summary>
146 protected static T _instance;
147
148 /// <summary>
149 /// Gets the current singleton instance, or null if no instance has awoken yet.
150 /// </summary>
151 public static T Instance => _instance;
152
153 #region Unity Methods
154
155 /// <summary>
156 /// Enforces the singleton rule and applies persistence to the surviving instance.
157 /// </summary>
158 protected virtual void Awake()
159 {
160 if (!Coalesce())
161 return;
162
163 Persist();
164 }
165
166 /// <summary>
167 /// Attempts to make this object the active singleton instance.
168 /// </summary>
169 /// <returns>
170 /// <c>true</c> if this object is the active instance; <c>false</c> if it was a duplicate and was destroyed.
171 /// </returns>
172 /// <remarks>
173 /// If another instance already exists, this object's GameObject is destroyed to preserve the singleton rule.
174 /// </remarks>
175 protected virtual bool Coalesce()
176 {
177 if (_instance != null && _instance != this)
178 {
179 Destroy(gameObject);
180 return false;
181 }
182
183 _instance = (T)this;
184 return true;
185 }
186
187 /// <summary>
188 /// Applies persistence behavior to the active singleton instance.
189 /// </summary>
190 /// <remarks>
191 /// The base implementation keeps the GameObject alive across scene loads.
192 /// Override this in derived singleton types that should remain scene-local.
193 /// </remarks>
194 protected virtual void Persist()
195 {
196 DontDestroyOnLoad(gameObject);
197 }
198
199 /// <summary>
200 /// Clears the static instance reference if this object currently owns it.
201 /// </summary>
202 protected virtual void OnDestroy()
203 {
204 if (_instance == this)
205 _instance = null;
206 }
207
208 #endregion Unity Methods
209 }
