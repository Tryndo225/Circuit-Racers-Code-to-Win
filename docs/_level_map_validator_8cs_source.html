4/// <summary>
5/// Validates the structural correctness of a generated <see cref="LevelMap"/>.
6/// </summary>
7/// <remarks>
8/// @ingroup level_gen
9/// @brief Checks map dimensions, start/finish validity, track connectivity, endpoint rules, checkpoint placement, and neighbour counts.
10///
11/// A valid map must:
12/// - Have a non-null tile grid with dimensions matching <see cref="LevelMap.Width"/> and <see cref="LevelMap.Height"/>.
13/// - Have start and finish points inside the map.
14/// - Have start and finish points placed on track-compatible tiles.
15/// - Contain one connected track component.
16/// - Follow circuit or point-to-point endpoint rules.
17/// - Keep track tiles connected with valid neighbour counts.
18/// - Place checkpoints only on straight track sections.
19///
20/// Track-compatible tiles are:
21/// - <see cref="LevelTileTypes.Track"/>
22/// - <see cref="LevelTileTypes.CP"/>
23/// </remarks>
24public static class LevelMapValidator
25{
26 /// <summary>
27 /// Validates a level map.
28 /// </summary>
29 /// <param name="lvl">Level map to validate.</param>
30 /// <returns><c>true</c> if the level map satisfies all validation rules; otherwise <c>false</c>.</returns>
31 public static bool Validate(LevelMap lvl)
32 {
33 if (lvl == null || lvl.Tiles == null)
34 return false;
35
36 if (lvl.Width <= 0 || lvl.Height <= 0)
37 return false;
38
39 if (lvl.Tiles.GetLength(0) != lvl.Width || lvl.Tiles.GetLength(1) != lvl.Height)
40 return false;
41
42 if (!IsInsideMap(lvl, lvl.StartPoint.X, lvl.StartPoint.Y) ||
43 !IsInsideMap(lvl, lvl.FinishPoint.X, lvl.FinishPoint.Y))
44 return false;
45
46 if (!IsTrackTileAt(lvl, lvl.StartPoint.X, lvl.StartPoint.Y) ||
47 !IsTrackTileAt(lvl, lvl.FinishPoint.X, lvl.FinishPoint.Y))
48 return false;
49
50 if (lvl.Circuit)
51 {
52 if (lvl.StartPoint.X != lvl.FinishPoint.X ||
53 lvl.StartPoint.Y != lvl.FinishPoint.Y)
54 return false;
55 }
56 else
57 {
58 if (lvl.StartPoint.X == lvl.FinishPoint.X &&
59 lvl.StartPoint.Y == lvl.FinishPoint.Y)
60 return false;
61 }
62
63 int trackTileCount = 0;
64 int connectedStartX = -1;
65 int connectedStartY = -1;
66 int endpointCount = 0;
67
68 for (int y = 0; y < lvl.Height; ++y)
69 {
70 for (int x = 0; x < lvl.Width; ++x)
71 {
72 if (!IsTrackTile(lvl.Tiles[x, y]))
73 continue;
74
75 ++trackTileCount;
76
77 if (connectedStartX == -1)
78 {
79 connectedStartX = x;
80 connectedStartY = y;
81 }
82
83 int neighbours = CountTrackNeighbours(lvl, x, y);
84
85 bool isStart = IsStartTile(lvl, x, y);
86 bool isFinish = IsFinishTile(lvl, x, y);
87 bool isCheckpoint = IsCheckpointTile(lvl, x, y);
88
89 if (lvl.Circuit)
90 {
91 if (neighbours != 2)
92 return false;
93
94 if (isCheckpoint || isStart || isFinish)
95 {
96 if (!HasOppositeTrackNeighbours(lvl, x, y))
97 return false;
98 }
99 }
100 else
101 {
102 if (isStart || isFinish)
103 {
104 if (neighbours != 1)
105 return false;
106
107 ++endpointCount;
108 }
109 else
110 {
111 if (neighbours != 2)
112 return false;
113
114 if (isCheckpoint)
115 {
116 if (!HasOppositeTrackNeighbours(lvl, x, y))
117 return false;
118 }
119 }
120 }
121 }
122 }
123
124 if (trackTileCount == 0)
125 return false;
126
127 if (!lvl.Circuit && endpointCount != 2)
128 return false;
129
130 int connectedTrackTileCount = CountConnectedTrackTiles(lvl, connectedStartX, connectedStartY);
131
132 return connectedTrackTileCount == trackTileCount;
133 }

197 /// <summary>
198 /// Counts drivable track neighbours around a tile using the four cardinal directions.
199 /// </summary>
200 /// <param name="lvl">Level map containing the tile grid.</param>
201 /// <param name="x">Tile X coordinate.</param>
202 /// <param name="y">Tile Y coordinate.</param>
203 /// <returns>Number of adjacent track-compatible tiles.</returns>
204 private static int CountTrackNeighbours(LevelMap lvl, int x, int y)
205 {
206 int count = 0;
207
208 foreach (Coordinates direction in CardinalDirections)
209 {
210 int neighbourX = x + direction.X;
211 int neighbourY = y + direction.Y;
212
213 if (IsTrackTileAt(lvl, neighbourX, neighbourY))
214 ++count;
215 }
216
217 return count;
218 }

220 /// <summary>
221 /// Checks whether a tile has two opposite track neighbours, forming a straight segment.
222 /// </summary>
223 /// <param name="lvl">Level map containing the tile grid.</param>
224 /// <param name="x">Tile X coordinate.</param>
225 /// <param name="y">Tile Y coordinate.</param>
226 /// <returns>
227 /// <c>true</c> if the tile has either left-right neighbours or up-down neighbours, without perpendicular neighbours.
228 /// Otherwise <c>false</c>.
229 /// </returns>
230 private static bool HasOppositeTrackNeighbours(LevelMap lvl, int x, int y)
231 {
232 bool left = IsTrackTileAt(lvl, x - 1, y);
233 bool right = IsTrackTileAt(lvl, x + 1, y);
234 bool down = IsTrackTileAt(lvl, x, y - 1);
235 bool up = IsTrackTileAt(lvl, x, y + 1);
236
237 bool horizontalStraight = left && right && !up && !down;
238 bool verticalStraight = up && down && !left && !right;
239
240 return horizontalStraight || verticalStraight;
241 }

243 /// <summary>
244 /// Checks whether a coordinate is inside the map and contains a track-compatible tile.
245 /// </summary>
246 /// <param name="lvl">Level map containing the tile grid.</param>
247 /// <param name="x">Tile X coordinate.</param>
248 /// <param name="y">Tile Y coordinate.</param>
249 /// <returns><c>true</c> if the coordinate is inside the map and contains a track-compatible tile; otherwise <c>false</c>.</returns>
250 private static bool IsTrackTileAt(LevelMap lvl, int x, int y)
251 {
252 if (!IsInsideMap(lvl, x, y))
253 return false;
254
255 return IsTrackTile(lvl.Tiles[x, y]);
256 }

258 /// <summary>
259 /// Counts how many track-compatible tiles are connected to the starting coordinate.
260 /// </summary>
261 /// <param name="lvl">Level map containing the tile grid.</param>
262 /// <param name="startX">Starting tile X coordinate.</param>
263 /// <param name="startY">Starting tile Y coordinate.</param>
264 /// <returns>Number of connected track-compatible tiles reachable from the starting coordinate.</returns>
265 /// <remarks>
266 /// Connectivity is checked with a breadth-first search over four-neighbour movement.
267 /// </remarks>
268 private static int CountConnectedTrackTiles(LevelMap lvl, int startX, int startY)
269 {
270 bool[,] visited = new bool[lvl.Width, lvl.Height];
271 Queue<int> queue = new Queue<int>();
272
273 visited[startX, startY] = true;
274 queue.Enqueue(startY * lvl.Width + startX);
275
276 int count = 0;
277
278 while (queue.Count > 0)
279 {
280 int encoded = queue.Dequeue();
281
282 int x = encoded % lvl.Width;
283 int y = encoded / lvl.Width;
284
285 ++count;
286
287 foreach (Coordinates direction in CardinalDirections)
288 {
289 int neighbourX = x + direction.X;
290 int neighbourY = y + direction.Y;
291
292 AddNeighbour(lvl, visited, queue, neighbourX, neighbourY);
293 }
294 }
295
296 return count;
297 }
298
299 /// <summary>
300 /// Adds a neighbouring track tile to the connectivity search queue when it is valid and unvisited.
301 /// </summary>
302 /// <param name="lvl">Level map containing the tile grid.</param>
303 /// <param name="visited">Visited tile lookup used by the breadth-first search.</param>
304 /// <param name="queue">Queue of encoded tile coordinates to visit.</param>
305 /// <param name="x">Neighbour tile X coordinate.</param>
306 /// <param name="y">Neighbour tile Y coordinate.</param>
307 private static void AddNeighbour(LevelMap lvl, bool[,] visited, Queue<int> queue, int x, int y)
308 {
309 if (x < 0 || x >= lvl.Width || y < 0 || y >= lvl.Height)
310 return;
311
312 if (visited[x, y])
313 return;
314
315 if (!IsTrackTile(lvl.Tiles[x, y]))
316 return;
317
318 visited[x, y] = true;
319 queue.Enqueue(y * lvl.Width + x);
320 }
