5/// <summary>
6/// Serializable, editor-friendly scene reference that stores a scene name and path.
7/// </summary>
8/// <remarks>
9/// @ingroup scene_mgmt
10/// @brief Wraps a Unity scene reference so it can be serialized, compared, converted, and executed as a scene load action.
11///
12/// This helper stores the scene name and scene path. In the Unity Editor, it can also keep those
13/// values synchronized with a selected <c>SceneAsset</c>.
14///
15/// It can be used as a runnable scene action through <see cref="Run"/>, which loads the scene through
16/// <see cref="SceneManagement"/>.
17///
18/// Threading:
19/// - Unity main thread only for <see cref="Run"/> and scene loading.
20/// </remarks>

163#if UNITY_EDITOR
164 if (sceneAsset != null)
165 {
166 sceneName = sceneAsset.name;
167 scenePath = UnityEditor.AssetDatabase.GetAssetPath(sceneAsset);

169 else
170 {
171 sceneName = string.Empty;
172 scenePath = string.Empty;
173 }
