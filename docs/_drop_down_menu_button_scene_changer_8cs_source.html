7/// <summary>
8/// Dropdown-driven scene (or action) selector that mirrors the current option onto a companion button,
9/// and invokes a configured action when the button is pressed.
10/// </summary>
11/// <remarks>
12/// @ingroup ui
13/// @brief Configures a <see cref="TMP_Dropdown"/> and a sibling <see cref="Button"/> so the dropdown chooses
14/// an action and the button triggers it. Also supports custom visuals (arrow sprite and color states).
15///
16/// Responsibilities:
17/// - Initialize and style the dropdown template, arrow, and selectable color states.
18/// - Create/configure an adjacent button (when resetting) that displays the selected option's label.
19/// - Keep the button label in sync with the currently selected dropdown item.
20/// - Execute the mapped <see cref="SceneAssetHelper"/> action when the button is clicked.
21///
22/// Threading:
23/// - Unity main thread only (MonoBehaviour lifecycle).
24///
25/// Usage:
26/// - Add to a GameObject with a <see cref="TMP_Dropdown"/> (and children "Template" and "Arrow").
27/// - Assign <see cref="options"/> to populate the dropdown and <see cref="scenes"/> for per-option actions.
28/// - Optionally tweak colors and arrow sprite under "Dropdown Appearance".
29/// - At runtime, user picks an option; the adjacent button shows that selection and triggers its action.
30/// </remarks>

97 /// <summary>

101 private void Reset()
102 {
103 dropdown = GetComponent<TMP_Dropdown>();
104 if (dropdown == null)
105 {
106 Debug.LogError("TMP_Dropdown component is missing. Please add this script to a newly created TMP_Dropdown");
107 }
108
109 if (dropdown.transform.Find("ButtonToChange") != null)
110 {
111 DestroyImmediate(dropdown.transform.Find("ButtonToChange").gameObject);

121 /// <summary>
122 /// Unity OnValidate: keep serialized arrays/sizing and visuals in sync when values change in the Inspector.
123 /// </summary>
124 private void OnValidate()
125 {
126 if (dropdown == null)
127 {
128 dropdown = GetComponent<TMP_Dropdown>();
129 }
130
131 if (scenes == null)
132 {
133 scenes = new SceneAssetHelper[dropdown.options.Count];
134 }
135 else if (scenes.Length != dropdown.options.Count)
136 {
137 var temp = scenes;
138 scenes = new SceneAssetHelper[dropdown.options.Count];
139 for (int i = 0; i < dropdown.options.Count; i++)
140 {
141 if (i < temp.Length)
142 {
143 scenes[i] = temp[i];
144 }
145 else
146 {
147 continue;
148 }
149 }
150 }
151 ConfigureDropdown();
152 ConfigureButton();
153 }

300 /// <summary>

303 public void OnButtonClicked()
