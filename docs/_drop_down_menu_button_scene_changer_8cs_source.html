10/// <summary>
11/// Dropdown-driven scene or action selector that mirrors the current dropdown choice onto a companion button.
12/// </summary>
13/// <remarks>
14/// @ingroup ui
15/// @ingroup scene_mgmt
16/// @brief Links a <see cref="TMP_Dropdown"/> to a generated/assigned <see cref="Button"/> so the dropdown selects an action
17/// and the button executes it.
18///
19/// This component is designed for menu screens where a dropdown should choose between multiple configured actions,
20/// while a large companion button displays the currently selected option and triggers the selected action.
21/// The actions are represented by <see cref="SceneAssetHelper"/> entries stored in <see cref="scenes"/>.
22///
23/// Responsibilities:
24/// - Configure the dropdown options, arrow sprite, template color, and selectable color states.
25/// - Create and configure the companion button during editor reset.
26/// - Keep the companion button label synchronized with the selected dropdown action.
27/// - Execute the selected <see cref="SceneAssetHelper"/> action when the companion button is clicked.
28///
29/// Requirements:
30/// - The GameObject should have a <see cref="TMP_Dropdown"/> component.
31/// - The dropdown hierarchy should contain the expected child objects such as <c>Template</c> and <c>Arrow</c>.
32/// - The <see cref="options"/> list and <see cref="scenes"/> array are index-matched.
33///
34/// Threading:
35/// - Unity main thread only.
36/// - Uses Unity lifecycle and UI callbacks.
37/// </remarks>

152 /// </summary>

178 /// <summary>
179 /// Unity editor validation callback that keeps serialized references, action arrays, and visuals synchronized.
180 /// </summary>
181 /// <remarks>
182 /// When Inspector values change, this method refreshes the dropdown reference, resizes the <see cref="scenes"/>
183 /// array to match the current dropdown option count, and reapplies dropdown/button configuration.
184 /// Existing action references are preserved by index where possible.
185 /// </remarks>
186 private void OnValidate()
187 {
188 if (dropdown == null)
189 {
190 dropdown = GetComponent<TMP_Dropdown>();
191 }
192
193 if (scenes == null)
194 {
195 scenes = new SceneAssetHelper[dropdown.options.Count];
196 }
197 else if (scenes.Length != dropdown.options.Count)
198 {
199 var temp = scenes;
200 scenes = new SceneAssetHelper[dropdown.options.Count];
201 for (int i = 0; i < dropdown.options.Count; i++)
202 {
203 if (i < temp.Length)
204 {
205 scenes[i] = temp[i];
206 }
207 else
208 {
209 continue;
210 }
211 }
212 }
213 ConfigureDropdown();
214 ConfigureButton();
215 }

368 /// <summary>
369 /// Handles dropdown selection changes by updating the companion button label.
370 /// </summary>
371 /// <param name="i">Selected dropdown index provided by <see cref="TMP_Dropdown.onValueChanged"/>.</param>
372 /// <remarks>
373 /// If the selected index has a valid <see cref="SceneAssetHelper"/> with a non-empty name,
374 /// that name is displayed on the companion button. Otherwise, the button displays
375 /// <c>No Action Assigned</c>.
376 /// </remarks>
377 public void OnChange(int i)
378 {
379 if (buttonToChange != null)
380 {
381 var text = buttonToChange.GetComponent<TMP_Text>();
382 if (text != null)
383 {
384 if (0 <= i && i < scenes.Length && scenes[i] != null && !string.IsNullOrEmpty(scenes[i].Name))
385 {
386 text.text = scenes[i].Name;
387 }
388 else
389 {
390 text.text = "No Action Assigned";
391 }
392 }
393 }
394 else
395 {
396 Debug.LogWarning("ButtonToChange is not assigned or found in the hierarchy.");
397 }
398 }

400 /// <summary>
