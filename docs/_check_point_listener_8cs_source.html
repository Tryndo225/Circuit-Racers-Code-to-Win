5/// <summary>
6/// Trigger-based checkpoint sensor that captures the player's pose and velocities on entry
7/// and notifies registered listeners. Intended for lap/track progression and respawn points.
8/// </summary>
9/// <remarks>
10/// @ingroup track_mng
11/// @invariant A <see cref="Collider"/> (set to trigger) and a <see cref="Renderer"/> exist on the same GameObject
12/// (enforced by attributes; validated in <see cref="GetReferences"/>).
13/// @thread Unity main thread only (standard MonoBehaviour lifecycle).
14/// @usage
15/// - Call <see cref="SetActive(bool)"/> to enable/disable the checkpoint (toggles collider &amp; renderer).
16/// - Register callbacks with <see cref="AddListener(Action)"/> to react when the player enters the trigger.
17/// - After a claim, read the captured respawn data via <see cref="cPClaimedPosition"/>, <see cref="cPClaimedRotation"/>,
18/// <see cref="cPClaimedRbLinearVelocity"/>, and <see cref="cPClaimedRbAngularVelocity"/>.
19/// </remarks>
20[Serializable]
21[DisallowMultipleComponent]
22[RequireComponent(typeof(Collider))]
23[RequireComponent(typeof(Renderer))]
24public class CheckPointListener : MonoBehaviour

111 /// <summary>
112 /// Physics callback: when active and a collider tagged <c>Player</c> enters,
113 /// captures position, rotation, and velocities, then notifies listeners.
114 /// </summary>
115 /// <param name="other">Incoming collider.</param>
116 private void OnTriggerEnter(Collider other)
117 {
118 if (_isActive && other.CompareTag("Player"))
119 {
120 other.GetComponent<Rigidbody>();
121
122 cPClaimedPosition = other.transform.position;
123 cPClaimedRotation = other.transform.rotation;
124
125 var playerRigidbody = other.GetComponent<Rigidbody>();
126
127 cPClaimedRbLinearVelocity = playerRigidbody.linearVelocity;
128 cPClaimedRbAngularVelocity = playerRigidbody.angularVelocity;
129
130 if (_listeners != null)
131 {
132 foreach (var listener in _listeners)
133 {
134 listener?.Invoke();
135 }
136 }
137 }
138 }
139
140 /// <summary>
141 /// Enables/disables the checkpoint. When disabled, the collider is turned off and the
142 /// renderer is hidden; the checkpoint will not claim or notify.
143 /// </summary>
144 /// <param name="isActive">True to enable; false to disable.</param>
145 public void SetActive(bool isActive)
146 {
147 if (checkPointCollider == null || checkPointRenderer == null)
148 {
149 GetReferences();
150 }
151
152 _isActive = isActive;
153 checkPointCollider.enabled = isActive;
154 checkPointRenderer.enabled = isActive;
155 }
