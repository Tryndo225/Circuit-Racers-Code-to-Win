134 /// <summary>
135 /// Repaints an individual terrain
136 /// </summary>
137 /// <param name="terrain"></param>
138 /// <param name="syncCPU">(Slow) Copy the internal splatmap render texture to the terrain's "Alphamaps" float arrays. Making it possible to use Terrain.TerrainData.GetAlphamaps()</param>
139 public void RepaintTerrain(Terrain terrain, bool syncCPU = false)
140 {
141 if (layerSettings.Count == 0 || terrain == null) return;
142
143 ModifierStack.Configure(terrain, bounds, splatmapResolution);
144
145 ModifierStack.ProcessLayers(terrain, layerSettings);
146
147 ApplyStampsToTerrain(terrain);
148
149 //Regenerate basemap
150 terrain.terrainData.baseMapResolution = colorMapResolution;
151 terrain.terrainData.SetBaseMapDirty();
152
153 RefreshVegetation(terrain);
154
155 OnTerrainRepaint?.Invoke(terrain);
156
157 if (syncCPU) FinalizeTerrain(terrain);
158
159#if UNITY_EDITOR
160 EditorUtility.SetDirty(terrain.terrainData);
161#endif
162 }

164 private void RefreshVegetation(Terrain terrain)
165 {
166#if VEGETATION_STUDIO_PRO
167 AwesomeTechnologies.VegetationStudio.VegetationStudioManager manager = AwesomeTechnologies.VegetationStudio.VegetationStudioManager.Instance;
168
169 if (refreshVegetationOnPaint && manager)
170 {
171#if UNITY_2019_1_OR_NEWER
172 //VS requires the CPU data
173 terrain.terrainData.SyncTexture(TerrainData.AlphamapTextureName);
174#endif
175 foreach (AwesomeTechnologies.VegetationSystem.VegetationSystemPro sys in manager.VegetationSystemList)
176 {
177 if (sys == null) continue;
178
179 sys.ClearCache();
180 sys.RefreshTerrainArea(terrain.terrainData.bounds);
181 }
182#if UNITY_EDITOR
183 SceneView.RepaintAll();
184#endif
185 }
186#endif

200 /// <summary>
201 /// Adds or removes the TerrainChangeListener component from all assigned terrains. If enabled, terrains will be repainted when their height is modified
202 /// </summary>
203 /// <param name="value"></param>
204 public void SetAutoRepaint(bool value)
205 {
206 autoRepaint = value;
207
208 if (value)
209 {
210 RemoveTerrainListeners();
211
212 foreach (Terrain terrain in terrains)
213 {
214 TerrainChangeListener listener = terrain.GetComponent<TerrainChangeListener>();
215 if (!listener) listener = terrain.gameObject.AddComponent<TerrainChangeListener>();
216
217 listener.terrain = terrain;
218 this.terrainListeners.Add(listener);
219 }
220 }
221 else
222 {
223 RemoveTerrainListeners();
224 }
225 }
226
227 private void RemoveTerrainListeners()
228 {
229 for (int i = 0; i < terrainListeners.Count; i++)
230 {
231 DestroyImmediate(terrainListeners[i]);
232 }
233
234 terrainListeners.Clear();
235 }
