27/// <summary>
28/// Conditional display attribute for Inspector fields.
29/// </summary>
30/// <remarks>
31/// @ingroup editor_attrs
32/// @brief Shows or hides a property based on the boolean value of another field on the same object.
33///
34/// This attribute is used to create more compact inspectors by revealing fields only when they are relevant.
35///
36/// Show when true:
37/// <code>
38/// [SerializeField] private bool useAdvanced;
39///
40/// [ShowIf(nameof(useAdvanced))]
41/// public float advancedGain;
42/// </code>
43///
44/// Show when false:
45/// <code>
46/// [SerializeField] private bool useDefaults = true;
47///
48/// [ShowIf(nameof(useDefaults), false)]
49/// public float customValue;
50/// </code>
51///
52/// Limitations:
53/// - The referenced field must be a <c>bool</c> member on the same component or object.
54/// - Requires a custom <c>PropertyDrawer</c> for <see cref="ShowIfAttribute"/> to affect the Inspector.
55/// - Has no runtime effect.
56/// </remarks>
57public class ShowIfAttribute : PropertyAttribute
58{
59 /// <summary>
60 /// Name of the boolean field whose value controls the visibility of the decorated property.
61 /// </summary>
62 public readonly string Field;
63
64 /// <summary>
65 /// Required state of <see cref="Field"/> for the decorated property to be shown.
66 /// </summary>
67 /// <remarks>
68 /// If the referenced field equals this value, the decorated property is visible.
69 /// Otherwise, it is hidden by the matching property drawer.
70 /// </remarks>
71 public readonly bool RequiredState;
72
73 /// <summary>
74 /// Creates an attribute that shows the property when the referenced boolean field is <c>true</c>.
75 /// </summary>
76 /// <param name="boolField">Name of a <c>bool</c> field on the same object.</param>
77 public ShowIfAttribute(string boolField)
78 {
79 Field = boolField;
80 RequiredState = true;
81 }

88 /// Required value of the referenced field for the property to be visible.
89 /// Use <c>true</c> to show when true, or <c>false</c> to show when false.
90 /// </param>
91 public ShowIfAttribute(string boolField, bool mustBeTrue)
