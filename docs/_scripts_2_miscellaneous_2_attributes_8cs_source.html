15/// <summary>
16/// Conditional display attribute for Inspector fields.
17/// Shows (or hides) a property based on the boolean value of another field on the same object.
18/// </summary>
19/// <remarks>
20/// @ingroup editor_attrs
21/// @purpose Create compact inspectors by revealing advanced options only when relevant.
22/// @usage
23/// <para>Show when true:</para>
24/// <code>
25/// [SerializeField] private bool useAdvanced;
26/// [ShowIf(nameof(useAdvanced))]
27/// public float advancedGain;
28/// </code>
29/// <para>Show when false:</para>
30/// <code>
31/// [SerializeField] private bool useDefaults = true;
32/// [ShowIf(nameof(useDefaults), false)]
33/// public float customValue;
34/// </code>
35/// @limits
36/// - The referenced field must be a <c>bool</c> member on the same component/object.
37/// - Requires a custom <c>PropertyDrawer</c> for <see cref="ShowIfAttribute"/> to actually affect the Inspector.
38/// - No runtime impact; purely an editor hint.
39/// </remarks>
40public class ShowIfAttribute : PropertyAttribute
41{
42 /// <summary>
43 /// Name of the boolean field whose value controls the visibility of the decorated property.
44 /// </summary>
45 public readonly string Field;
46
47 /// <summary>
48 /// Required state of <see cref="Field"/> for the decorated property to be shown.
49 /// If the field equals this value, the property is visible; otherwise it is hidden.

52
53 /// <summary>
54 /// Shows the property when the referenced boolean field is <c>true</c>.
