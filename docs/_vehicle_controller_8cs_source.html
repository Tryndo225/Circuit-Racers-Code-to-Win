6/// <summary>
7/// High-level vehicle controller that connects player input to vehicle subsystems.
8/// </summary>
9/// <remarks>
10/// @ingroup car_ctrl
11/// @brief Wires input actions to drivetrain, transmission, engine audio, and lights.
12///
13/// This component acts as the main setup and coordination point for a playable vehicle. It owns the
14/// inspector-facing tuning values and pushes them into the specialized runtime components:
15/// - <see cref="DriveTrainController"/> for wheel physics and vehicle movement.
16/// - <see cref="TransmissionController"/> for RPM and shifting.
17/// - <see cref="EngineSound"/> for layered engine audio.
18/// - <see cref="LightsController"/> for vehicle lights.
19///
20/// Requirements:
21/// - Exactly four wheels should be configured in FL, FR, RL, RR order.
22/// - Each wheel entry should contain both a <see cref="WheelCollider"/> and a visual wheel transform.
23/// - A <see cref="Rigidbody"/> must be present on the same object.
24///
25/// Threading:
26/// - Unity main thread only.
27/// - Vehicle control is applied from <see cref="FixedUpdate"/>.
28/// </remarks>
29[RequireComponent(typeof(Rigidbody))]
30public class VehicleController : MonoBehaviour
31{
