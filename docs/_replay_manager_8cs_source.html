125 /// <summary>
126 /// Captures the current race and vehicle state as a replay snapshot.
127 /// </summary>
128 /// <param name="type">Replay snapshot type describing why the snapshot is being recorded.</param>
129 /// <remarks>
130 /// The vehicle reference is resolved lazily by finding a <see cref="VehicleController"/> in the scene.
131 /// Snapshot capture is skipped if required race managers or vehicle references are unavailable.
132 /// </remarks>
133 private void TakeSnapshot(ReplaySnapshot.SnapshotType type)
134 {
135 if (currentReplay_ == null)
136 return;
137
138 if (RaceTimeManager.Instance == null || TrackManager.Instance == null)
139 return;
140
141 if (vehicleTransform_ == null)
142 {
143 VehicleController vehicle = FindFirstObjectByType<VehicleController>();
144
145 if (vehicle == null)
146 return;
147
148 vehicleTransform_ = vehicle.transform;
149 vehicleRigidBody_ = vehicle.GetComponent<Rigidbody>();
150 driveTrainController_ = vehicle.GetComponent<DriveTrainController>();
151 }
152
153 float vehicleSpeed = 0f;
154
155 if (vehicleRigidBody_ != null)
156 vehicleSpeed = vehicleRigidBody_.linearVelocity.magnitude;
157
158 float steeringAngle = 0f;
159
160 if (driveTrainController_ != null)
161 steeringAngle = driveTrainController_.GetSteeringAngle();
162
163 currentReplay_.AddSnapshot(RaceTimeManager.Instance.GetCurrentRaceTime(), vehicleTransform_.position, vehicleTransform_.rotation, TrackManager.Instance.CurrentLap, vehicleSpeed, steeringAngle, type);
164 }
