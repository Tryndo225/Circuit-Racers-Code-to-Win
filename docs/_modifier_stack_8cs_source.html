31 /// <summary>
32 /// Call this once, per terrain. Sets up the Blit material and generates a normal map
33 /// </summary>
34 /// <param name="terrain"></param>
35 /// <param name="bounds"></param>
36 /// <param name="rtDsc"></param>
37 public static void Configure(Terrain terrain, Bounds bounds, int resolution)
38 {
39 if (m_resolution != resolution || alphaMap == null)
40 {
41 alphaMap = new RenderTexture(resolution, resolution, 0, GraphicsFormat.R8_UNorm);
42 }
43 m_resolution = resolution;
44
45 if (!filterMat) filterMat = new Material(Shader.Find("Hidden/TerrainPainter/Modifier"));
46
47 //Note: heightmap borders aren't blended with neighboring texels, causes visible seams at low resolutions
48 filterMat.SetTexture(HeightmapID, terrain.terrainData.heightmapTexture);
49 filterMat.SetTexture(NormalMapID, terrain.normalmapTexture);
50
51 heightScale = bounds.max.y - bounds.min.y;
52 filterMat.SetFloat(HeightmapScaleID, heightScale);
53
54 //Used to reconstruct the global-bounds aligned UV
55 float invWidth = 1.0f / bounds.size.x;
56 float invHeight = 1.0f / bounds.size.z;
57
58 var terrainPosScale = new Vector4(
59 (terrain.GetPosition().x * invWidth) - (bounds.min.x * invWidth),
60 (terrain.GetPosition().z * invHeight) - (bounds.min.z * invHeight),
61 terrain.terrainData.size.x / bounds.size.x,
62 terrain.terrainData.size.z / bounds.size.z
63 );
64
65 filterMat.SetVector(TerrainPosScaleID, terrainPosScale);
66 filterMat.SetVector(TerrainBoundsID, new Vector4(bounds.min.x, bounds.max.z, bounds.size.x, bounds.size.z));
67 }

78 /// <summary>
79 /// Executes the modifiers in order and generates the necessary splatmaps
80 /// </summary>
81 /// <param name="layer"></param>
82 /// <param name="modifiers"></param>
83 /// <param name="output"></param>
84 public static void ProcessSingleLayer(Terrain terrain, LayerSettings settings)
85 {
86 //Disable or with no settings (result in a fill with black)
87 if (settings.enabled == false || settings.layer == false) return;
88
89 Graphics.SetRenderTarget(alphaMap);
90
91 //Start with a full white base
92 Graphics.Blit(Texture2D.whiteTexture, alphaMap);
93
94 //Reverse order
95 for (int i = settings.modifierStack.Count-1; i >= 0; i--)
96 {
97 settings.modifierStack[i].Configure(filterMat);
98 settings.modifierStack[i].Execute(alphaMap);
99 }
100
101 //Scale splatmap to fit in terrain
102 Vector2 scaledSplatmapSize = new Vector2((terrain.terrainData.size.x / m_resolution) * m_resolution, (terrain.terrainData.size.z / m_resolution) * m_resolution);
103
104 //PaintContext handles creation of splatmaps. Subtracting the weights of a splatmap, from the ones before it.
105 //A single pixels of all combined alpha maps must not exceed a value of one. The 2nd pass of Hidden/TerrainEngine/TerrainLayerUtils is used internally to do this
106 //Note: Paintcontext must use a serialized terrain reference, otherwise breaks when executing "ApplyDelayedActions" when the project is saved!
107 PaintContext c = TerrainPaintUtility.BeginPaintTexture(terrain, new Rect(0, 0, scaledSplatmapSize.x, scaledSplatmapSize.y), settings.layer);
108
109 Graphics.Blit(alphaMap, c.destinationRenderTexture);
110
111 TerrainPaintUtility.EndPaintTexture(c, UndoActionName);
112 }
