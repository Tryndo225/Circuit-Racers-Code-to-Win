6/// <summary>
7/// Renders a small top-down preview image of a <see cref="LevelMap"/> into a UI <see cref="RawImage"/>.
8/// </summary>
9/// <remarks>
10/// @ingroup ui
11/// @brief Lightweight level thumbnail generator for menus and saved-level lists.
12///
13/// The preview draws:
14/// - grass/background cells,
15/// - road cells from <see cref="LevelMap.LevelTileTypes.Track"/>,
16/// - checkpoint cells from <see cref="LevelMap.LevelTileTypes.CP"/>,
17/// - start and finish markers.
18///
19/// Responsibilities:
20/// - Choose a pixels-per-cell value based on the target rect width and device max texture size.
21/// - Convert the tile grid into a CPU color buffer on a worker thread.
22/// - Upload the buffer into a point-filtered <see cref="Texture2D"/> and assign it to the UI.
23///
24/// Threading:
25/// - CPU buffer generation is done off the main thread through <c>Task.Run</c>.
26/// - <see cref="Texture2D"/> creation and UI assignment must occur on the Unity main thread after the await.
27///
28/// Performance notes:
29/// - Uses <see cref="TextureFormat.RGBA32"/> and <see cref="FilterMode.Point"/> for crisp pixel art.
30/// - Avoid calling <see cref="ShowPreviewAsync(LevelMap)"/> every frame; call it only when data or size changes.
31/// </remarks>
32public class LevelPreviewer : MonoBehaviour
33{

95 /// <summary>
96 /// Asynchronously builds and displays a preview texture for the given level map.
97 /// </summary>
98 /// <param name="map">Level definition to visualize.</param>
99 /// <remarks>
100 /// The CPU color buffer is generated on a worker thread. The <see cref="Texture2D"/> is created
101 /// and assigned on the Unity main thread after the asynchronous work completes.
102 /// </remarks>
103 public async Task ShowPreviewAsync(LevelMap map)
104 {
105 if (target == null) return;
106
107 int ppc = Mathf.Max(1, Mathf.FloorToInt(target.rectTransform.rect.width / map.Width));
108
109 int maxTex = SystemInfo.maxTextureSize;
110 ppc = Mathf.Min(ppc, Mathf.Max(1, maxTex / Math.Max(map.Width, map.Height)));
111
112 var result = await Task.Run(() =>
113 {
114 return BuildPreviewBuffer(map, ppc);
115 });
116
117 var tex = new Texture2D(result.texWidth, result.texHeight, TextureFormat.RGBA32, false)
118 {
119 filterMode = FilterMode.Point,
120 wrapMode = TextureWrapMode.Clamp
121 };
122 tex.SetPixels32(result.buffer);
123 tex.Apply(false, false);
124
125 target.texture = tex;
126 }

128 /// <summary>
129 /// Constructs a CPU color buffer for the preview.
130 /// </summary>
131 /// <param name="map">Level grid to render.</param>
132 /// <param name="pixelsPerCell">Preview resolution in pixels per tile.</param>
133 /// <returns>Color buffer and texture dimensions ready for <see cref="Texture2D.SetPixels32(Color32[])"/>.</returns>
134 private (Color32[] buffer, int texWidth, int texHeight) BuildPreviewBuffer(LevelMap map, int pixelsPerCell)
135 {
136 const int TrackTile = (int)LevelMap.LevelTileTypes.Track;
137 const int CheckpointTile = (int)LevelMap.LevelTileTypes.CP;
138
139 int texWidth = map.Width * pixelsPerCell;
140 int texHeight = map.Height * pixelsPerCell;
141
142 var buffer = new Color32[texWidth * texHeight];
143
144 for (int i = 0; i < buffer.Length; i++)
145 buffer[i] = grass;
146
147 for (int y = 0; y < map.Height; y++)
148 {
149 for (int x = 0; x < map.Width; x++)
150 {
151 int v = map.Tiles[x, y];
152
153 if (v == TrackTile || v == CheckpointTile)
154 FillCell(buffer, texWidth, texHeight, x, y, pixelsPerCell, road);
155
156 if (v == CheckpointTile)
157 DrawMarker(buffer, texWidth, texHeight, new Coordinates(x, y), pixelsPerCell, checkPoint);
158 }
159 }
160
161 DrawMarker(buffer, texWidth, texHeight, map.StartPoint, pixelsPerCell, start);
162 DrawMarker(buffer, texWidth, texHeight, map.FinishPoint, pixelsPerCell, finish);
163
164 return (buffer, texWidth, texHeight);
165 }

167 /// <summary>
168 /// Fills a single tile cell area with a solid color in the buffer.
169 /// </summary>
170 /// <param name="buffer">Destination color buffer.</param>
171 /// <param name="texWidth">Total texture width in pixels.</param>
172 /// <param name="texHeight">Total texture height in pixels.</param>
173 /// <param name="cellX">Tile X coordinate.</param>
174 /// <param name="cellY">Tile Y coordinate.</param>
175 /// <param name="pixelsPerCell">Pixels per tile.</param>
176 /// <param name="color">Fill color.</param>
177 private static void FillCell(Color32[] buffer, int texWidth, int texHeight, int cellX, int cellY, int pixelsPerCell, Color32 color)
178 {
179 int pxX = cellX * pixelsPerCell;
180 int pxY = cellY * pixelsPerCell;
181
182 pxY = texHeight - pixelsPerCell - pxY;
183
184 for (int dy = 0; dy < pixelsPerCell; dy++)
185 {
186 int row = (pxY + dy) * texWidth;
187 int idx = row + pxX;
188 for (int dx = 0; dx < pixelsPerCell; dx++)
189 {
190 buffer[idx + dx] = color;
191 }
192 }
193 }
194
195 /// <summary>
196 /// Draws a square marker centered within a tile.
197 /// </summary>
198 /// <param name="buffer">Destination color buffer.</param>
199 /// <param name="texWidth">Total texture width in pixels.</param>
200 /// <param name="texHeight">Total texture height in pixels.</param>
201 /// <param name="cell">Tile coordinate to mark.</param>
202 /// <param name="pixelsPerCell">Pixels per tile.</param>
203 /// <param name="color">Marker color.</param>
204 private static void DrawMarker(Color32[] buffer, int texWidth, int texHeight, Coordinates cell, int pixelsPerCell, Color32 color)
205 {
206 if (cell.X < 0 || cell.Y < 0) return;
207
208 int size = Math.Max(1, pixelsPerCell / 2);
209 int pxX = cell.X * pixelsPerCell + (pixelsPerCell - size) / 2;
210 int pxY = cell.Y * pixelsPerCell + (pixelsPerCell - size) / 2;
211
212 pxY = texHeight - size - pxY;
213
214 for (int dy = 0; dy < size; dy++)
215 {
216 int row = (pxY + dy) * texWidth;
217 int idx = row + pxX;
218 for (int dx = 0; dx < size; dx++)
219 {
220 buffer[idx + dx] = color;
221 }
222 }
223 }
