3/// <summary>
4/// Simple popup controller that previews a generated <see cref="LevelMap"/> and lets the user
5/// accept (add to game data) or dismiss it.
6/// </summary>
7/// <remarks>
8/// @ingroup ui
9/// @brief Shows/hides a preview panel and forwards the map to <see cref="GameDataManager"/> when accepted.
10///
11/// Responsibilities:
12/// - Toggle a visual container on/off for the preview.
13/// - Drive a co-located <see cref="LevelPreviewer"/> to render the given <see cref="LevelMap"/>.
14/// - Add the accepted map to the persistent list via <see cref="GameDataManager.AddLevel(LevelMap)"/>.
15///
16/// Threading:
17/// - Unity main thread only (standard MonoBehaviour lifecycle).
18///
19/// Usage:
20/// - Call <see cref="ShowMap(LevelMap)"/> with a generated map to display the popup preview.
21/// - Call <see cref="KeepMap()"/> to persist the currently previewed map; <see cref="HideMap()"/> to close.
22/// </remarks>
23public class LevelPopUp : MonoBehaviour
24{
