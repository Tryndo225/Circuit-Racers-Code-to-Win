106 /// <summary>
107 /// Keeps this object at the listener position and services SFX pooling/continuations.
108 /// </summary>
109 private void Update()
110 {
111 // Follow the main camera so “global” SFX use the listener position.
112 if (_camera == null) _camera = Camera.main;
113 if (_camera != null) transform.position = _camera.transform.position;
114
115 // Pin all helper sources to this position (spatialized per-clip via blend on helper).
116 for (int i = 0; i < _sfxSources.Count; i++)
117 {
118 var helper = _sfxSources[i];
119 if (helper != null) helper.transform.position = transform.position;
120 }
121
122 // Manage “continue” countdowns.
123 _toBeRemoved.Clear();
124 _keys = _keys ?? new List<AudioClip>(4);
125 _keys.Clear();
126 _keys.AddRange(_toBeContinued.Keys);
127
128 for (int i = 0; i < _keys.Count; i++)
129 {
130 var clip = _keys[i];
131 float t = _toBeContinued[clip] - Time.deltaTime;
132 if (t > 0f)
133 {
134 _toBeContinued[clip] = t;
135 }
136 else
137 {
138 if (_sfxMapping.TryGetValue(clip, out var helper) && helper != null)
139 helper.StopSound();
140 _toBeRemoved.Add(clip);
141 }
142 }
143
144 for (int i = 0; i < _toBeRemoved.Count; i++)
145 _toBeContinued.Remove(_toBeRemoved[i]);
146 }

152 /// <summary>

154 /// and playback begins; if the same clip is already set but not playing, it resumes.
155 /// </summary>

157 public void PlayMusic(AudioClip clip)
158 {
159 if (clip == null) return;

161 if (_musicSource.clip != clip)
162 {
163 _musicSource.clip = clip;
164 _musicSource.Play();

168 _musicSource.Play();

193 /// <summary>
194 /// Plays (or reuses) a pooled SFX helper for the given clip at a target transform, with
195 /// optional volume/pitch/loop and spatial blend control.
196 /// </summary>
197 /// <param name="clip">Clip to play (ignored if null).</param>
198 /// <param name="transform">World target whose position is tracked by the helper.</param>
199 /// <param name="volume">Per-call volume scalar (multiplied by global <see cref="sfxVolume"/>).</param>
200 /// <param name="pitch">Per-call pitch multiplier.</param>
201 /// <param name="loop">Whether the SFX should loop.</param>
202 /// <param name="specialBlend">
203 /// Spatial blend for the helper [0..1]. 0 = 2D, 1 = fully 3D.
204 /// </param>
205 /// <param name="reset">
206 /// If true and the clip is already playing, the helper is reset and restarted. If false,
207 /// an already-playing clip continues uninterrupted.
208 /// </param>
209 public void PlaySFXClip(
210 AudioClip clip,
211 Transform transform,
212 float volume = 1f,
213 float pitch = 1f,
214 bool loop = false,
215 float specialBlend = 0.0f,
216 bool reset = true)
217 {
218 if (clip == null) return;
219
220 // Try to reuse existing helper for this clip.
221 if (!_sfxMapping.ContainsKey(clip))
222 {
223 // Find an unused helper first.
224 for (int i = 0; i < _sfxSources.Count; ++i)
225 {
226 if (!_sfxSources[i].IsUsed)
227 {
228 _sfxMapping.Remove(_sfxSources[i].Clip);
229 SetUpHelper(_sfxSources[i], clip, transform, volume * sfxVolume, pitch, loop, specialBlend);
230 _sfxMapping[clip] = _sfxSources[i];
231 _sfxSources[i].PlaySound();
232 return;
233 }
234 }
235
236 // None free—create a new helper.
237 var helper = new GameObject("SFXHelper").AddComponent<SoundManagerHelper>();
238 SetUpHelper(helper, clip, transform, volume * sfxVolume, pitch, loop, specialBlend);
239 _sfxSources.Add(helper);
240 _sfxMapping[clip] = helper;
241 helper.PlaySound();
242 }
243 else
244 {
245 var helper = _sfxMapping[clip];
246 if (helper == null)
247 {
248 // Defensive: mapping exists but helper missing—create a fresh one.
249 helper = new GameObject("SFXHelper").AddComponent<SoundManagerHelper>();
250 SetUpHelper(helper, clip, transform, volume * sfxVolume, pitch, loop, specialBlend);
251 _sfxSources.Add(helper);
252 _sfxMapping[clip] = helper;
253 helper.PlaySound();
254 return;
255 }
256
257 if (helper.IsPlaying)
258 {
259 if (reset)
260 {
261 helper.ResetSound();
262 helper.PlaySound();
263 }
264 }
265 else
266 {
267 helper.PlaySound();
268 }
269 }
270 }
