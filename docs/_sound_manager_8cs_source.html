6/// <summary>
7/// Centralized audio service for background music and pooled sound-effect playback.
8/// </summary>
9/// <remarks>
10/// @ingroup audio_mgr
11/// @brief Manages looping music, pooled one-shot SFX, looping SFX, and short continuation windows for repeated sounds.
12///
13/// The manager owns one music <see cref="AudioSource"/> and a pool of <see cref="SoundManagerHelper"/>
14/// objects used for sound effects.
15///
16/// Behaviour:
17/// - Background music is played through a dedicated looping audio source.
18/// - Non-looping SFX are not mapped by clip, so repeated one-shots can overlap.
19/// - Looping SFX are mapped by clip, so one looping helper is reused per clip.
20/// - Continued SFX are kept alive for a short duration and stopped unless continued again.
21/// - The manager follows the main camera so manager-position SFX play near the listener.
22///
23/// Threading:
24/// - Unity main thread only.
25/// </remarks>
26public class SoundManager : Singleton<SoundManager>

134 /// <summary>
135 /// Keeps this object at the listener position and updates SFX continuation countdowns.
136 /// </summary>
137 private void Update()
138 {
139 if (_camera == null) _camera = Camera.main;
140 if (_camera != null) transform.position = _camera.transform.position;
141
142 _toBeRemoved.Clear();
143 _keys = _keys ?? new List<AudioClip>(4);
144 _keys.Clear();
145 _keys.AddRange(_toBeContinued.Keys);
146
147 for (int i = 0; i < _keys.Count; i++)
148 {
149 var clip = _keys[i];
150 float t = _toBeContinued[clip] - Time.deltaTime;
151 if (t > 0f)
152 {
153 _toBeContinued[clip] = t;
154 }
155 else
156 {
157 if (_sfxMapping.TryGetValue(clip, out var helper) && helper != null)
158 {
159 helper.StopSound();
160 _sfxMapping.Remove(clip);
161 }
162
163 _toBeRemoved.Add(clip);
164 }
165 }
166
167 for (int i = 0; i < _toBeRemoved.Count; i++)
168 _toBeContinued.Remove(_toBeRemoved[i]);
169 }

179 /// <remarks>

191 {
192 _musicSource.clip = clip;
193 _musicSource.Play();
194 }
195 else if (!_musicSource.isPlaying)

197 _musicSource.Play();
198 }

224 /// <summary>
225 /// Plays a sound effect through the pooled helper system.
226 /// </summary>
227 /// <param name="clip">Clip to play.</param>
228 /// <param name="target">Target transform followed by the helper during playback.</param>
229 /// <param name="volume">Per-call volume multiplier.</param>
230 /// <param name="pitch">Per-call pitch multiplier.</param>
231 /// <param name="loop">Whether the sound effect should loop.</param>
232 /// <param name="spatialBlend">Spatial blend in the range 0 to 1. A value of 0 is 2D; a value of 1 is fully 3D.</param>
233 /// <param name="reset">
234 /// Whether an already-playing mapped clip should be reset and restarted.
235 /// If false, an already-playing mapped clip continues uninterrupted.
236 /// </param>
237 /// <remarks>
238 /// Non-looping sounds always use a free helper and are allowed to overlap. Looping sounds are mapped by clip
239 /// so repeated calls reuse the same helper.
240 /// </remarks>
241 public void PlaySFXClip(AudioClip clip, Transform target, float volume = 1f, float pitch = 1f, bool loop = false, float spatialBlend = 0.0f, bool reset = true)
242 {
243 if (clip == null)
244 {
245 return;
246 }
247
248 float finalVolume = Mathf.Clamp01(volume * sfxVolume);
249
250 // One-shot sounds, such as crashes, should NOT be mapped by clip.
251 // Otherwise the same crash clip cuts/restarts itself.
252 if (!loop)
253 {
254 SoundManagerHelper oneShotHelper = GetFreeHelper();
255 SetUpHelper(oneShotHelper, clip, target, finalVolume, pitch, false, spatialBlend);
256 oneShotHelper.PlaySound();
257 return;
258 }
259
260 // Looping sounds still use mapping, because engine/continuous sounds
261 // should usually have one active source per clip.
262 if (!_sfxMapping.TryGetValue(clip, out SoundManagerHelper helper) || helper == null)
263 {
264 helper = GetFreeHelper();
265 _sfxMapping[clip] = helper;
266
267 SetUpHelper(helper, clip, target, finalVolume, pitch, true, spatialBlend);
268 helper.PlaySound();
269 return;
270 }
271
272 SetUpHelper(helper, clip, target, finalVolume, pitch, true, spatialBlend);
273
274 if (helper.IsPlaying)
275 {
276 if (reset)
277 {
278 helper.ResetSound();
279 helper.PlaySound();
280 }
281 }
282 else
283 {
284 helper.PlaySound();
285 }
286 }

422 /// <summary>
423 /// Gets an unused helper from the pool, or creates a new helper when none are free.
424 /// </summary>
425 /// <returns>Free helper ready for setup.</returns>
426 /// <remarks>
427 /// Any stale clip mappings pointing to a reused helper are removed before reuse.
428 /// </remarks>
429 private SoundManagerHelper GetFreeHelper()
430 {
431 for (int i = 0; i < _sfxSources.Count; i++)
432 {
433 SoundManagerHelper helper = _sfxSources[i];
434
435 if (helper != null && !helper.IsUsed)
436 {
437 RemoveMappingsForHelper(helper);
438 return helper;
439 }
440 }
441
442 SoundManagerHelper newHelper = new GameObject("SFXHelper").AddComponent<SoundManagerHelper>();
443 _sfxSources.Add(newHelper);
444 return newHelper;
445 }
446
447 /// <summary>
448 /// Removes all clip mappings that point to the given helper.
449 /// </summary>
450 /// <param name="helper">Helper whose previous mappings should be removed.</param>
451 /// <remarks>
452 /// This prevents stale mappings when a pooled helper is reused for another sound.
453 /// </remarks>
454 private void RemoveMappingsForHelper(SoundManagerHelper helper)
455 {
456 if (helper == null)
457 {
458 return;
459 }
460
461 _keys = _keys ?? new List<AudioClip>(4);
462 _keys.Clear();
463 _keys.AddRange(_sfxMapping.Keys);
464
465 for (int i = 0; i < _keys.Count; i++)
466 {
467 AudioClip clip = _keys[i];
468
469 if (_sfxMapping.TryGetValue(clip, out SoundManagerHelper mappedHelper) &&
470 mappedHelper == helper)
471 {
472 _sfxMapping.Remove(clip);
473 }
474 }
475 }
