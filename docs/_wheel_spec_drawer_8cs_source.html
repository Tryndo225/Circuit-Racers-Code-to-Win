6/// <summary>
7/// Compact 1-row drawer for <see cref="WheelSpec"/> showing collider, visual, powered, and steering columns.
8/// </summary>
9/// <remarks>
10/// @ingroup editor_util
11/// Purpose:
12/// - Presents a dense, horizontal layout for the four fields of <see cref="WheelSpec"/> in the Inspector.
13/// - Keeps the footprint to a single line per element for faster wheel setup.
14///
15/// Behavior:
16/// - Lays out four equal-width columns with a small padding: Collider | Visual | Powered | Steering.
17/// - Does not draw field labels (uses <see cref="GUIContent.none"/>); rely on column order and tooltips from Unity.
18///
19/// Threading:
20/// - Unity Editor thread only; compiled out of player builds with <c>#if UNITY_EDITOR</c>.
21///
22/// Notes:
23/// - Intended for use inside arrays or structs on vehicle setup components.
24/// - Pairs well with a header/tooltip on the parent field describing the column order.
25/// </remarks>
26[CustomPropertyDrawer(typeof(WheelSpec))]
27public class WheelSpecDrawer : PropertyDrawer
28{
29 /// <summary>
30 /// Returns a fixed one-line height plus a small padding to separate rows visually.
31 /// </summary>
32 /// <param name="property">Property being drawn.</param>
33 /// <param name="label">GUI label.</param>
34 /// <returns>Pixel height for this drawer.</returns>
35 public override float GetPropertyHeight(SerializedProperty property, GUIContent label)
36 {
37 return EditorGUIUtility.singleLineHeight + 4f;
38 }
39
40 /// <summary>
41 /// Renders the four <see cref="WheelSpec"/> fields in a single row: collider, visual, powered, steering.
42 /// </summary>
43 /// <param name="position">Target rect to draw within.</param>
44 /// <param name="property">Root serialized property (a <see cref="WheelSpec"/> instance).</param>
45 /// <param name="label">GUI label (unused; columns omit labels).</param>
46 public override void OnGUI(Rect position, SerializedProperty property, GUIContent label)
47 {
48 EditorGUI.BeginProperty(position, label, property);
49 position = EditorGUI.IndentedRect(position);
50
51 float pad = 4f;
52 int cols = 4;
53 float w = (position.width - pad * (cols - 1)) / cols;
54
55 var r0 = new Rect(position.x + (w + pad) * 0, position.y, w, EditorGUIUtility.singleLineHeight);
56 var r1 = new Rect(position.x + (w + pad) * 1, position.y, w, EditorGUIUtility.singleLineHeight);
57 var r2 = new Rect(position.x + (w + pad) * 2, position.y, w, EditorGUIUtility.singleLineHeight);
58 var r3 = new Rect(position.x + (w + pad) * 3, position.y, w, EditorGUIUtility.singleLineHeight);
59
60 EditorGUI.PropertyField(r0, property.FindPropertyRelative("collider"), GUIContent.none);
61 EditorGUI.PropertyField(r1, property.FindPropertyRelative("visual"), GUIContent.none);
62 EditorGUI.PropertyField(r2, property.FindPropertyRelative("powered"), GUIContent.none);
63 EditorGUI.PropertyField(r3, property.FindPropertyRelative("steering"), GUIContent.none);
64
65 EditorGUI.EndProperty();
66 }
67}
