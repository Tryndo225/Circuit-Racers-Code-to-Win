5/// <summary>
6/// Controls vehicle light groups and state-dependent light effects.
7/// </summary>
8/// <remarks>
9/// @ingroup car_ctrl
10/// @brief Manages front, day, rear, reverse, and brake lights with optional fading and shared-bulb restoration.
11///
12/// The controller supports:
13/// - Toggling front, day, and rear lights as a group.
14/// - Fading normal light groups on and off.
15/// - Instantly enabling brake and reverse lights.
16/// - Reusing rear light bulbs for brake or reverse lights.
17/// - Restoring rear light color and intensity when shared brake/reverse lights are deactivated.
18///
19/// Lists may be empty and may contain null entries. Null lights are skipped.
20///
21/// Threading:
22/// - Unity main thread only.
23/// - Uses Unity coroutines for fading.
24/// </remarks>

244 /// If a reverse light is also listed as a rear light, turning reverse off restores the rear-light
245 /// color and intensity when the normal lights are currently enabled.
246 /// </remarks>
247 public void SetReverseLights(bool active)
248 {
249 if (_isReversing == active) return;
250
251 _isReversing = active;
252
253 foreach (var light in reverseLights)
254 {
255 if (light != null)
256 {
257 if (rearLights.Contains(light) && _lightsToggle && !active)
258 {
259 light.color = rearLightsColor;
260 light.intensity = rearLightsIntensity;
261 }
262 else
263 {
264 light.color = reverseLightsColor;
265 light.intensity = active ? reverseLightsIntensity : 0f;
266 }
267 }
268 }
269
270 if (!_isReversing)
271 SetLights(_lightsToggle);
272 }

310 /// <summary>
311 /// Fades a light from its current intensity to a target intensity.
312 /// </summary>
313 /// <param name="light">Light to fade.</param>
314 /// <param name="targetIntensity">Target intensity.</param>
315 /// <returns>Coroutine enumerator.</returns>
316 private IEnumerator FadeLight(Light light, float targetIntensity)
317 {
318 float startIntensity = light.intensity;
319 float elapsedTime = 0f;
320 while (elapsedTime < fadeDuration)
321 {
322 elapsedTime += Time.deltaTime;
323 light.intensity = Mathf.Lerp(startIntensity, targetIntensity, elapsedTime / fadeDuration);
324 yield return null;
325 }
326
327 light.intensity = targetIntensity;
328 }
