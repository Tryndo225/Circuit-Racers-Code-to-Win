249 /// <summary>
250 /// Fades a light's intensity from its current value to targetIntensity over fadeDuration seconds.
251 /// </summary>
252 /// <param name="light">Light to fade.</param>
253 /// <param name="targetIntensity">Target intensity.</param>
254 /// <returns>Coroutine enumerator.</returns>
255 private IEnumerator FadeLight(Light light, float targetIntensity)
256 {
257 float startIntensity = light.intensity;
258 float elapsedTime = 0f;
259 while (elapsedTime < fadeDuration)
260 {
261 elapsedTime += Time.deltaTime;
262 light.intensity = Mathf.Lerp(startIntensity, targetIntensity, elapsedTime / fadeDuration);
263 yield return null;
264 }
265
266 light.intensity = targetIntensity;
267 }
