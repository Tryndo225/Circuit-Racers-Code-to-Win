228 /// <summary>Returns average RPM of driven wheels that are grounded.</summary>
229 private float AverageDrivenWheelRPM()
230 {
231 float sum = 0f;
232 int count = 0;
233 for (int i = 0; i < _wheelColliders.Length; i++)
234 {
235 if (!_driven[i]) continue;
236
237 WheelHit hit;
238 if (_wheelColliders[i].GetGroundHit(out hit))
239 {
240 sum += _wheelColliders[i].rpm;
241 count++;
242 }
243 }
244
245 return (count > 0) ? (sum / count) : 0f;
246 }
247
248 /// <summary>Returns average combined slip (abs(forwardSlip + sidewaysSlip)) for grounded driven wheels.</summary>
249 private float AverageDrivenWheelSlip()
250 {
251 float sum = 0f;
252 int count = 0;
253 for (int i = 0; i < _wheelColliders.Length; i++)
254 {
255 if (!_driven[i]) continue;
256 WheelHit hit;
257
258 if (_wheelColliders[i].GetGroundHit(out hit))
259 {
260 sum += Mathf.Abs(hit.forwardSlip + hit.sidewaysSlip);
261 count++;
262 }
263 }
264 return (count > 0) ? (sum / count) : 0f;
265 }

278 /// <summary>
279 /// Applies steering input. If gamepadSteering is true, applies exponent shaping.
280 /// Otherwise, limits max angle by speed and rate-limits by steerSpeedDegPerSec.
281 /// </summary>
282 private void ControlSteering(float steerInput, bool gamepadSteering)
283 {
284 if (gamepadSteering)
285 {
286 float shaped = Mathf.Sign(steerInput) * Mathf.Pow(Mathf.Abs(steerInput), inputExponent);
287 ApplySteering(steerInput); // using original variable as in provided code (logic preserved)
288 }
289 else
290 {
291 float kph = CalculateSpeed();
292 float speedCoeficient = Mathf.InverseLerp(0f, maxSpeed, kph);
293 float maxAngle = Mathf.Lerp(maxSteerAngle, maxSteerAngleAtTopSpeed, speedCoeficient);
294 float targetAngle = steerInput * maxAngle;
295
296 steeringAngle = Mathf.MoveTowards(steeringAngle, targetAngle, steerSpeedDegPerSec * Time.fixedDeltaTime);
297
298 ApplySteering(steeringAngle);
299 }
300 }
301
302 /// <summary>Sets steerAngle on steering wheels, with optional Ackermann scaling.</summary>
303 private void ApplySteering(float steer)
304 {
305 for (int i = 0; i < _wheelColliders.Length; i++)
306 {
307 if (!_steering[i])
308 continue;
309
310 bool isLeft = (i % 2 == 0);
311 float angle = steer;
312
313 if (ackermannFactor != 1f)
314 angle *= isLeft ? (1f + 0.1f * ackermannFactor) : (1f - 0.1f * ackermannFactor);
315
316 _wheelColliders[i].steerAngle = angle;
317 }
318 }
319
320 /// <summary>
321 /// Updates reverse state. Allows reversing if braking and speed is low.
322 /// </summary>
323 private bool ReverseCheck(bool braking)
324 {
325 if (braking && CalculateSpeed() < 1f)
326 {
327 _isReversing = true;
328 return true;
329 }
330 else if (!braking && _isReversing)
331 {
332 _isReversing = false;
333 return false;
334 }

336 return _isReversing;

341 #region Private Helpers: Drive, Braking, Mesh Sync

343 /// <summary>
344 /// Applies motor torque to driven wheels, limiting by max speeds and transmission shifting,
345 /// and reducing torque when traction control detects excessive slip.
346 /// </summary>
347 private void ApplyDrive(float throttle)
348 {
349 if (CalculateSpeed() > maxSpeed
350 || (_isReversing && CalculateSpeed() > maxReverseSpeed)
351 || _transmissionController.HandleShifting(AverageDrivenWheelRPM(), AverageDrivenWheelSlip()))
352 {
353 throttle = 0f;
354 }
355
356 for (int i = 0; i < _driven.Length; i++)
357 {
358 if (!_driven[i])
359 continue;
360
361 var wc = _wheelColliders[i];
362 float motor = 0f;
363
364 WheelHit hit;
365 bool grounded = wc.GetGroundHit(out hit);
366 float targetTorque = maxMotorPower / _driven.Length * throttle;
367
368 if (grounded && Mathf.Abs(hit.forwardSlip) > tractionSlipLimit && tractionControlEnabled)
369 targetTorque *= 0.5f;
370
371 motor = targetTorque;
372 wc.motorTorque = motor;
373 }
374 }
375
376 /// <summary>
377 /// Applies service brake and handbrake torques per wheel. ABS reduces brake torque when slip is high.
378 /// </summary>
379 private void ApplyBraking(bool braking, bool handbrake)
380 {
381 for (int i = 0; i < _wheelColliders.Length; i++)
382 {
383 var wc = _wheelColliders[i];
384 var brake = 0f;
385
386 WheelHit hit;
387 bool grounded = wc.GetGroundHit(out hit);
388
389 if (braking)
390 {
391 float targetBrake = maxBrakeTorque;
392
393 if (grounded && Mathf.Abs(hit.forwardSlip) > absSlipLimit && absEnabled)
394 targetBrake *= 0.6f;
395
396 brake = targetBrake;
397 }
398
399 if (handbrake && i >= 2)
400 brake = Mathf.Max(brake, handbrakeTorque);
401
402 wc.brakeTorque = brake;
403 }
404 }

418 /// <summary>Applies anti-roll forces to front and rear axle pairs.</summary>
419 private void ApplyAntiRoll()
420 {
421 ApplyAntiRollPair(0, 1, antiRollStiffnessFront);
422 ApplyAntiRollPair(2, 3, antiRollStiffnessRear);
423 }

425 /// <summary>Applies anti-roll force between a left-right wheel pair.</summary>
426 private void ApplyAntiRollPair(int leftIndex, int rightIndex, float stiffness)
427 {
428 WheelCollider wl = _wheelColliders[leftIndex];
429 WheelCollider wr = _wheelColliders[rightIndex];
430
431 float travelL = GetSuspensionTravel(wl);
432 float travelR = GetSuspensionTravel(wr);
433
434 float antiRollForce = (travelL - travelR) * stiffness;
435
436 if (travelL > -1f)
437 _carRigidBody.AddForceAtPosition(wl.transform.up * -antiRollForce, wl.transform.position, ForceMode.Force);
438
439 if (travelR > -1f)
440 _carRigidBody.AddForceAtPosition(wr.transform.up * antiRollForce, wr.transform.position, ForceMode.Force);
441 }
442
443 /// <summary>Computes normalized suspension travel for a wheel in [0,1], or -1 if not grounded.</summary>
444 private float GetSuspensionTravel(WheelCollider wc)
445 {
446 WheelHit hit;
447
448 if (wc.GetGroundHit(out hit))
449 return (-wc.transform.InverseTransformPoint(hit.point).y - wc.radius) / wc.suspensionDistance;
450
451 return -1f;
452 }
