87 /// <summary>
88 /// Attempts to find and play a music clip for the given scene using the configured list.
89 /// </summary>
90 /// <param name="scene">Scene helper used for name matching.</param>
91 /// <remarks>
92 /// A match is determined by equality against the pair's scene. Only the first match triggers music.
93 /// Requires SoundManager.Instance to be present in the scene.
94 /// </remarks>
95 private void MatchMusicClip(SceneAssetHelper scene)
96 {
97 for (int i = 0; i < sceneAudioClipPairs.Count; i++)
98 {
99 if (scene == sceneAudioClipPairs[i])
100 {
101 // Defensive: SoundManager may not yet have initialized in edge cases.
102 if (SoundManager.Instance != null)
103 {
104 SoundManager.Instance.PlayMusic(sceneAudioClipPairs[i].AudioClip);
105 }
106 else
107 {
108 Debug.LogWarning("SoundManager.Instance is null; cannot play scene music.");
109 }
110
111 // Stop after first match to avoid overlapping PlayMusic calls.
112 return;
113 }
114 }
115 }
