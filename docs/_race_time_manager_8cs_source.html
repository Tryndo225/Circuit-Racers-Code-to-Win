5/// <summary>
6/// Manages race timing, lap timing, checkpoint splits, and race completion.
7/// </summary>
8/// <remarks>
9/// @ingroup track_mng
10/// @brief Tracks race progress times and forwards completed race data to replay and game-data systems.
11///
12/// This manager records:
13/// - Race start time.
14/// - Lap start and lap durations.
15/// - Checkpoint split times.
16/// - Final race time.
17///
18/// It also connects checkpoint events to split recording, starts replay recording when the race begins,
19/// saves the replay when the race ends, and passes completed race results to <see cref="GameDataManager"/>.
20/// </remarks>

100 /// <summary>
101 /// Records a checkpoint split for the current race.
102 /// </summary>
103 /// <remarks>
104 /// The split is displayed through <see cref="RaceOverlay"/>. When previous saved split data exists,
105 /// the displayed difference compares the current split against the stored split for the same checkpoint index.
106 /// </remarks>
107 public void CheckpointPassed()
108 {
109 if (IsRaceFinished)
110 return;
111
112 float currentCpSplit = GetCurrentRaceTime();
113 CheckPointSplitsTimes.Add(currentCpSplit);
114
115 float lastSplitTime = GameDataManager.Instance.GetCurrentMapSplit(CheckPointSplitsTimes.Count - 1);
116
117 if (lastSplitTime == 0)
118 RaceOverlay.Instance.DisplaySplit(currentCpSplit, 0);
119 else
120 RaceOverlay.Instance.DisplaySplit(currentCpSplit, currentCpSplit - lastSplitTime);
121 }

145 /// <summary>
146 /// Finishes the race, saves race results, and returns the final race time.
147 /// </summary>
148 /// <returns>Total race duration in seconds.</returns>
149 /// <remarks>
150 /// Records the final lap time, saves the replay through <see cref="ReplayManager"/>,
151 /// and sends the completed result to <see cref="GameDataManager"/>.
152 /// </remarks>
153 public float RaceEnd()
154 {
155 if (LapStartTime > 0)
156 {
157 LapTimes.Add(Time.time - LapStartTime);
158 }
159 float totalRaceTime = Time.time - RaceStartTime;
160 Debug.Log($"Race finished! Total time: {totalRaceTime} seconds");
161 Replay replay = ReplayManager.Instance.SaveReplay();
162 GameDataManager.Instance.CompleteLevel(totalRaceTime, CheckPointSplitsTimes.ToArray(), replay);
163 RaceEndTime = totalRaceTime;
164 return totalRaceTime;
165 }
