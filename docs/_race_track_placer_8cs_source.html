8/// <summary>
9/// Describes a single placeable track piece: prefab, world-space pose, pattern size, and source pattern.
10/// </summary>
11/// <remarks>
12/// @ingroup level_gen
13/// @brief Stores the prefab and placement data used by <see cref="RaceTrackPlacer"/>.
14///
15/// In the generated legend, <see cref="Position"/> is used as a local placement offset.
16/// During track placement, the placer converts it into the final world-space position.
17///
18/// <see cref="PatternKey"/> stores the flattened pattern that produced the piece. This is used
19/// for larger pieces where the matching pattern is larger than the actual occupied footprint.
20/// </remarks>
21[Serializable]
22public struct TrackPiece
23{
24 /// <summary>
25 /// Prefab to instantiate for this track cell.
26 /// </summary>
27 [Tooltip("Prefab to instantiate for this track cell.")]
28 public GameObject Prefab;
29
30 /// <summary>
31 /// World-space position used when placing the prefab.
32 /// </summary>
33 /// <remarks>
34 /// In the generated rule legend this value is treated as a local offset. During placement,
35 /// the final world-space position is filled based on the grid coordinate and block size.
36 /// </remarks>
37 [Tooltip("Placement position. In rules this acts as a local offset; during placement it becomes the final world-space position.")]
38 public Vector3 Position;
39
40 /// <summary>
41 /// World-space rotation used when placing the prefab.
42 /// </summary>
43 [Tooltip("World-space rotation used when placing the prefab.")]
44 public Quaternion Rotation;
45
46 /// <summary>
47 /// Pattern size used by this piece, usually 3 or 5.
48 /// </summary>
49 /// <remarks>
50 /// The pattern size includes the matching border. For example, a 5x5 pattern may describe
51 /// a larger prefab whose actual occupied footprint is the inner 3x3 area.
52 /// </remarks>
53 [Tooltip("Pattern size used by this piece, usually 3 or 5.")]
54 public int Size;
55
56 /// <summary>
57 /// Flattened pattern key that produced this track piece.
58 /// </summary>
59 /// <remarks>
60 /// This is used by <see cref="RaceTrackPlacer"/> to decide which cells are actually covered
61 /// by larger pieces. Without this, a large bend would remove the whole surrounding square,
62 /// including road tiles that are not part of the bend.
63 /// </remarks>
64 [Tooltip("Generated flattened pattern key that produced this track piece.")]
65 public string PatternKey;
66
67 /// <summary>
68 /// Creates a new <see cref="TrackPiece"/> with explicit prefab, pose, pattern size, and source pattern.
69 /// </summary>
70 /// <param name="prefab">Prefab reference. May be null for an empty/default legend entry.</param>
71 /// <param name="position">Placement position or local placement offset.</param>
72 /// <param name="rotation">Placement rotation.</param>
73 /// <param name="size">Pattern size used by this piece.</param>
74 /// <param name="patternKey">Flattened pattern key that produced this piece.</param>
75 public TrackPiece(GameObject prefab, Vector3 position, Quaternion rotation, int size = 3, string patternKey = "")
76 {
77 Prefab = prefab;
78 Position = position;
79 Rotation = rotation;
80 Size = size;
81 PatternKey = patternKey;
82 }

206 /// <summary>
207 /// Converts the visual pattern into a compact flattened string key.
208 /// </summary>
209 /// <returns>Pattern string using <c>X</c>, <c>1</c>, and <c>C</c>.</returns>
210 public string ToPatternKey()
211 {
212 NormalizePatternLength();
213
214 StringBuilder builder = new StringBuilder(Size * Size);
215
216 for (int i = 0; i < Pattern.Length; i++)
217 {
218 builder.Append(Pattern[i] switch
219 {
220 TrackPatternCell.Track => '1',
221 TrackPatternCell.Checkpoint => 'C',
222 _ => 'X'
223 });
224 }
225
226 return builder.ToString();
227 }
228
229 /// <summary>
230 /// Ensures that the pattern size is valid and that the pattern array has the correct length.
231 /// </summary>
232 /// <remarks>
233 /// Pattern sizes must be odd so that there is a clear center cell.
234 /// </remarks>
235 public void NormalizePatternLength()
236 {
237 if (Size < 3)
238 Size = 3;
239
240 if (Size % 2 == 0)
241 Size += 1;
242
243 int requiredLength = Size * Size;
244
245 if (Pattern == null)
246 {
247 Pattern = new TrackPatternCell[requiredLength];
248 return;
249 }
250
251 if (Pattern.Length != requiredLength)
252 Array.Resize(ref Pattern, requiredLength);
253 }
254
255 /// <summary>
256 /// Converts a compact pattern string into visual pattern cells.
257 /// </summary>
258 /// <param name="key">Flattened pattern string using <c>X</c>, <c>1</c>, and <c>C</c>.</param>
259 /// <param name="size">Square pattern size.</param>
260 /// <returns>Pattern cell array.</returns>
261 private static TrackPatternCell[] PatternFromKey(string key, int size)
262 {
263 int length = size * size;
264 TrackPatternCell[] pattern = new TrackPatternCell[length];
265
266 for (int i = 0; i < length; i++)
267 {
268 char c = i < key.Length ? key[i] : 'X';
269
270 pattern[i] = c switch
271 {
272 '1' => TrackPatternCell.Track,
273 'C' => TrackPatternCell.Checkpoint,
274 _ => TrackPatternCell.Empty
275 };
276 }
277
278 return pattern;
279 }
280}

584 /// <summary>
585 /// Rebuilds the string-keyed legend from the visual <see cref="trackPieceRules"/> list.
586 /// </summary>
587 /// <param name="logDuplicates">Whether duplicate pattern keys should be reported to the console.</param>
588 /// <remarks>
589 /// This method is used both by <see cref="OnValidate"/> and <see cref="Start"/>. The generated
590 /// dictionary remains serialized and visible in the Inspector, but the visual rule list is the
591 /// source of truth.
592 /// </remarks>
593 private void RebuildTrackPieceLegend(bool logDuplicates)
594 {
595 if (trackPieceLegend == null)
596 trackPieceLegend = new StringTrackPieceDictionary();
597
598 trackPieceLegend.Clear();
599
600 if (trackPieceRules == null)
601 return;
602
603 foreach (TrackPieceRule rule in trackPieceRules)
604 {
605 if (rule == null)
606 continue;
607
608 rule.NormalizePatternLength();
609
610 string key = rule.ToPatternKey();
611 TrackPiece piece = rule.ToTrackPiece();
612
613 if (trackPieceLegend.ContainsKey(key) && logDuplicates)
614 {
615 Debug.LogWarning($"[RaceTrackPlacer] Duplicate track-piece pattern found: {key}. The later rule will overwrite the earlier one.");
616 }
617
618 trackPieceLegend[key] = piece;
619 }
620 }

630 {

648 /// <summary>
649 /// Creates a placement map, walks the track from the start point, instantiates prefabs,
650 /// registers checkpoints, and starts either race or replay mode.
651 /// </summary>
652 private void PlaceTrackPieces()
653 {
654 if (levelMap == null || levelMap.Tiles == null)
655 {
656 Debug.LogError("LevelMap or its Tiles are not set.");
657 return;
658 }
659
660 var piecesToPlace = CreateTrack();
661
662 Coordinates? position = levelMap.StartPoint;
663 HashSet<Coordinates> visited = new HashSet<Coordinates>();
664
665 int checkpointAmount = 0;
666
667 while (true)
668 {
669 if (position == null)
670 break;
671
672 if (piecesToPlace[position.Value.X, position.Value.Y].Prefab != null && piecesToPlace[position.Value.X, position.Value.Y].Prefab != traversalPrefab)
673 {
674 var newGameObject = Instantiate(piecesToPlace[position.Value.X, position.Value.Y].Prefab, piecesToPlace[position.Value.X, position.Value.Y].Position, piecesToPlace[position.Value.X, position.Value.Y].Rotation, this.transform);
675
676 if (IsCheckpointTile(position.Value))
677 {
678 CheckPointListener checkpoint = newGameObject.GetComponentInChildren<CheckPointListener>();
679
680 if (checkpoint == null)
681 {
682 Debug.LogError($"[RaceTrackPlacer] Checkpoint tile at {position.Value} was placed, but prefab has no CheckPointListener.");
683 }
684 else
685 {
686 checkpoint.CheckpointOrder = checkpointAmount;
687 ++checkpointAmount;
688 }
689
690 if (IsStartTile(position.Value) && !isBuildForReplay)
691 TrackManager.Instance.CarSpawn = newGameObject.transform;
692 }
693 }
694
695 visited.Add(position.Value);
696 position = Step(piecesToPlace, position.Value, visited);
697 }
698
699 CheckPointManager.Instance.ClearCheckPoints();
700 CheckPointManager.Instance.AutoAddCheckpoints();
701
702 if (!isBuildForReplay)
703 {
704 TrackManager.Instance.StartRace(levelMap);
705 }
706 else
707 {
708 Debug.Log("Building finished setting Replay");
709 ReplayPreviewer.Instance.SetReplay(GameDataManager.Instance.CurrentGameData.GetBestReplay(GameDataManager.Instance.CurrentLevelMap));
710 Debug.Log("Replay Set.");
711 }
712 }

714 /// <summary>
715 /// Replaces cells covered by larger placed pieces with traversal placeholders to avoid visual overlap.
716 /// </summary>
717 /// <param name="pieces">Placement map to adjust.</param>
718 /// <remarks>
719 /// Larger pieces use a pattern that is one layer larger than their physical footprint. For example,
720 /// a 5x5 pattern can represent a piece whose actual footprint is the inner 3x3 area.
721 ///
722 /// This method therefore does not remove the whole surrounding square. Instead, it removes only
723 /// cells that are part of the inner footprint and are marked as covered by the placed piece's
724 /// <see cref="TrackPiece.PatternKey"/>.
725 /// </remarks>
726 private void StopOverlaying(TrackPiece[,] pieces)
727 {
728 List<Coordinates> largePiecePositions = new List<Coordinates>();
729
730 for (int x = 0; x < pieces.GetLength(0); x++)
731 {
732 for (int y = 0; y < pieces.GetLength(1); y++)
733 {
734 if (pieces[x, y].Prefab == null || pieces[x, y].Prefab == traversalPrefab)
735 continue;
736
737 int coveredArea = GetCoveredAreaRadius(pieces[x, y].Size);
738
739 if (coveredArea > 0)
740 {
741 largePiecePositions.Add(new Coordinates(x, y));
742 }
743 }
744 }
745
746 foreach (Coordinates position in largePiecePositions)
747 {
748 if (!pieces.InBounds(position.X, position.Y))
749 continue;
750
751 if (pieces[position.X, position.Y].Prefab == null || pieces[position.X, position.Y].Prefab == traversalPrefab)
752 continue;
753
754 RemoveAllSurrounding(position, pieces, pieces[position.X, position.Y]);
755 }
756 }

852 /// <summary>
853 /// Builds a flattened pattern string centered at the given coordinate.
854 /// </summary>
855 /// <param name="center">Center coordinate of the pattern.</param>
856 /// <param name="size">Pattern width and height.</param>
857 /// <param name="allowOverlap">Whether internally used cells may count as track.</param>
858 /// <returns>Flattened pattern string.</returns>
859 /// <remarks>
860 /// Pattern symbols:
861 /// - <c>1</c> marks track-compatible cells.
862 /// - <c>X</c> marks empty cells.
863 /// - <c>C</c> marks checkpoint, start, or finish cells when they must be matched explicitly.
864 ///
865 /// For 3x3 patterns, checkpoint/start/finish neighbours are treated as normal track cells. This
866 /// allows ordinary road pieces to be placed next to checkpoints without needing extra pattern variants.
867 /// The center checkpoint/start/finish cell is still represented as <c>C</c>, so dedicated checkpoint
868 /// rules can be matched.
869 ///
870 /// For larger patterns, checkpoint/start/finish cells are represented as <c>C</c> even when they are
871 /// neighbours. This prevents larger prefabs, such as 5x5 turns, from being placed over checkpoint,
872 /// start, or finish tiles unless a rule explicitly supports such a case.
873 ///
874 /// Isolated single track symbols are converted to <c>X</c> to avoid stray matches.
875 /// </remarks>
876 private string ExtractPattern(Coordinates center, int size, bool allowOverlap)
877 {
878 int halfSize = size / 2;
879 char[,] pattern = new char[size, size];
880
881 for (int dy = -halfSize; dy <= halfSize; dy++)
882 {
883 for (int dx = -halfSize; dx <= halfSize; dx++)
884 {
885 int x = center.X + dx;
886 int y = center.Y + dy;
887
888 int px = dx + halfSize;
889 int py = dy + halfSize;
890
891 char c = 'X';
892
893 if (dx != 0 && dy != 0)
894 {
895 c = 'X';
896 }
897 else if (levelMap.Tiles.InBounds(x, y))
898 {
899 bool isCheckpointLike = levelMap.Tiles[x, y] == CheckpointTile || (x == levelMap.StartPoint.X && y == levelMap.StartPoint.Y) || (x == levelMap.FinishPoint.X && y == levelMap.FinishPoint.Y);
900
901 bool isCenter = dx == 0 && dy == 0;
902
903 if (isCheckpointLike)
904 {
905 if (isCenter)
906 {
907 c = 'C';
908 }
909 else if (halfSize > 1)
910 {
911 c = 'C';
912 }
913 else
914 {
915 c = '1';
916 }
917 }
918 else if (levelMap.Tiles[x, y] == TrackTile || allowOverlap && levelMap.Tiles[x, y] == UsedTileValue)
919 {
920 c = '1';
921 }
922 }
923
924 pattern[py, px] = c;
925 }
926 }
927
928 for (int y = 0; y < size; y++)
929 {
930 for (int x = 0; x < size; x++)
931 {
932 if (pattern[y, x] == '1' && IsAlone(new Coordinates(y, x), pattern))
933 {
934 pattern[y, x] = 'X';
935 }
936 }
937 }
938
939 patternBuilder.Clear();
940
941 for (int y = 0; y < size; y++)
942 {
943 for (int x = 0; x < size; x++)
944 {
945 patternBuilder.Append(pattern[y, x]);
946 }
947 }
948
949 return patternBuilder.ToString();
950 }
