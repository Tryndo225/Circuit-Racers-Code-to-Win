73/// <summary>
74/// Procedural track placer that scans a <see cref="LevelMap"/>, matches local tile patterns,
75/// and instantiates track-piece prefabs.
76/// </summary>

78/// @ingroup level_gen
79/// @brief Builds the visible track scene from a grid-based level map.

81/// The placer reads the selected level from <see cref="GameDataManager.CurrentLevelMap"/>,

288 /// <summary>
289 /// Creates a placement map, walks the track from the start point, instantiates prefabs,
290 /// registers checkpoints, and starts either race or replay mode.
291 /// </summary>
292 private void PlaceTrackPieces()
293 {
294 if (levelMap == null || levelMap.Tiles == null)
295 {
296 Debug.LogError("LevelMap or its Tiles are not set.");
297 return;
298 }
299 var piecesToPlace = CreateTrack();
300
301 Coordinates? position = levelMap.StartPoint;
302 HashSet<Coordinates> visited = new HashSet<Coordinates>();
303
304 int checkpointAmount = 0;
305
306 while (true)
307 {
308 if (position == null)
309 break;
310
311 if (piecesToPlace[position.Value.X, position.Value.Y].Prefab != null && piecesToPlace[position.Value.X, position.Value.Y].Prefab != traversalPrefab)
312 {
313 var newGameObject = Instantiate(piecesToPlace[position.Value.X, position.Value.Y].Prefab, piecesToPlace[position.Value.X, position.Value.Y].Position, piecesToPlace[position.Value.X, position.Value.Y].Rotation, this.transform);
314
315 if (IsCheckpointTile(position.Value))
316 {
317 CheckPointListener checkpoint = newGameObject.GetComponentInChildren<CheckPointListener>();
318
319 if (checkpoint == null)
320 {
321 Debug.LogError($"[RaceTrackPlacer] Checkpoint tile at {position.Value} was placed, but prefab has no CheckPointListener.");
322 }
323 else
324 {
325 checkpoint.CheckpointOrder = checkpointAmount;
326 ++checkpointAmount;
327 }
328
329 if (IsStartTile(position.Value) && !isBuildForReplay)
330 TrackManager.Instance.CarSpawn = newGameObject.transform;
331 }
332 }
333
334 visited.Add(position.Value);
335 position = Step(piecesToPlace, position.Value, visited);
336 }
337
338 CheckPointManager.Instance.ClearCheckPoints();
339 CheckPointManager.Instance.AutoAddCheckpoints();
340
341 if (!isBuildForReplay)
342 {
343 TrackManager.Instance.StartRace(levelMap);
344 }
345 else
346 {
347 Debug.Log("Building finished setting Replay");
348 ReplayPreviewer.Instance.SetReplay(GameDataManager.Instance.CurrentGameData.GetBestReplay(GameDataManager.Instance.CurrentLevelMap));
349 Debug.Log("Replay Set.");
350 }

378 /// <summary>
379 /// Chooses the next unvisited neighboring cell with a placeable piece.
380 /// </summary>
381 /// <param name="piecesToPlace">Placement map being traversed.</param>
382 /// <param name="position">Current track coordinate.</param>
383 /// <param name="visited">Already visited coordinates.</param>
384 /// <returns>Next coordinate, or <c>null</c> when no continuation exists.</returns>
385 private Coordinates? Step(TrackPiece[,] piecesToPlace, Coordinates position, HashSet<Coordinates> visited)
386 {
387 foreach (var offset in offsets)
388 {
389 var next = position + offset;
390 if (piecesToPlace.InBounds(next.X, next.Y) && piecesToPlace[next.X, next.Y].Prefab != null && !visited.Contains(next))
391 return next;
392 }
393
394 return null;
395 }

401 /// <summary>
402 /// Produces a 2D map of <see cref="TrackPiece"/> descriptors for track and checkpoint cells.
403 /// </summary>
404 /// <returns>Placement map containing track-piece descriptors.</returns>
405 private TrackPiece[,] CreateTrack()
406 {
407 TrackPiece[,] piecesToPlace = new TrackPiece[levelMap.Tiles.GetLength(0), levelMap.Tiles.GetLength(1)];
408 TrackPiece? trackPiece;
409 Coordinates coordinates;
410
411 foreach (int size in possibleSizes)
412 {
413 for (int x = 0; x < levelMap.Tiles.GetLength(0); x++)
414 {
415 for (int y = 0; y < levelMap.Tiles.GetLength(1); y++)
416 {
417 if (piecesToPlace[x, y].Prefab != null || !IsTrackOrCheckpointValue(levelMap.Tiles[x, y]))
418 continue;
419
420 coordinates = new Coordinates(x, y);
421 trackPiece = GetTrackPiece(coordinates, size, size == possibleSizes[possibleSizes.Count - 1]);
422
423 if (trackPiece != null && trackPiece.Value.Prefab != null)
424 {
425 MarkSpaceAsUsed(coordinates, size);
426 AddTraversalPlaceHolders(coordinates, size, piecesToPlace);
427 piecesToPlace[x, y] = trackPiece.Value;
428 }
429 }
430 }
431 }
432
433 return piecesToPlace;
434 }

438 /// <summary>
439 /// Extracts a local pattern, finds a matching legend entry, and returns a filled <see cref="TrackPiece"/>.
440 /// </summary>
441 /// <param name="coordinates">Center coordinate to match.</param>
442 /// <param name="size">Pattern size to extract.</param>
443 /// <param name="allowOverlap">Whether internally used cells may still count as track while matching.</param>
444 /// <returns>Matched track piece, or <c>null</c> if no matching pattern exists.</returns>
445 private TrackPiece? GetTrackPiece(Coordinates coordinates, int size, bool allowOverlap)
446 {
447 string pattern = ExtractPattern(coordinates, size, allowOverlap);
448 if (trackPieceLegend.TryGetValue(pattern, out TrackPiece trackPiece))
449 {
450 Vector3 positionOffset = transform.position + new Vector3(coordinates.X * blockOffset, 0f, coordinates.Y * blockOffset);
451 trackPiece.Position = positionOffset;
452
453 if (coordinates == levelMap.StartPoint || coordinates == levelMap.FinishPoint)
454 {
455 Debug.Log($"[RaceTrackPlacer] Assigning start/finish prefab at {coordinates}");
456 trackPiece.Prefab = levelMap.Circuit ? raceTrackStartFinishPrefab : raceTrackStartFinishP2pPrefab;
457 }
458 else if (levelMap.Tiles.At(coordinates) == CheckpointTile)
459 {
460 trackPiece.Prefab = checkpointPrefab;
461 }
462
463 return trackPiece;
464 }
465
466 return null;
467 }

469 /// <summary>
470 /// Builds a flattened pattern string centered at the given coordinate.
471 /// </summary>
472 /// <param name="center">Center coordinate of the pattern.</param>
473 /// <param name="size">Pattern width and height.</param>
474 /// <param name="allowOverlap">Whether internally used cells may count as track.</param>
475 /// <returns>Flattened pattern string.</returns>
476 /// <remarks>
477 /// Pattern symbols:
478 /// - <c>1</c> marks track-compatible cells.
479 /// - <c>X</c> marks empty cells.
480 /// - <c>C</c> marks checkpoints, start, or finish in larger kernels.
481 ///
482 /// Isolated single track symbols are converted to <c>X</c> to avoid stray matches.
483 /// </remarks>
484 private string ExtractPattern(Coordinates center, int size, bool allowOverlap)
485 {
486 int halfSize = size / 2;
487 char[,] pattern = new char[size, size];
488
489 for (int dy = -halfSize; dy <= halfSize; dy++)
490 {
491 for (int dx = -halfSize; dx <= halfSize; dx++)
492 {
493 int x = center.X + dx;
494 int y = center.Y + dy;
495
496 int px = dx + halfSize;
497 int py = dy + halfSize;
498
499 char c = 'X';
500 if (dx != 0 && dy != 0)
501 {
502 c = 'X';
503 }
504 else if (levelMap.Tiles.InBounds(x, y))
505 {
506 if (levelMap.Tiles[x, y] == CheckpointTile ||
507 (x == levelMap.StartPoint.X && y == levelMap.StartPoint.Y) ||
508 (x == levelMap.FinishPoint.X && y == levelMap.FinishPoint.Y))
509 {
510 if ((Mathf.Abs(dx) < 2 || Mathf.Abs(dy) < 2) && halfSize > 1)
511 c = 'C';
512 else
513 c = '1';
514 }
515 else if (levelMap.Tiles[x, y] == TrackTile || allowOverlap && levelMap.Tiles[x, y] == UsedTileValue)
516 {
517 c = '1';
518 }
519 }
520 pattern[py, px] = c;
521 }
522 }
523
524 for (int y = 0; y < size; y++)
525 {
526 for (int x = 0; x < size; x++)
527 {
528 if (pattern[y, x] == '1' && IsAlone(new Coordinates(y, x), pattern))
529 {
530 pattern[y, x] = 'X';
531 }
532 }
533 }
534
535 patternBuilder.Clear();
536 for (int y = 0; y < size; y++)
537 {
538 for (int x = 0; x < size; x++)
539 {
540 patternBuilder.Append(pattern[y, x]);
541 }
542 }
543
544 return patternBuilder.ToString();
545 }

563 /// <summary>

565 /// </summary>

567 /// <param name="pattern">Pattern grid.</param>

572 {
573 var neighbor = position + offset;

607 /// <summary>
608 /// Marks cells occupied by a larger piece so they are not placed as independent visible pieces.
609 /// </summary>
610 /// <param name="position">Center coordinate of the placed piece.</param>
611 /// <param name="size">Pattern size of the placed piece.</param>
612 private void MarkSpaceAsUsed(Coordinates position, int size)
613 {
614 int usedArea = (size / 2) - 1;
615 Coordinates coords;
616
617 for (int dx = -usedArea; dx <= usedArea; dx++)
618 {
619 for (int dy = -usedArea; dy <= usedArea; dy++)
620 {
621 if (dx != 0 && dy != 0)
622 continue;
623
624 coords = new Coordinates(position.X + dx, position.Y + dy);
625
626 if (!levelMap.Tiles.InBounds(coords.X, coords.Y))
627 continue;
628
629 if (coords == levelMap.StartPoint || coords == levelMap.FinishPoint || levelMap.Tiles.At(coords) == CheckpointTile)
630 continue;
631
632 if (levelMap.Tiles.At(coords) == TrackTile)
633 levelMap.Tiles.At(coords) = UsedTileValue;
634 }
635 }
636 }
