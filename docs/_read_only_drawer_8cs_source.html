5/// <summary>
6/// Inspector drawer that renders fields marked with <see cref="ReadOnlyAttribute"/> as disabled.
7/// </summary>
8/// <remarks>
9/// @ingroup editor_util

11///
12/// Usage:
13/// <code>
14/// [SerializeField, ReadOnly] private int debugValue;
15/// </code>
16///
17/// The drawer preserves Unity's native property rendering, including child fields for complex types,
18/// arrays, and object references. It only disables user input while drawing the property.
19///
20/// Threading:
21/// - Unity Editor thread only.
22/// - Editor-only script, excluded from player builds by the <c>UNITY_EDITOR</c> guard.
23/// </remarks>
24[CustomPropertyDrawer(typeof(ReadOnlyAttribute))]
25public class ReadOnlyDrawer : PropertyDrawer
26{

29 /// </summary>
