123 /// <summary>

125 /// </summary>

127 /// <remarks>
128 /// The method updates:
129 /// - The optional level preview, if a <see cref="LevelPreviewer"/> component is attached.

132 /// - The day/night icon.
133 /// </remarks>
134 public void SetUp(GameDataManager.LevelData levelData)
135 {
136 this.levelData = levelData;
137
138 // Attempt to show a visual preview, if a LevelPreviewer component is present.
139 _ = GetComponent<LevelPreviewer>()?.ShowPreviewAsync(levelData.LevelMap);
140
141 // Update displayed time.
142 if (levelData.Time < float.MaxValue)
143 {
144 TimeSpan timeSpan = TimeSpan.FromSeconds(levelData.Time);
145 levelTime.text = string.Format("{0:D2}:{1:D2}:{2:D2}", timeSpan.Minutes, timeSpan.Seconds, timeSpan.Milliseconds);
146 }
147 else
148 {
149 levelTime.text = "No Time";
150 }
151
152 levelName.text = levelData.LevelMap.Name;
153
154 if (levelData.LevelMap.IsDayTrack)
155 dayNightIconImage.texture = dayIcon;
156 else
157 dayNightIconImage.texture = nightIcon;
158 }
