5/// <summary>
6/// Inspector drawer for <see cref="ShowIfAttribute"/> that conditionally shows or hides
7/// a serialized field based on the value of a boolean sibling/parent property.
8/// </summary>
9/// <remarks>
10/// @ingroup editor_util
11/// @thread Unity Editor thread only. Editor-only script, excluded from player builds via <c>UNITY_EDITOR</c>.
12///
13/// Behavior:
14/// - If the referenced condition property is found and is a <c>bool</c>, the decorated field is only drawn
15/// when its value equals <see cref="ShowIfAttribute.RequiredState"/> (default: <c>true</c>).
16/// - If the condition cannot be found or is not a boolean, the field is shown (fails open).
17///
18/// Field path resolution:
19/// - First attempts a sibling/ancestor-relative lookup by replacing the tail of <c>property.propertyPath</c>
20/// with the condition field name.
21/// - Falls back to a root-level lookup using just <see cref="ShowIfAttribute.Field"/>.
22///
23/// Usage:
24/// <code>
25/// public class Example : MonoBehaviour
26/// {
27/// [SerializeField] private bool advancedMode;
28/// [SerializeField, ShowIf(nameof(advancedMode))]
29/// private float advancedSetting;
30/// }
31/// </code>
32/// </remarks>
33[CustomPropertyDrawer(typeof(ShowIfAttribute))]
34public class ShowIfDrawer : PropertyDrawer
35{
36 /// <summary>
37 /// Returns either the normal property height (when visible) or 0 when hidden,
38 /// allowing the drawer to collapse layout space in the Inspector.
39 /// </summary>
40 /// <param name="property">Serialized property being measured.</param>
41 /// <param name="label">GUI label.</param>
42 /// <returns>Height in pixels for the property field.</returns>
43 public override float GetPropertyHeight(SerializedProperty property, GUIContent label)
44 {
45 return IsVisible(property) ? EditorGUI.GetPropertyHeight(property, label, true) : 0f;
46 }
47
48 /// <summary>
49 /// Draws the property only if the visibility condition evaluates to true.
50 /// </summary>
51 /// <param name="position">Drawing rect.</param>
52 /// <param name="property">Serialized property to draw.</param>
53 /// <param name="label">GUI label.</param>
54 public override void OnGUI(Rect position, SerializedProperty property, GUIContent label)
55 {
56 if (IsVisible(property))
57 {
58 EditorGUI.PropertyField(position, property, label, true);
59 }
60 }
61
62 /// <summary>
63 /// Evaluates the condition specified by the attached <see cref="ShowIfAttribute"/>.
64 /// </summary>
65 /// <param name="property">Serialized property whose visibility is being evaluated.</param>
66 /// <returns>True if the field should be shown; otherwise false.</returns>
67 private bool IsVisible(SerializedProperty property)
68 {
69 var attrib = (ShowIfAttribute)attribute;
70
71 // Try relative path first: replace tail of the current property path with the condition field name.
72 string path = property.propertyPath;
73 int lastDot = path.LastIndexOf('.');
74 string pathToField = (lastDot >= 0 ? path.Substring(0, lastDot + 1) : "") + attrib.Field;
75
76 // Attempt relative lookup, then fallback to root-level lookup.
77 SerializedProperty cond = property.serializedObject.FindProperty(pathToField)
78 ?? property.serializedObject.FindProperty(attrib.Field);
79
80 // Fail open (visible) if condition missing or not a bool.
81 if (cond == null || cond.propertyType != SerializedPropertyType.Boolean)
82 {
83 return true;
84 }
85
86 // Visible only when the boolean matches the required state.
87 return cond.boolValue == attrib.RequiredState;
88 }
89}
