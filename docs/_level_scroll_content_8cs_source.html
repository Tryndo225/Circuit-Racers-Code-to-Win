5/// <summary>
6/// Dynamic grid layout and population of level preview items inside a scrollable content RectTransform.
7/// Subscribes to <see cref="GameDataManager"/> changes, instantiates preview prefabs, and lays them out responsively.
8/// </summary>
9/// <remarks>
10/// @ingroup ui
11/// @brief Read-only UI that mirrors the current set of saved levels.
12///
13/// Responsibilities:
14/// - Listen for game-data changes and (re)build the preview list.
15/// - Instantiate a preview prefab per level (under <see cref="contentRect"/>).
16/// - Compute a responsive, column-based grid using viewport-relative spacing/padding ratios.
17/// - Adjust content height so a parent <c>ScrollRect</c> can scroll the full grid.
18///
19/// Threading:
20/// - Unity main thread only.
21///
22/// Usage:
23/// - Place this on a GameObject under a <c>ScrollRect</c>. Assign <see cref="contentRect"/> to the content transform,
24/// and set <see cref="levelPreviewerPrefab"/> to a prefab that contains a <see cref="LevelEntry"/> component.
25/// - The grid reflows on dimension changes and one frame after (to catch late layout updates).

135 /// <summary>
136 /// Destroys old preview items, instantiates one per saved level, wires the data, and updates the layout.
137 /// </summary>
138 private void CreateLevelPreviews()
139 {
140 foreach (var go in levelPreviewers) Destroy(go);
141 levelPreviewers.Clear();
142
143 var levels = GameDataManager.Instance.CurrentGameData.Levels;
144 foreach (var levelMap in levels)
145 {
146 var previewerObj = Instantiate(levelPreviewerPrefab, contentRect);
147 var previewer = previewerObj.GetComponent<LevelEntry>();
148 if (previewer != null) previewer.SetUp(levelMap);
149 levelPreviewers.Add(previewerObj);
150 }
151
152 UpdateLayout();
153 StartCoroutine(DelayedLayout());
154 }
155
156 /// <summary>

158 /// </summary>
159 private IEnumerator DelayedLayout()
160 {

162 UpdateLayout();
163 }

165 /// <summary>
166 /// Computes a responsive, column-based grid for all preview items and adjusts the content height to fit.
167 /// Item size is square and derived from viewport width, column count, and spacing/padding ratios.
168 /// </summary>
169 private void UpdateLayout()
170 {
171 if (contentRect == null || levelPreviewers.Count == 0) return;
172
173 var viewport = contentRect.parent as RectTransform;
174 if (viewport == null) return;
175
176 float fieldWidth = viewport.rect.width;
177 if (fieldWidth <= 0f) return;
178
179 contentRect.anchorMin = new Vector2(0f, 1f);
180 contentRect.anchorMax = new Vector2(1f, 1f);
181 contentRect.pivot = new Vector2(0.5f, 1f);
182 contentRect.offsetMin = new Vector2(0f, contentRect.offsetMin.y);
183 contentRect.offsetMax = new Vector2(0f, contentRect.offsetMax.y);
184
185 int cols = Mathf.Max(1, itemsInLine);
186
187 float hPad = horizontalPaddingRatio * fieldWidth;
188 float hGap = horizontalSpacingRatio * fieldWidth;
189 float topPad = topPaddingRatio * fieldWidth;
190 float botPad = bottomPaddingRatio * fieldWidth;
191
192 float totalGutters = (cols - 1) * hGap;
193 float usableWidth = Mathf.Max(0f, fieldWidth - 2f * hPad - totalGutters);
194 float itemWidth = usableWidth / cols;
195 float itemHeight = itemWidth; // square tiles
196
197 float vGap = verticalSpacingRatio * fieldWidth;
198
199 for (int i = 0; i < levelPreviewers.Count; i++)
200 {
201 var rt = levelPreviewers[i].GetComponent<RectTransform>();
202 if (!rt) continue;
203
204 rt.anchorMin = new Vector2(0f, 1f);
205 rt.anchorMax = new Vector2(0f, 1f);
206 rt.pivot = new Vector2(0f, 1f);
207 rt.sizeDelta = new Vector2(itemWidth, itemHeight);
208
209 int row = i / cols;
210 int col = i % cols;
211
212 float x = hPad + col * (itemWidth + hGap);
213 float y = -topPad - row * (itemHeight + vGap);
214
215 rt.anchoredPosition = new Vector2(x, y);
216 }
217
218 int totalRows = Mathf.CeilToInt((float)levelPreviewers.Count / cols);
219 float totalHeight =
220 topPad +
221 (totalRows > 0 ? (totalRows * itemHeight + (totalRows - 1) * vGap) : 0f) +
222 botPad;
223
224 contentRect.sizeDelta = new Vector2(0f, totalHeight);
225 }
