6#region Generic Button Handler
7/// <summary>
8/// Generic UI button driver that delegates its click behavior to a pluggable <see cref="ButtonType"/>.
9/// </summary>
10/// <remarks>
11/// @ingroup ui
12/// @brief Keeps scene/UI logic decoupled by holding a serialized strategy object that implements <see cref="ButtonType.Action"/>.
13///
14/// Usage:
15/// - Assign a <see cref="ButtonType"/> (e.g., <see cref="ChangeSceneButton"/>, <see cref="GenerateLevelButton"/>) in the Inspector.
16/// - Hook the Unity Button's OnClick() to <see cref="OnButtonClick"/> (or call it from a UnityEvent).
17///
18/// Threading:
19/// - Unity main thread for <see cref="OnButtonClick"/>; specific strategies may spawn background work (see <see cref="GenerateLevelButton"/>).
20/// </remarks>
21public class ButtonScript : MonoBehaviour
22{
23 /// <summary>
24 /// Serialized behavior object that performs the action when the button is clicked.
25 /// </summary>
26 [SerializeReference] private ButtonType properties;
27
28 /// <summary>
29 /// Read/write accessor for the underlying button behavior.
30 /// </summary>
31 public ButtonType Properties
32 {
33 get => properties;
34 set => properties = value;
35 }
36
37 /// <summary>
38 /// Invoked by the UI Button. Validates configuration and executes <see cref="ButtonType.Action"/>.
39 /// </summary>
40 public void OnButtonClick()
41 {
42 if (properties == null)
43 {
44 Debug.LogWarning("Button properties are not set.");
45 return;
46 }
47 properties.Action();
48 }
49}

52#region Button Types
53/// <summary>
54/// Abstract strategy for a button action. Concrete implementations encapsulate the behavior
55/// (scene change, level selection, generation, etc.) without coupling UI to systems.
56/// </summary>
57/// <remarks>
58/// @ingroup ui
59/// Implementations must remain safe to call from the main thread. Long-running work should be
60/// pushed to background tasks and marshalled back to main (see <see cref="GenerateLevelButton"/>).
61/// </remarks>
62[Serializable]
63public abstract class ButtonType
64{
65 /// <summary>
66 /// Execute the action represented by this strategy.
67 /// </summary>
68 public abstract void Action();
69}

170/// <summary>
171/// Strategy: procedurally generate a level (optionally circuit), add checkpoints,
172/// and display it in a popup preview.
173/// </summary>
174/// <remarks>
175/// @ingroup ui @ingroup level_gen
176/// Workflow:
177/// - Reads UI sliders/toggles to update generation parameters.
178/// - Runs generation and checkpoint placement on background threads via <see cref="Task.Run"/>.
179/// - On success, shows the result in <see cref="LevelPopUp"/> attached to <see cref="popUp"/>.
180///
181/// Threading:
182/// - Kicks off CPU work on background threads; UI interactions remain on main.
183///
184/// Invariants:
185/// - <see cref="popUp"/> must reference a GameObject with a <see cref="LevelPopUp"/> component.
186/// </remarks>
187[Serializable]
188public class GenerateLevelButton : ButtonType
189{
190 #region References (UI)
191 /// <summary>Popup container that owns a <see cref="LevelPopUp"/> for previewing the generated map.</summary>
192 [Header("References")]
193 [SerializeField] private GameObject popUp;
194 /// <summary>Slider for width (X) in tiles.</summary>
195 [SerializeField] private Slider sizeXSlider;
196 /// <summary>Slider for height (Y) in tiles.</summary>
197 [SerializeField] private Slider sizeYSlider;
198 /// <summary>Slider for step count (carving iterations).</summary>
199 [SerializeField] private Slider stepCountSlider;
200 /// <summary>Toggle to generate circuits (loop) instead of point-to-point tracks.</summary>
201 [SerializeField] private Toggle createCircuitsToggle;
202 #endregion
203
204 #region Generation Settings (defaults)
205 /// <summary>Whether to attempt creating closed-loop tracks.</summary>
206 [Header("Generation Settings")]
207 [SerializeField] public bool createCircuits = true;
208 /// <summary>Grid width in tiles.</summary>
209 [SerializeField] public int sizeX = 50;
210 /// <summary>Grid height in tiles.</summary>
211 [SerializeField] public int sizeY = 50;
212 /// <summary>Total carving steps.</summary>
213 [SerializeField] private int stepCount = 20;
214 /// <summary>Computed step size (derived from area and count).</summary>
215 [SerializeField, ReadOnly] private int stepSize;
216 /// <summary>Max attempts per step to find a valid target.</summary>
217 [SerializeField] private int maxAttemps = 1000;
218 #endregion
219
220 /// <summary>Parameterless ctor for serialization.</summary>
221 public GenerateLevelButton()
222 { }
223
224 /// <summary>
225 /// Pulls latest values from the bound UI widgets into the generation fields.
226 /// </summary>
227 private void UpdateValues()
228 {
229 if (sizeXSlider != null)
230 sizeX = Mathf.RoundToInt(sizeXSlider.value);
231 if (sizeYSlider != null)
232 sizeY = Mathf.RoundToInt(sizeYSlider.value);
233 if (stepCountSlider != null)
234 stepCount = Mathf.RoundToInt(stepCountSlider.value);
235 if (createCircuitsToggle != null)
236 createCircuits = createCircuitsToggle.isOn;
237 }
238
239 /// <summary>
240 /// Execute: generate a level, add checkpoints, and show the preview popup.
241 /// </summary>
242 /// <remarks>
243 /// Uses <see cref="Task.Run(Func{Task})"/> to keep the main thread responsive.
244 /// Logs/guards against missing references and exceptions.
245 /// </remarks>
246 public override async void Action() // ok for top-level UI handlers
247 {
248 try
249 {
250 if (popUp == null)
251 {
252 Debug.LogError("Pop-up is not assigned.");
253 return;
254 }
255
256 UpdateValues();
257
258 // Heuristic: scale step size from area and desired step count.
259 stepSize = Mathf.RoundToInt(Mathf.Sqrt(sizeX * sizeY)) * 4 / stepCount;
260
261 Debug.Log("Starting level generation...");
262 var map = await Task.Run(() =>
263 {
264 return LevelGenerator.GenerateLevel(sizeX, sizeY, createCircuits, stepCount, stepSize, maxAttemps, SeedFactory.Next());
265 });
266
267 Debug.Log("Level generated, generating checkpoints...");
268
269 await Task.Run(() =>
270 {
271 LevelCheckPointMaker.GenerateCheckPoints(map);
272 });
273
274 Debug.Log("Checkpoints generated, showing map...");
275
276 if (map != null)
277 {
278 popUp.GetComponent<LevelPopUp>().ShowMap(map);
279 }
280 }
281 catch (Exception ex)
282 {
283 Debug.LogError($"Level generation failed: {ex}");
284 }
285 }
286}
