9/// <summary>
10/// Generic UI button driver that delegates its click behavior to a pluggable <see cref="ButtonType"/> strategy.
11/// </summary>
12/// <remarks>
13/// @ingroup ui
14/// @brief Connects reusable Unity UI buttons to serialized action objects.
15///
16/// This component is intended to be attached to a Unity UI button object. Instead of hard-coding
17/// behavior directly into the button component, the actual action is stored in <see cref="properties"/>
18/// as a serialized <see cref="ButtonType"/> implementation.
19///
20/// Typical usage:
21/// - Attach this component to a UI button object.
22/// - Assign a concrete <see cref="ButtonType"/> strategy in the Inspector.
23/// - Hook the Unity Button OnClick() event to <see cref="OnButtonClick"/>.
24///
25/// This keeps menu buttons, level buttons, popup buttons, import buttons, and scene-transition buttons
26/// visually configurable while sharing the same click-handling entry point.
27/// </remarks>
28public class ButtonScript : MonoBehaviour
29{
30 /// <summary>
31 /// Serialized strategy object that performs the actual action when the button is clicked.
32 /// </summary>
33 [Tooltip("Serialized button action executed when this UI button is clicked.")]
34 [SerializeReference] private ButtonType properties;
35
36 /// <summary>
37 /// Gets or assigns the serialized button behavior strategy.
38 /// </summary>
39 /// <remarks>
40 /// This property allows code-generated UI elements to configure the button action at runtime,
41 /// while still supporting Inspector-based assignment through <see cref="properties"/>.
42 /// </remarks>
43 public ButtonType Properties
44 {
45 get => properties;
46 set => properties = value;
47 }
48
49 /// <summary>
50 /// Invoked by the Unity UI Button OnClick event.
51 /// </summary>
52 /// <remarks>
53 /// The method validates that a concrete <see cref="ButtonType"/> was assigned before executing it.
54 /// Missing configuration is reported as a warning rather than an exception so that incorrectly
55 /// configured menu buttons do not crash the scene.
56 /// </remarks>
57 public void OnButtonClick()
58 {
59 if (properties == null)
60 {
61 Debug.LogWarning("Button properties are not set.");
62 return;
63 }
64 properties.Action();
65 }
66}

72/// <summary>
73/// Abstract base class for serialized UI button actions.
74/// </summary>
75/// <remarks>
76/// @ingroup ui
77/// @brief Defines the strategy interface used by <see cref="ButtonScript"/>.
78///
79/// Concrete subclasses encapsulate individual button behaviors, such as changing scenes,
80/// selecting or removing levels, generating tracks, opening popups, importing levels,
81/// showing notifications, or starting replay/gameplay scenes.
82///
83/// Because these objects are triggered from Unity UI events, implementations should remain safe
84/// to call from the Unity main thread. Long-running work should be moved to background tasks where
85/// appropriate, as done by <see cref="GenerateLevelButton"/>.
86/// </remarks>
87[Serializable]
88public abstract class ButtonType
89{
90 /// <summary>
91 /// Executes the action represented by this button strategy.
92 /// </summary>
93 public abstract void Action();
94}

218/// <summary>
219/// Button strategy that procedurally generates a level, creates checkpoints, and displays the result in a popup preview.
220/// </summary>
221/// <remarks>
222/// @ingroup ui
223/// @ingroup level_gen
224///
225/// Workflow:
226/// - Reads current size/circuit settings from the assigned UI controls.
227/// - Generates a <see cref="LevelMap"/> using <see cref="LevelGenerator"/>.
228/// - Adds checkpoint data using <see cref="LevelCheckPointMaker"/>.
229/// - Displays the generated level through a <see cref="LevelPopUp"/> component on <see cref="popUp"/>.
230///
231/// Threading:
232/// - Generation and checkpoint placement are run through <see cref="Task.Run"/> to avoid blocking the UI.
233/// - Unity object access remains in the surrounding async UI action.
234///
235/// Requirements:
236/// - <see cref="popUp"/> must reference a GameObject with a <see cref="LevelPopUp"/> component.
237/// - <see cref="sizeSlider"/> and <see cref="createCircuitsToggle"/> are optional; if left unassigned,
238/// the serialized default values are used.
239/// </remarks>

241public class GenerateLevelButton : ButtonType
242{

244

246 /// Popup container that owns the <see cref="LevelPopUp"/> used to preview the generated map.
247 /// </summary>
248 [Header("References")]
249 [Tooltip("Popup GameObject containing the LevelPopUp component used to preview the generated level.")]
250 [SerializeField] private GameObject popUp;
251
252 /// <summary>
253 /// UI slider that controls the generated level size in tiles.
254 /// </summary>
255 [Tooltip("Slider used to choose the generated level size in tiles.")]
256 [SerializeField] private Slider sizeSlider;
257
258 /// <summary>
259 /// UI toggle that controls whether generated tracks should be circuits or point-to-point tracks.
260 /// </summary>

269 /// Determines whether the generator should attempt to create a closed-loop circuit.

271 [Header("Generation Settings")]
272 [Tooltip("Whether the generator should create a closed circuit instead of a point-to-point track.")]
273 [SerializeField] public bool createCircuits = true;
274

277 /// </summary>
278 [Tooltip("Width and height of the generated square level grid, measured in tiles.")]
279 [SerializeField] public int size = 50;
280
281 /// <summary>
282 /// Number of generation steps used by the level generator.
283 /// </summary>
284 [Tooltip("Number of carving/generation steps derived from the selected level size.")]
285 [SerializeField, ReadOnly] private int stepCount = 20;
286

288 /// Size of each generation step, derived from <see cref="size"/>.
289 /// </summary>
290 [Tooltip("Step size used by the generator, derived from the selected level size.")]
291 [SerializeField, ReadOnly] private int stepSize;
292
293 /// <summary>
294 /// Maximum number of attempts used by the generator when searching for a valid step target.
295 /// </summary>
296 [Tooltip("Maximum number of attempts allowed when the generator searches for a valid target.")]
297 [SerializeField] private int maxAttempts = 1000;

299 #endregion Generation Settings (defaults)

301 /// <summary>
302 /// Creates an empty level-generation strategy for Unity serialization.
303 /// </summary>
304 public GenerateLevelButton()
305 { }
306
307 /// <summary>
308 /// Copies current values from the assigned UI widgets into the serialized generation fields.
309 /// </summary>
310 /// <remarks>
311 /// Missing UI references are ignored so the strategy can still run with the serialized default values.
312 /// </remarks>
313 private void UpdateValues()
314 {
315 if (sizeSlider != null)
316 size = Mathf.RoundToInt(sizeSlider.value);
317 if (createCircuitsToggle != null)
318 createCircuits = createCircuitsToggle.isOn;
319 }
320
321 /// <summary>
322 /// Generates a level, generates its checkpoints, and opens the preview popup.
323 /// </summary>
324 /// <remarks>
325 /// This is an async UI entry point. It catches and logs generation exceptions so errors do not
326 /// propagate through Unity's UI event system.
327 /// </remarks>
328 public override async void Action() // ok for top-level UI handlers
329 {
330 try
331 {
332 if (popUp == null)
333 {
334 Debug.LogError("Pop-up is not assigned.");
335 return;
336 }
337
338 UpdateValues();
339
340 stepCount = size;
341 stepSize = Mathf.Max(1, size / 5);
342
343 Debug.Log("Starting level generation...");
344 var map = await Task.Run(() =>
345 {
346 return LevelGenerator.GenerateLevel(size, size, createCircuits, stepCount, stepSize, maxAttempts, SeedFactory.Next());
347 });
348
349 Debug.Log("Level generated, generating checkpoints...");
350
351 await Task.Run(() =>
352 {
353 LevelCheckPointMaker.GenerateCheckPoints(map);
354 });
355
356 Debug.Log("Checkpoints generated, showing map...");
357
358 if (map != null)
359 {
360 popUp.GetComponent<LevelPopUp>().ShowMap(map);
361 }
362 }
363 catch (Exception ex)
364 {
365 Debug.LogError($"Level generation failed: {ex}");
366 }
367 }
368}

649/// <summary>
650/// Button strategy that starts gameplay by selecting either the day or night scene for the current level.
651/// </summary>
652/// <remarks>
653/// @ingroup ui
654/// @ingroup game_data
655/// @ingroup scene_mgmt
656///
657/// The class name intentionally keeps the existing spelling <c>GoPlayButton</c> to preserve Unity
658/// serialized references.
659///
660/// The action checks <see cref="GameDataManager.CurrentLevelMap"/> and chooses a scene based on
661/// <see cref="LevelMap.IsDayTrack"/>:
662/// - Day tracks load <see cref="_sceneToLoadDay"/>.
663/// - Night tracks load <see cref="_sceneToLoadNight"/>.
664///
665/// If no level is selected, or if the required scene helper is missing, the method reports the problem
666/// and does not attempt a scene transition.
667/// </remarks>
668[Serializable]
669public class GoPlayButton : ButtonType
670{
671 /// <summary>
672 /// Scene helper used when the selected level is marked as a day track.
673 /// </summary>
674 [Tooltip("Scene loaded when the selected level is marked as a day track.")]
675 [SerializeField] private SceneAssetHelper _sceneToLoadDay;
676
677 /// <summary>
678 /// Scene helper used when the selected level is marked as a night track.
679 /// </summary>
680 [Tooltip("Scene loaded when the selected level is marked as a night track.")]
681 [SerializeField] private SceneAssetHelper _sceneToLoadNight;
682
683 /// <inheritdoc/>
684 public override void Action()
685 {
686 LevelMap levelMap = GameDataManager.Instance.CurrentLevelMap;
687 if (levelMap == null)
688 {
689 NotificationManager.Instance.Show("No level selected.");
690 }
691 else if (levelMap.IsDayTrack)
692 {
693 if (_sceneToLoadDay == null)
694 {
695 Debug.LogError("Day scene to load is not assigned.");
696 return;
697 }
698 SceneManagement.Instance.ChangeScene(_sceneToLoadDay);
699 }
700 else if (levelMap.IsDayTrack == false)
701 {
702 if (_sceneToLoadNight == null)
703 {
704 Debug.LogError("Night scene to load is not assigned.");
705 return;
706 }
707 SceneManagement.Instance.ChangeScene(_sceneToLoadNight);
708 }
709 }
