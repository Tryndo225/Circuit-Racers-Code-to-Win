88 /// <summary>
89 /// Physics callback: computes collision severity from impulse, shapes volume and pitch,
90 /// and plays a crash clip if above threshold and not on cooldown.
91 /// </summary>
92 /// <param name="c">Collision data provided by Unity.</param>
93 /// <remarks>
94 /// Skips if: cooldown active, other object is on an ignored layer,
95 /// no clips configured, or computed severity <= 0.
96 /// </remarks>
97 private void OnCollisionEnter(Collision c)
98 {
99 if (Time.time < nextAllowedTime) return;
100
101 if (ignoreLayers.Contains(c.gameObject.layer)) return;
102
103 if (crashClips == null || crashClips.Length == 0) return;
104
105 float severity = GetSeverity01(c);
106 if (severity <= 0f) return;
107
108 var clip = crashClips[Random.Range(0, crashClips.Length)];
109
110 // Volume shaping: curve(severity) * base, clamped by minimumVolume.
111 float volume = volumeCurve.Evaluate(severity) * baseVolume;
112 volume = Mathf.Max(volume, minimumVolume);
113
114 // Pitch shaping: lerp by severity with slight randomization.
115 float pitch = Mathf.Lerp(minPitch, maxPitch, severity) * Random.Range(0.98f, 1.02f);
116
117 SoundManager.Instance.PlaySFXClip(clip, transform, volume, pitch);
118
119 nextAllowedTime = Time.time + cooldown;
120 }
121
122 #endregion

124 #region Private Helpers

126 /// <summary>
127 /// Maps the collision impulse magnitude (N*s) into a normalized severity in [0,1].
128 /// </summary>
129 /// <param name="c">Collision data.</param>
130 /// <returns>
131 /// Normalized severity where 0 <= result <= 1; returns 0 if impulse is negligible.
132 /// </returns>
133 /// <complexity>O(1)</complexity>
134 private float GetSeverity01(Collision c)
135 {
136 float j = c.impulse.magnitude;
137 Debug.Log("Collision impulse: " + j + " N*s");
138 if (j > 0.01f)
139 {
140 return Mathf.InverseLerp(minImpulse, maxImpulse, j);
141 }
142 return 0f;
143 }
