1using UnityEngine;
2/// <summary>
3/// Third-person follow camera for a racing vehicle.
4/// </summary>
5/// <remarks>
6/// @ingroup car_ctrl
7/// @brief Tracks a target using configurable offset, smoothing, fixed pitch, and movement-direction-based positioning.
8///
9/// The camera follows the target from behind its current movement direction. When a target
10/// <see cref="Rigidbody"/> is assigned and moving fast enough, the camera uses the rigidbody's planar
11/// velocity as the heading. This allows the camera to flip naturally when the vehicle reverses.
12///
13/// Behaviour:
14/// - Smoothly follows the desired position with <see cref="Vector3.SmoothDamp"/>.
15/// - Smoothly aligns yaw with <see cref="Mathf.SmoothDampAngle"/>.
16/// - Keeps a fixed pitch angle.
17/// - Adds look-ahead in the current movement direction.
18/// - Can be snapped instantly with <see cref="SyncCamera"/> after respawns or teleports.
19///
20/// Threading:
21/// - Unity main thread only.
22/// - Camera motion is updated in <see cref="LateUpdate"/>.
23/// </remarks>
24public class FollowCamera : MonoBehaviour
25{
26 #region Inspector
27
28 [Header("Target")]
29 /// <summary>
30 /// Transform followed by the camera.
31 /// </summary>
32 /// <remarks>
33 /// If this is null, camera updates are skipped.
34 /// </remarks>
35 [Tooltip("Transform followed by the camera. If empty, camera updates are skipped.")]
36 [SerializeField] public Transform target;
37
38 [Header("Physics")]
39 /// <summary>
40 /// Optional rigidbody used to determine actual movement direction.
41 /// </summary>
42 /// <remarks>
43 /// If assigned, the camera can use vehicle velocity instead of only target forward direction.
44 /// </remarks>
45 [Tooltip("Optional Rigidbody used to determine actual movement direction, including reversing.")]
46 public Rigidbody targetRb;
47
48 [Header("Offset")]
49 /// <summary>
50 /// Camera offset from the target.
51 /// </summary>
52 /// <remarks>
53 /// X is side offset, Y is height, and Z controls follow distance. The sign of Z is ignored
54 /// because distance is calculated from its absolute value.
55 /// </remarks>
56 [Tooltip("Camera offset. X = side offset, Y = height, Z = follow distance.")]
57 public Vector3 offset = new Vector3(0f, 6f, -12f);
58
59 [Header("Position Smoothing")]
60 /// <summary>
61 /// Position smoothing time used by <see cref="Vector3.SmoothDamp"/>.
62 /// </summary>
63 /// <remarks>
64 /// Smaller values make the camera more responsive. Larger values make it smoother.
65 /// </remarks>
66 [Tooltip("Position smoothing time. Smaller values are more responsive; larger values are smoother.")]
67 public float positionSmoothTime = 0.25f;
68
69 [Header("Rotation Smoothing")]
70 /// <summary>
71 /// Yaw smoothing time used by <see cref="Mathf.SmoothDampAngle"/>.
72 /// </summary>
73 /// <remarks>
74 /// Smaller values make yaw react faster. Larger values make it turn more gently.
75 /// </remarks>
76 [Tooltip("Yaw smoothing time. Smaller values react faster; larger values turn more gently.")]
77 public float rotationSmoothTime = 0.2f;
78
79 [Header("Pitch")]
80 /// <summary>
81 /// Fixed pitch angle applied to the camera rotation.
82 /// </summary>
83 [Tooltip("Fixed pitch angle in degrees applied to the camera rotation.")]
84 public float fixedPitchAngle = 15f;
85
86 [Header("Look Ahead")]
87 /// <summary>
88 /// Distance added along the selected movement heading.
89 /// </summary>
90 /// <remarks>
91 /// This makes the camera look and move slightly ahead of the target.
92 /// </remarks>
93 [Tooltip("Distance added along the movement heading to make the camera anticipate motion.")]
94 public float lookAheadDistance = 4f;
95
96 [Header("Movement Direction")]
97 /// <summary>
98 /// Minimum planar speed required before rigidbody velocity is used as the camera heading.
99 /// </summary>
100 /// <remarks>
101 /// Below this threshold, the camera falls back to the target's forward direction.
102 /// </remarks>
103 [Tooltip("Minimum planar speed before Rigidbody velocity is used as the camera heading.")]
104 public float minVelocityForDirection = 0.5f;

111 /// Velocity accumulator used internally by <see cref="Vector3.SmoothDamp"/>.

122 #region Unity Methods
123
124 /// <summary>
125 /// Updates the camera after target movement for the current frame.
126 /// </summary>
127 /// <remarks>
128 /// The camera moves toward a movement-direction-based follow position and smoothly rotates toward
129 /// the same heading while keeping <see cref="fixedPitchAngle"/>.
130 /// </remarks>
131 private void LateUpdate()
132 {
133 if (target == null)
134 {
135 return;
136 }
137
138 Vector3 flatDirection = GetCameraDirection();
139
140 Vector3 desiredPosition = GetDesiredCameraPosition();
141 transform.position = Vector3.SmoothDamp(transform.position, desiredPosition, ref positionVelocity, positionSmoothTime);
142
143 float targetYaw = Mathf.Atan2(flatDirection.x, flatDirection.z) * Mathf.Rad2Deg;
144 float currentYaw = transform.eulerAngles.y;
145
146 float smoothYaw = Mathf.SmoothDampAngle(currentYaw, targetYaw, ref yawVelocity, rotationSmoothTime);
147
148 transform.rotation = Quaternion.Euler(fixedPitchAngle, smoothYaw, 0f);
149 }
150
151 #endregion
152
153 #region Public API
154
155 /// <summary>
156 /// Instantly snaps the camera to the target using the configured movement-based offset.
157 /// </summary>
158 /// <remarks>
159 /// Useful after teleports, respawns, or scene loads. Also clears smoothing velocity accumulators
160 /// so the next frame does not continue old smoothing momentum.
161 /// </remarks>
162 public void SyncCamera()
163 {
164 if (target == null)
165 {
166 return;
167 }
168
169 transform.position = GetDesiredCameraPosition();
170
171 Vector3 flatDirection = GetCameraDirection();
172 float targetYaw = Mathf.Atan2(flatDirection.x, flatDirection.z) * Mathf.Rad2Deg;
173 transform.rotation = Quaternion.Euler(fixedPitchAngle, targetYaw, 0f);
174
175 positionVelocity = Vector3.zero;
176 yawVelocity = 0f;
177 }
178
179 /// <summary>
180 /// Assigns a new target transform and tries to cache its rigidbody.
181 /// </summary>
182 /// <param name="newTarget">New transform to follow.</param>
183 /// <remarks>
184 /// After assigning the target, the camera is immediately synchronized to avoid a delayed smooth movement
185 /// from the previous target position.
186 /// </remarks>
187 public void SetTarget(Transform newTarget)
188 {
189 target = newTarget;
190 newTarget.TryGetComponent<Rigidbody>(out targetRb);
191 SyncCamera();
192 }
193
194 #endregion

196 #region Helpers
197
198 /// <summary>
199 /// Gets the flattened heading used by the camera.
200 /// </summary>
201 /// <returns>Normalized planar direction used for camera positioning and yaw.</returns>
202 /// <remarks>
203 /// Rigidbody planar velocity is preferred when the target is moving faster than
204 /// <see cref="minVelocityForDirection"/>. Otherwise, the method falls back to the target's forward direction.
205 /// </remarks>
206 private Vector3 GetCameraDirection()
207 {
208 if (targetRb != null)
209 {
210 Vector3 flatVelocity = targetRb.linearVelocity;
211 flatVelocity.y = 0f;
212
213 if (flatVelocity.sqrMagnitude > minVelocityForDirection * minVelocityForDirection)
214 {
215 return flatVelocity.normalized;
216 }
217 }
218
219 Vector3 flatForward = target.forward;
220 flatForward.y = 0f;
221
222 if (flatForward.sqrMagnitude < 0.001f)
223 {
224 return Vector3.forward;
225 }
226
227 return flatForward.normalized;
228 }
229
230 /// <summary>
231 /// Calculates the desired camera position from target position, movement direction, offset, and look-ahead.
232 /// </summary>
233 /// <returns>Desired world-space camera position.</returns>
234 private Vector3 GetDesiredCameraPosition()
235 {
236 Vector3 flatDirection = GetCameraDirection();
237 float distance = Mathf.Abs(offset.z);
238 Vector3 right = Vector3.Cross(Vector3.up, flatDirection).normalized;
239 Vector3 lookAhead = flatDirection * lookAheadDistance;
240 return target.position - flatDirection * distance + Vector3.up * offset.y + right * offset.x + lookAhead;
241 }
242
243 #endregion
