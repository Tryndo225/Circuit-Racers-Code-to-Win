3/// <summary>
4/// Third-person follow camera that tracks a target with a configurable local-space offset,
5/// smooth positional damping, and yaw aligned to the target with a fixed pitch angle.

23 /// <summary>Follow responsiveness (higher = snappier). Used as 1 / followSpeed in SmoothDamp.</summary>

44 /// <summary>
45 /// Updates position using SmoothDamp toward target.TransformPoint(offset) and
46 /// rotates toward target yaw while enforcing a fixed pitch angle.
47 /// </summary>
48 private void Update()
49 {
50 if (target == null) return;
51
52 // Smooth follow toward local-space offset
53 Vector3 desiredPosition = target.TransformPoint(offset);
54 transform.position = Vector3.SmoothDamp(
55 transform.position,
56 desiredPosition,
57 ref velocity,
58 1f / followSpeed
59 );
60
61 // Compute yaw from target forward projected onto XZ plane
62 Vector3 flatForward = target.forward;
63 flatForward.y = 0f;
64
65 if (flatForward.sqrMagnitude > 0.001f)
66 {
67 Quaternion yawRotation = Quaternion.LookRotation(flatForward);
68
69 // Apply fixed pitch on top of yaw
70 Vector3 euler = yawRotation.eulerAngles;
71 euler.x = fixedPitchAngle;
72 Quaternion finalRotation = Quaternion.Euler(euler);
73
74 // Smooth yaw toward target orientation
75 transform.rotation = Quaternion.Slerp(
76 transform.rotation,
77 finalRotation,
78 Time.fixedDeltaTime * yawSmoothness
79 );
80 }
81 }
82
83 #endregion
84
85 #region Public API
