63 RectInt pixelRect = new RectInt(0, 0, heightmapTexture.width, heightmapTexture.height);
64 Vector2 pixelSize = new Vector2(terrain.terrainData.size.x / heightmapTexture.width, terrain.terrainData.size.z / heightmapTexture.height);
65 BrushTransform brushXform = TerrainPaintUtility.CalculateBrushTransform(terrain, new Vector2(0.5f, 0.5f), terrain.terrainData.size.x, 0.0f);
66
67 // we want to build a quad mesh, with one vertex for each pixel in the heightmap
68 // i.e. a 3x3 heightmap would create a mesh that looks like this:
69 //
70 // +-+-+
71 // |\|\|
72 // +-+-+
73 // |\|\|
74 // +-+-+
75 //
76 int quadsX = pixelRect.width+1;
77 int quadsY = pixelRect.height+1;
78 int vertexCount = quadsX * quadsY * (2 * 3); // two triangles (2 * 3 vertices) per quad
79
80 // issue: the 'int vertexID' in the shader is often stored in an fp32
81 // which can only represent exact integers up to 16777216 ~== 6 * 1672^2
82 // once we have more than 16777216 vertices, the vertexIDs start skipping odd values, resulting in missing triangles
83 // the solution is to reduce vertex count by halving our mesh resolution before we hit that point
84 const int kMaxFP32Int = 16777216;
85 int vertSkip = 1;
86 while (vertexCount > kMaxFP32Int / 2) // in practice we want to stay well below 16 million verts, for perf sanity
87 {
88 quadsX = (quadsX + 1) / 2;
89 quadsY = (quadsY + 1) / 2;
90 vertexCount = quadsX * quadsY * (2 * 3);
91 vertSkip *= 2;
92 }
93
94 // this is used to tessellate the quad mesh (from within the vertex shader)
95 heatmapMat.SetVector("_QuadRez", new Vector4(quadsX, quadsY, vertexCount, vertSkip));
96
97 // paint context pixels to heightmap uv: uv = (pixels + 0.5) / width
98 float invWidth = 1.0f / heightmapTexture.width;
99 float invHeight = 1.0f / heightmapTexture.height;
100 heatmapMat.SetVector("_HeightmapUV_PCPixelsX", new Vector4(invWidth, 0.0f, 0.0f, 0.0f));
101 heatmapMat.SetVector("_HeightmapUV_PCPixelsY", new Vector4(0.0f, invHeight, 0.0f, 0.0f));
102 heatmapMat.SetVector("_HeightmapUV_Offset", new Vector4(0.5f * invWidth, 0.5f * invHeight, 0.0f, 0.0f));
103
104 heatmapMat.SetTexture("_Heightmap", heightmapTexture);
105 heatmapMat.SetTexture("_NormalMap", terrain.normalmapTexture);
106
107 // paint context pixels to object (terrain) position
108 // objectPos.x = scaleX * pcPixels.x + heightmapRect.xMin * scaleX
109 // objectPos.y = scaleY * H
110 // objectPos.z = scaleZ * pcPixels.y + heightmapRect.yMin * scaleZ
111 float scaleX = pixelSize.x;
112 float scaleY = (terrain.terrainData.heightmapScale.y) / kNormalizedHeightScale;
113 float scaleZ = pixelSize.y;
114 heatmapMat.SetVector("_ObjectPos_PCPixelsX", new Vector4(scaleX, 0.0f, 0.0f, 0.0f));
115 heatmapMat.SetVector("_ObjectPos_HeightMapSample", new Vector4(0.0f, scaleY, 0.0f, 0.0f));
116 heatmapMat.SetVector("_ObjectPos_PCPixelsY", new Vector4(0.0f, 0.0f, scaleZ, 0.0f));
117 //Note slightly offset, so raise up so it doesn't clip through terrain
118 heatmapMat.SetVector("_ObjectPos_Offset", new Vector4((pixelRect.xMin * scaleX), 3f , (pixelRect.yMin * scaleZ) + (pixelSize.y * 0.0f) , 1.0f));
119
120 // paint context origin in terrain space
121 // (note this is the UV space origin and size, not the mesh origin & size)
122 float pcOriginX = pixelRect.xMin * pixelSize.x;
123 float pcOriginZ = pixelRect.yMin * pixelSize.y;
124 float pcSizeX = pixelSize.x;
125 float pcSizeZ = pixelSize.y;
126
127 Vector2 scaleU = pcSizeX * brushXform.targetX;
128 Vector2 scaleV = pcSizeZ * brushXform.targetY;
129 Vector2 offset = brushXform.targetOrigin + pcOriginX * brushXform.targetX + pcOriginZ * brushXform.targetY;
130 heatmapMat.SetVector("_BrushUV_PCPixelsX", new Vector4(scaleU.x, scaleU.y, 0.0f, 0.0f));
131 heatmapMat.SetVector("_BrushUV_PCPixelsY", new Vector4(scaleV.x, scaleV.y, 0.0f, 0.0f));
132 heatmapMat.SetVector("_BrushUV_Offset", new Vector4(offset.x, offset.y, 0.0f, 1.0f));
133 heatmapMat.SetTexture("_LayerMaskTex", alphaMap);
134
135 heatmapMat.SetFloat("_ContourOn", contour ? 1 : 0);
136 heatmapMat.SetFloat("_TilingOn", tilingPreview ? 1 : 0);
137 //Tiling in world-space
138 heatmapMat.SetVector("_LayerTiling", new Vector4(layer.diffuseTexture.width / terrain.terrainData.size.x / layer.tileSize.x, layer.diffuseTexture.height / terrain.terrainData.size.z / layer.tileSize.y, 0, 0));
139
