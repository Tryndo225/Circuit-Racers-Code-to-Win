5/// <summary>
6/// Persistent UI notification manager that displays one text popup at a time.
7/// </summary>
8/// <remarks>
9/// @ingroup ui
10/// @brief Creates short-lived notification popups, optionally with custom text color and timing.
11///
12/// The manager instantiates a <see cref="NotificationPopup"/> prefab as a child of itself.
13/// Showing a new notification clears the currently visible notification first, so only one popup
14/// can be visible at a time.
15///
16/// Timing uses unscaled time, so notifications continue to count down even when
17/// <see cref="Time.timeScale"/> is zero.
18/// </remarks>

124 /// <summary>
125 /// Shows a notification with optional text color and custom timing.
126 /// </summary>
127 /// <param name="message">Notification text to display.</param>
128 /// <param name="textColor">Optional text color. When null, the prefab's current/default text color is used.</param>
129 /// <param name="visibleTime">Time in seconds the notification remains fully visible.</param>
130 /// <param name="fadeTime">Time in seconds used to fade the notification out.</param>
131 /// <remarks>
132 /// Empty messages are ignored. If a popup is already visible, it is destroyed before the new popup is created.
133 /// </remarks>
134 public void Show(string message, Color? textColor, float visibleTime, float fadeTime)
135 {
136 if (string.IsNullOrEmpty(message))
137 {
138 return;
139 }
140
141 if (notificationPrefab == null)
142 {
143 Debug.LogWarning("NotificationManager has no notification prefab assigned.");
144 return;
145 }
146
147 ClearCurrent();
148
149 currentPopup = Instantiate(notificationPrefab, transform);
150 currentPopup.SetText(message);
151
152 if (textColor.HasValue)
153 {
154 currentPopup.SetTextColor(textColor.Value);
155 }
156
157 currentPopup.SetAlpha(1f);
158
159 currentCoroutine = StartCoroutine(NotificationCoroutine(visibleTime, fadeTime));
160 }
161
162 /// <summary>

174 /// <summary>
175 /// Stops the active notification coroutine and destroys the current popup instance.
176 /// </summary>
177 private void ClearCurrent()
178 {
179 if (currentCoroutine != null)
180 {
181 StopCoroutine(currentCoroutine);
182 currentCoroutine = null;

192 /// <summary>
193 /// Keeps the notification visible, fades it out, and then clears it.
194 /// </summary>
195 /// <param name="visibleTime">Time in seconds the popup remains fully visible.</param>
196 /// <param name="fadeTime">Time in seconds used for fade-out.</param>
197 /// <returns>Coroutine enumerator.</returns>
198 /// <remarks>
199 /// Uses <see cref="Time.unscaledTime"/> so notification timing is independent of gameplay time scale.
200 /// </remarks>
201 private IEnumerator NotificationCoroutine(float visibleTime, float fadeTime)
202 {
203 float startTime = Time.unscaledTime;
204
205 while (Time.unscaledTime - startTime < visibleTime)
206 {
207 yield return null;
208 }
209
210 startTime = Time.unscaledTime;
211
212 while (Time.unscaledTime - startTime < fadeTime)
213 {
214 if (currentPopup == null)
215 {
216 yield break;
217 }
218
219 float t = Mathf.Clamp01((Time.unscaledTime - startTime) / fadeTime);
220 currentPopup.SetAlpha(Mathf.Lerp(1f, 0f, t));
221
222 yield return null;
223 }
224
225 ClearCurrent();
226 }
227
228 #endregion
