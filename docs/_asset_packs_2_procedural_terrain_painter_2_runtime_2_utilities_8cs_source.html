49 public static Bounds RecalculateBounds(Terrain[] terrains)
50 {
51 Bounds bounds = new Bounds(Vector3.zero, Vector3.one);
52
53 Vector3 minSum = Vector3.one * Mathf.Infinity;
54 Vector3 maxSum = Vector3.one * Mathf.NegativeInfinity;
55 Vector3 min = Vector3.zero;
56 Vector3 max = Vector3.zero;
57
58 foreach (Terrain terrain in terrains)
59 {
60 if (terrain == null || terrain.terrainData == null)
61 {
62 throw new Exception("Failed to calculate total terrain bounds, one or more terrain objects are missing or broken");
63 }
64
65 //Min/max bounds corners in world-space
66 min = terrain.GetPosition();
67 max = terrain.GetPosition() + terrain.terrainData.size;
68
69 minSum.x = Mathf.Min(minSum.x, min.x);
70 minSum.y = Mathf.Min(minSum.y, min.y);
71 minSum.z = Mathf.Min(minSum.z, min.z);
72
73 //Must handle each axis separately, terrain may be further away, but not necessarily higher
74 maxSum.x = Mathf.Max(maxSum.x, max.x);
75 maxSum.y = Mathf.Max(maxSum.y, max.y);
76 maxSum.z = Mathf.Max(maxSum.z, max.z);
77 }
78
79 bounds.SetMinMax(minSum, maxSum);
80
81 //Increase bounds height for flat terrains
82 if (bounds.size.y < 1f)
83 {
84 bounds.Encapsulate(new Vector3(bounds.center.x, bounds.center.y + 1f, bounds.center.z));
85 }
86
87 return bounds;
88 }
