1/// <summary>
2/// Utility that scans a generated <see cref="LevelMap"/> road and places checkpoint tiles on long straight segments.
3/// </summary>
4/// <remarks>
5/// @ingroup level_gen
6/// @brief Converts viable road tiles into checkpoint tiles based on straight-segment length.
7///
8/// Traversal starts at <see cref="LevelMap.StartPoint"/> and follows connected road tiles until no
9/// forward step is possible. A checkpoint is placed once per sufficiently long straight segment, roughly
10/// at the middle of that straight. Start and finish tiles are never converted to checkpoints.
11///
12/// Tile values used by this utility come from <see cref="LevelMap.LevelTileTypes"/>:
13/// - <see cref="LevelMap.LevelTileTypes.Track"/>: road tile.
14/// - <see cref="LevelMap.LevelTileTypes.CP"/>: generated checkpoint tile.
15///
16/// Threading:
17/// - This mutates <see cref="LevelMap.Tiles"/> directly and should be called from code that owns the level data.
18/// </remarks>
19public static class LevelCheckPointMaker
20{
21 /// <summary>
22 /// Integer tile value used for road/track cells.
23 /// </summary>
24 private const int TrackTile = (int)LevelMap.LevelTileTypes.Track;
25
26 /// <summary>
27 /// Integer tile value used for checkpoint cells.
28 /// </summary>
29 private const int CheckpointTile = (int)LevelMap.LevelTileTypes.CP;
30
31 /// <summary>
32 /// Sentinel coordinate used to indicate an invalid direction or position.
33 /// </summary>
34 private static readonly Coordinates Invalid = new Coordinates(-1, -1);
35
36 /// <summary>
37 /// Walks the road network in <paramref name="levelMap"/> and converts midpoint tiles of sufficiently long
38 /// straight segments into checkpoint tiles.
39 /// </summary>
40 /// <param name="levelMap">Target level definition containing the road grid.</param>
41 /// <param name="minStraightLengthForCheckPoint">
42 /// Minimum number of consecutive road tiles required before a checkpoint is placed on a straight segment.
43 /// </param>
44 /// <remarks>
45 /// @ingroup level_gen
46 ///
47 /// Preconditions:
48 /// - <paramref name="levelMap"/> is non-null.
49 /// - <c>levelMap.Tiles</c> is a valid 2D grid.
50 /// - Road cells are marked with <see cref="LevelMap.LevelTileTypes.Track"/> and checkpoint cells with <see cref="LevelMap.LevelTileTypes.CP"/>.
51 ///
52 /// Effects:
53 /// - Some road cells may be converted to checkpoint cells with <see cref="LevelMap.LevelTileTypes.CP"/>.
54 /// - <see cref="LevelMap.CheckpointCountPerLap"/> is incremented for each generated checkpoint.
55 /// - Start and finish cells are not modified.
56 /// </remarks>
57 public static void GenerateCheckPoints(LevelMap levelMap, int minStraightLengthForCheckPoint = 4)
58 {
59 Coordinates position = levelMap.StartPoint;
60 Coordinates lastVisited = levelMap.StartPoint;
61
62 Coordinates direction = Invalid;
63 Coordinates straightStart = Invalid;
64 int straightLength = 0;
65
66 while (true)
67 {
68 Coordinates newDirection = Step(levelMap, ref position, ref lastVisited);
69 if (newDirection == Invalid)
70 break;
71
72 if (direction == Invalid)
73 {
74 direction = newDirection;
75 continue;
76 }
77
78 if (newDirection == direction)
79 {
80 straightLength++;
81 continue;
82 }
83
84 PlaceCheckpointIfPossible(levelMap, straightStart, direction, straightLength, minStraightLengthForCheckPoint);
85
86 direction = newDirection;
87 straightStart = position - direction;
88 straightLength = 2;
89 }
90 }

92 /// <summary>
93 /// Places a checkpoint on a straight segment when the segment is long enough and the target tile is valid.
94 /// </summary>
95 /// <param name="levelMap">Level map whose tile grid should be modified.</param>
96 /// <param name="straightStart">First coordinate of the straight segment.</param>
97 /// <param name="direction">Direction of the straight segment.</param>
98 /// <param name="straightLength">Length of the straight segment in tiles.</param>
99 /// <param name="minStraightLengthForCheckPoint">Minimum straight length required to place a checkpoint.</param>
100 /// <remarks>
101 /// The checkpoint is placed at the middle of the straight segment. The method skips invalid starts,
102 /// out-of-bounds coordinates, and coordinates equal to the level start or finish point.
103 /// </remarks>
104 private static void PlaceCheckpointIfPossible(LevelMap levelMap, Coordinates straightStart, Coordinates direction, int straightLength, int minStraightLengthForCheckPoint)
105 {
106 if (straightLength >= minStraightLengthForCheckPoint && straightStart != Invalid)
107 {
108 int midOffset = (straightLength - 1) / 2;
109
110 Coordinates checkpoint = straightStart + midOffset * direction;
111
112 if (checkpoint != levelMap.StartPoint && checkpoint != levelMap.FinishPoint && levelMap.Tiles.InBounds(checkpoint.X, checkpoint.Y))
113 {
114 levelMap.Tiles.At(checkpoint) = CheckpointTile;
115 ++levelMap.CheckpointCountPerLap;
116 }
117 }
118 }
119
120 /// <summary>
121 /// Advances one step along the road from <paramref name="position"/> to a valid adjacent road tile.
122 /// </summary>
123 /// <param name="levelMap">Level map containing the road grid.</param>
124 /// <param name="position">Current traversal position; updated when a valid step is found.</param>
125 /// <param name="lastVisited">Previous traversal position used to avoid immediate backtracking; updated on success.</param>
126 /// <returns>
127 /// Direction offset used for the step if a move was made; otherwise <see cref="Invalid"/> when no forward step is possible.
128 /// </returns>
129 /// <remarks>
130 /// The method checks the cardinal directions from <see cref="LevelMap.CardinalDirections"/>.
131 /// It only steps onto tiles marked with <see cref="LevelMap.LevelTileTypes.Track"/> or <see cref="LevelMap.LevelTileTypes.CP"/>, avoids the immediately previous tile,
132 /// and does not step onto <see cref="LevelMap.FinishPoint"/>.
133 /// </remarks>
134 private static Coordinates Step(LevelMap levelMap, ref Coordinates position, ref Coordinates lastVisited)
135 {
136 foreach (var offset in LevelMap.CardinalDirections)
137 {
138 var next = position + offset;
139
140 if (!levelMap.Tiles.InBounds(next.X, next.Y))
141 continue;
142
143 if ((levelMap.Tiles[next.X, next.Y] == TrackTile || levelMap.Tiles[next.X, next.Y] == CheckpointTile) && next != lastVisited && next != levelMap.FinishPoint)
144 {
145 lastVisited = position;
146 position = next;
147 return offset;
148 }
149 }
150
151 return Invalid;
152 }
