3/// <summary>
4/// Utility that scans a generated <see cref="LevelMap"/> road (tile value 1) and stamps
5/// checkpoint tiles (-2) along sufficiently long straight segments.
6/// </summary>
7/// <remarks>
8/// @ingroup level_gen
9/// Traversal starts at <see cref="LevelMap.StartPoint"/> and follows the road until no
10/// forward step is possible. A checkpoint is placed once per straight segment whose
11/// length meets <paramref name="minStraightCountForChackPoint"/>; the checkpoint is
12/// positioned roughly at the middle of that straight. Start/finish tiles are never
13/// converted to checkpoints.
14/// @thread Unity main thread (mutates <see cref="LevelMap.Tiles"/>).
15/// </remarks>
16public static class LevelCheckPointMaker
17{
18 /// <summary>
19 /// Sentinel coordinate used to indicate an invalid direction/position.
20 /// </summary>
21 private static readonly Coordinates Invalid = new Coordinates(-1, -1);
22
23 /// <summary>
24 /// 4-neighbor step offsets (right, left, up, down) for road traversal.
25 /// </summary>
26 private static readonly Coordinates[] offsets = new Coordinates[]
27 {
28 new Coordinates(1, 0),
29 new Coordinates(-1, 0),
30 new Coordinates(0, 1),
31 new Coordinates(0, -1)
32 };
33
34 /// <summary>
35 /// Walks the road network in <paramref name="levelMap"/> and converts the midpoint of each
36 /// sufficiently long straight segment into a checkpoint tile (-2).
37 /// </summary>
38 /// <param name="levelMap">Target level definition containing the road grid.</param>
39 /// <param name="minStraightCountForChackPoint">
40 /// Minimum number of consecutive road tiles (including the first) required
41 /// before a checkpoint is placed on that straight segment. (Typo preserved.)
42 /// </param>
43 /// <remarks>
44 /// @ingroup level_gen
45 /// Preconditions:
46 /// <list type="bullet">
47 /// <item><description><paramref name="levelMap"/> is non-null.</description></item>
48 /// <item><description><c>levelMap.Tiles</c> is a valid 2D grid with road cells marked as 1.</description></item>
49 /// </list>
50 /// Postconditions:
51 /// <list type="bullet">
52 /// <item><description>Some road cells (value 1) may be converted to checkpoints (value -2).</description></item>
53 /// <item><description>Start/finish cells are not modified.</description></item>
54 /// </list>
55 /// Complexity: O(N) over visited road tiles (single pass with 4-neighborhood checks).
56 /// </remarks>
57 public static void GenerateCheckPoints(LevelMap levelMap, int minStraightCountForChackPoint = 3)
58 {
59 Coordinates position = levelMap.StartPoint;
60 Coordinates lastVisited = levelMap.StartPoint;
61
62 Coordinates direction = Invalid;
63 Coordinates newDirection = Invalid;
64 int straightCount = 0;
65 Coordinates straightStart = Invalid;
66 bool checkpointPlacedForThisStraight = true;
67
68 while (true)
69 {
70 newDirection = Step(levelMap, ref position, ref lastVisited);
71
72 if (newDirection == Invalid)
73 break;
74
75 if (newDirection == direction)
76 {
77 straightCount++;
78 }
79 else
80 {
81 if (direction != Invalid)
82 {
83 straightStart = position;
84 straightCount = 1;
85 checkpointPlacedForThisStraight = false;
86 }
87 direction = newDirection;
88 }
89
90 if (!checkpointPlacedForThisStraight && straightCount >= minStraightCountForChackPoint)
91 {
92 if (straightStart != levelMap.StartPoint && straightStart != levelMap.FinishPoint)
93 {
94 levelMap.Tiles.At(straightStart + ((straightCount - 1) / 2) * direction) = -2;
95 }
96 checkpointPlacedForThisStraight = true;
97 }
98 }
99 }
100
101 /// <summary>
102 /// Advances one step along the road from <paramref name="position"/> to an adjacent
103 /// road tile (value 1), avoiding immediate backtracking and skipping the finish tile.
104 /// </summary>
105 /// <param name="levelMap">Level map containing the road grid.</param>
106 /// <param name="position">Current traversal position; updated on success.</param>
107 /// <param name="lastVisited">Previous position used to prevent backtracking; updated on success.</param>
108 /// <returns>
109 /// The offset (direction) used for the step if a move was made; otherwise
110 /// <see cref="Invalid"/> when no forward step is possible.
111 /// </returns>
112 private static Coordinates Step(LevelMap levelMap, ref Coordinates position, ref Coordinates lastVisited)
113 {
114 foreach (var offset in offsets)
115 {
116 var next = position + offset;
117
118 if (!levelMap.Tiles.InBounds(next.X, next.Y))
119 continue;
120
121 if (levelMap.Tiles[next.X, next.Y] == 1 && next != lastVisited && next != levelMap.FinishPoint)
122 {
123 lastVisited = position;
124 position = next;
125 return offset;
126 }
127 }
128
129 return Invalid;
130 }
131}
